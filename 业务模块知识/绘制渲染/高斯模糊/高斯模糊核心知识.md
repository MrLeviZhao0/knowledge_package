// frameworks/base/libs/hwui/Blur.cpp (Android 4.4)
class BlurFilter {
public:
    static void applyGaussianBlur(Bitmap& src, Bitmap& dst, float radius) {
        // 使用RenderScript的ScriptIntrinsicBlur
        sp<RS> rs = new RS();
        sp<Allocation> input = Allocation::createFromBitmap(rs, src);
        sp<Allocation> output = Allocation::createFromBitmap(rs, dst);
        
        ScriptIntrinsicBlur::create(rs, Element::U8_4(rs))
            .setRadius(radius)
            .setInput(input)
            .forEach(output);
        
        output->copyTo(dst);
    }
};

### 1.2 SystemUI高斯模糊具体实现

#### 状态栏模糊实现
```java
// frameworks/base/packages/SystemUI/src/com/android/systemui/statusbar/BlurUtils.java
public class BlurUtils {
    private static final String BLUR_SHADER = 
        "uniform sampler2D uTexture;\n" +
        "uniform vec2 uTexelSize;\n" +
        "uniform float uRadius;\n" +
        "void main() {\n" +
        "    vec4 sum = vec4(0.0);\n" +
        "    float total = 0.0;\n" +
        "    for (float x = -uRadius; x <= uRadius; x++) {\n" +
        "        float weight = exp(-0.5 * pow(x / (uRadius/3.0), 2.0));\n" +
        "        sum += texture2D(uTexture, vTexCoord + vec2(x * uTexelSize.x, 0.0)) * weight;\n" +
        "        total += weight;\n" +
        "    }\n" +
        "    gl_FragColor = sum / total;\n" +
        "}\n";
    
    public static void applyStatusBarBlur(Surface surface, float radius) {
        // 获取当前屏幕截图
        Bitmap screenshot = takeScreenshot();
        
        // 创建模糊纹理
        GLuint blurTexture = createBlurTexture(screenshot);
        
        // 应用模糊着色器
        applyShaderProgram(BLUR_SHADER, blurTexture, radius);
        
        // 渲染到状态栏Surface
        renderToSurface(surface, blurTexture);
    }
}
```

```glsl
// frameworks/base/libs/hwui/shaders/blur.frag
#version 310 es
#extension GL_EXT_shader_pixel_local_storage : enable

layout(binding = 0) uniform sampler2D uTexture;
layout(location = 0) in vec2 vTexCoord;
layout(location = 0) out vec4 outColor;

uniform vec2 uTexelSize;
uniform float uRadius;
uniform int uQuality;  // 新增：质量等级控制

void main() {
    vec4 sum = vec4(0.0);
    float total = 0.0;
    
    // 根据质量等级调整采样步长
    float step = 1.0 / float(uQuality);
    
    for (float i = -uRadius; i <= uRadius; i += step) {
        float weight = exp(-0.5 * pow(i / (uRadius/3.0), 2.0));
        
        // 水平模糊
        vec2 offsetH = vec2(i * uTexelSize.x, 0.0);
        sum += texture(uTexture, vTexCoord + offsetH) * weight;
        
        // 垂直模糊
        vec2 offsetV = vec2(0.0, i * uTexelSize.y);
        sum += texture(uTexture, vTexCoord + offsetV) * weight;
        
        total += weight * 2.0;
    }
    
    outColor = sum / total;
}
```

## 4. 面试常考点

### 4.1 技术实现细节

#### 高斯核计算
**答案**：高斯核权重计算基于高斯函数：
```java
// 高斯函数公式：G(x) = (1/(σ√(2π))) * e^(-x²/(2σ²))
// 简化版本（忽略归一化常数）：G(x) = e^(-x²/(2σ²))

public float[] generateGaussianKernel(float sigma, int radius) {
    int size = radius * 2 + 1;
    float[] kernel = new float[size];
    float sum = 0.0f;
    
    for (int i = -radius; i <= radius; i++) {
        float x = (float) i;
        // 计算高斯权重
        kernel[i + radius] = (float) Math.exp(-(x * x) / (2 * sigma * sigma));
        sum += kernel[i + radius];
    }
    
    // 归一化，确保所有权重之和为1
    for (int i = 0; i < size; i++) {
        kernel[i] /= sum;
    }
    
    return kernel;
}
```

**sigma参数的意义**：
- **控制模糊程度**：sigma越大，模糊效果越强
- **影响核大小**：通常半径取3*sigma，覆盖99.7%的能量
- **质量与性能平衡**：sigma小则质量高但性能差，sigma大则性能好但质量低

#### 分离卷积原理
**答案**：分离卷积将2D卷积分解为两个1D卷积：
```java
// 传统2D卷积：O(n²)复杂度
for (int y = 0; y < height; y++) {
    for (int x = 0; x < width; x++) {
        for (int ky = -radius; ky <= radius; ky++) {
            for (int kx = -radius; kx <= radius; kx++) {
                // 2D卷积计算
            }
        }
    }
}

// 分离卷积：O(2n)复杂度
// 1. 水平卷积
for (int y = 0; y < height; y++) {
    for (int x = 0; x < width; x++) {
        for (int kx = -radius; kx <= radius; kx++) {
            // 水平方向卷积
        }
    }
}

// 2. 垂直卷积
for (int y = 0; y < height; y++) {
    for (int x = 0; x < width; x++) {
        for (int ky = -radius; ky <= radius; ky++) {
            // 垂直方向卷积
        }
    }
}
```

**性能提升原理**：
- **复杂度降低**：从O(n²)降到O(2n)，n为核半径
- **内存访问优化**：更好的缓存局部性
- **并行化机会**：两个1D卷积可以并行执行

#### 着色器优化
**答案**：减少纹理采样次数的方法：
1. **使用纹理缓存**：利用GPU的纹理缓存机制
2. **局部性采样**：按内存顺序访问纹理数据
3. **预计算权重**：在CPU端预计算高斯权重
4. **使用mipmap**：对模糊纹理使用mipmap优化

**优化着色器示例**：
```glsl
#version 310 es
precision mediump float;

uniform sampler2D uTexture;
uniform vec2 uTexelSize;
uniform float uRadius;

in vec2 vTexCoord;
out vec4 fragColor;

void main() {
    // 预计算权重，减少运行时计算
    float weights[9];
    float total = 0.0;
    
    for (int i = 0; i < 9; i++) {
        float x = float(i - 4);
        weights[i] = exp(-0.5 * (x * x) / (uRadius * uRadius));
        total += weights[i];
    }
    
    // 归一化
    for (int i = 0; i < 9; i++) {
        weights[i] /= total;
    }
    
    vec4 sum = vec4(0.0);
    
    // 优化采样顺序，提高缓存命中率
    for (int i = 0; i < 9; i++) {
        float offset = float(i - 4);
        vec2 sampleCoord = vTexCoord + vec2(offset * uTexelSize.x, 0.0);
        sum += texture(uTexture, sampleCoord) * weights[i];
    }
    
    fragColor = sum;
}
```

public class MultiWindowBlurManager {
    private Map<Window, BlurRegion> mBlurRegions = new HashMap<>();
    
    public void onWindowAdded(Window window) {
        // 检测重叠区域
        BlurRegion newRegion = calculateBlurRegion(window);
        
        for (BlurRegion existing : mBlurRegions.values()) {
            if (regionsOverlap(newRegion, existing)) {
                // 合并重叠区域
                mergeRegions(newRegion, existing);
            }
        }
        
        mBlurRegions.put(window, newRegion);

    public void optimizeBlurRegions() {
        // 使用空间划分算法优化区域管理

            partitioner.addRegion(region);
        }
        
        // 优化后的区域
        List<BlurRegion> optimized = partitioner.optimize();
        
        // 应用优化
        applyOptimizedRegions(optimized);
    }
}
