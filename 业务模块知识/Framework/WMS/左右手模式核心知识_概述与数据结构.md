# 左右手模式核心知识 - 概述与数据结构

## 1. 左右手模式概述

左右手模式是Android系统中的一项重要功能，它允许用户根据使用习惯调整屏幕布局，提高单手操作体验。该功能主要通过窗口缩放和布局调整来实现，是WindowManagerService中的一个重要模块。

### 1.1 左右手模式的作用与重要性

左右手模式在Android系统中提供以下价值：

- **单手操作便利**：减少大屏设备单手操作难度
- **用户体验优化**：满足不同用户的操作习惯
- **无障碍支持**：为有特殊需求的用户提供便利
- **差异化特性**：作为厂商定制化的重要功能点

### 1.2 左右手模式的实现方式

不同厂商对左右手模式的实现方式有所不同，主要包括：

| 实现方式 | 技术原理 | 优点 | 缺点 |
|----------|----------|------|------|
| 窗口整体缩放 | 调整窗口大小和位置 | 实现简单，兼容性好 | 可能影响应用显示效果 |
| 内容布局调整 | 修改应用内部布局 | 保持应用完整性 | 需要应用适配 |
| 系统级UI调整 | 调整系统UI元素位置 | 对应用透明 | 实现复杂 |
| 混合模式 | 结合多种方式 | 平衡效果和兼容性 | 实现最复杂 |

### 1.3 左右手模式与WindowManagerService的关系

左右手模式是WindowManagerService中的一个重要功能模块：

```
+-------------------+
|   WindowManagerService |
+-------------------+
|   左右手模式管理器    |

+-------------------+

+-------------------+
```

## 4. 左右手模式与WindowManagerService集成

### 4.1 WMS中的左右手模式集成

```java
// WindowManagerService中的左右手模式集成
public class WindowManagerService {
    // 左右手模式管理器
    private LeftRightModeManager mLeftRightModeManager;
    
    // 窗口缩放控制器
    private WindowScaleController mScaleController;
    
    // 触摸事件处理器
    private TouchEventHandler mTouchHandler;
    
    // 初始化左右手模式
    private void initializeLeftRightMode() {
        // 创建管理器
        mLeftRightModeManager = new LeftRightModeManager();
        
        // 创建控制器
        mScaleController = new WindowScaleController();
        mTouchHandler = new TouchEventHandler();
        
        // 设置监听器
        mLeftRightModeManager.addListener(new LeftRightModeListener() {
            @Override
            public void onModeChanged(LeftRightMode oldMode, LeftRightMode newMode) {
                handleModeChange(oldMode, newMode);
            }
            
            @Override
            public void onScaleFactorChanged(float oldFactor, float newFactor) {
                handleScaleFactorChange(oldFactor, newFactor);
            }
            
            @Override
            public void onOffsetChanged(int oldX, int oldY, int newX, int newY) {
                handleOffsetChange(oldX, oldY, newX, newY);
            }
        });
    }
    
    // 处理模式变化
    private void handleModeChange(LeftRightMode oldMode, LeftRightMode newMode) {
        // 更新所有窗口
        for (WindowState win : mWindowList) {
            updateWindowForMode(win, newMode);
        }
        
        // 更新触摸事件处理
        updateTouchHandling(newMode);
        
        // 通知系统UI
        notifySystemUIOfModeChange(newMode);
    }
    
    // 更新窗口以适应新模式
    private void updateWindowForMode(WindowState win, LeftRightMode mode) {
        // 获取窗口缩放参数
        WindowScaleParams params = calculateScaleParams(win, mode);
        
        // 应用缩放
        if (params != null) {
            mScaleController.startScale(win);
            mScaleController.setScaleParams(win, params);
        }
    }
    
    // 更新触摸事件处理
    private void updateTouchHandling(LeftRightMode mode) {
        // 计算触摸转换参数
        TouchTransformParams params = calculateTouchTransformParams(mode);
        
        // 应用转换
        mTouchHandler.setTransformParams(params);
    }
}
```

```java
// 窗口布局调整
public class WindowLayoutAdjuster {
    // 布局参数
    private WindowManager.LayoutParams mLayoutParams;
    private Rect mDisplayFrame;
    private Rect mParentFrame;
    private Rect mContentFrame;
    
    // 左右手模式参数
    private LeftRightMode mMode;
    private float mScaleFactor;
    private int mOffsetX;
    private int mOffsetY;
    
    // 调整窗口布局
    public void adjustWindowLayout(WindowState win, LeftRightMode mode) {
        // 获取原始布局参数
        mLayoutParams = win.mAttrs;
        mDisplayFrame = win.mDisplayFrame;
        mParentFrame = win.mParentFrame;
        mContentFrame = win.mContentFrame;
        
        // 设置模式参数
        mMode = mode;
        calculateModeParameters();
        
        // 调整布局
        adjustDisplayFrame();
        adjustParentFrame();
        adjustContentFrame();
        
        // 应用调整后的布局
        win.mDisplayFrame = mDisplayFrame;
        win.mParentFrame = mParentFrame;
        win.mContentFrame = mContentFrame;
    }
    
    // 计算模式参数
    private void calculateModeParameters() {
        switch (mMode) {
            case LEFT_HAND:
                mScaleFactor = 0.85f;
                mOffsetX = 0;
                mOffsetY = 0;
                break;
            case RIGHT_HAND:
                mScaleFactor = 0.85f;
                mOffsetX = (int)(mDisplayFrame.width() * (1 - mScaleFactor));
                mOffsetY = 0;
                break;
            case NONE:
            default:
                mScaleFactor = 1.0f;
                mOffsetX = 0;
                mOffsetY = 0;
                break;
        }
    }
    
    // 调整显示区域
    private void adjustDisplayFrame() {
        int width = (int)(mDisplayFrame.width() * mScaleFactor);
        int height = (int)(mDisplayFrame.height() * mScaleFactor);
        int left = mDisplayFrame.left + mOffsetX;
        int top = mDisplayFrame.top + mOffsetY;
        int right = left + width;
        int bottom = top + height;
        
        mDisplayFrame.set(left, top, right, bottom);
    }
    
    // 调整父窗口区域
    private void adjustParentFrame() {
        int width = (int)(mParentFrame.width() * mScaleFactor);
        int height = (int)(mParentFrame.height() * mScaleFactor);
        int left = mParentFrame.left + mOffsetX;
        int top = mParentFrame.top + mOffsetY;
        int right = left + width;
        int bottom = top + height;
        
        mParentFrame.set(left, top, right, bottom);
    }
    
    // 调整内容区域
    private void adjustContentFrame() {
        int width = (int)(mContentFrame.width() * mScaleFactor);
        int height = (int)(mContentFrame.height() * mScaleFactor);

        int top = mContentFrame.top + mOffsetY;

        int bottom = top + height;
```

### 5.1 窗口缩放实现

```java
// 窗口缩放实现
public class WindowScaleImpl {
    // 缩放矩阵
    private Matrix mScaleMatrix;
    private Matrix mInverseMatrix;
    
    // 缩放参数
    private float mScaleFactor;
    private int mPivotX;
    private int mPivotY;
    
    // 初始化缩放
    public void initializeScale(float factor, int pivotX, int pivotY) {
        mScaleFactor = factor;
        mPivotX = pivotX;
        mPivotY = pivotY;
        
        // 创建缩放矩阵
        mScaleMatrix = new Matrix();
        mScaleMatrix.postScale(factor, factor, pivotX, pivotY);
        
        // 创建逆矩阵
        mInverseMatrix = new Matrix();
        mScaleMatrix.invert(mInverseMatrix);
    }
    
    // 应用缩放到窗口
    public void applyScale(WindowState win) {
        // 获取窗口表面
        SurfaceControl surface = win.mSurfaceControl;
        
        // 应用缩放变换
        SurfaceControl.Transaction transaction = new SurfaceControl.Transaction();
        transaction.setMatrix(surface, mScaleMatrix, mInverseMatrix);
        transaction.apply();
    }
    
    // 转换坐标
    public Point transformPoint(Point point) {
        float[] coords = {point.x, point.y};
        mScaleMatrix.mapPoints(coords);
        return new Point((int)coords[0], (int)coords[1]);
    }
    
    // 逆转换坐标
    public Point inverseTransformPoint(Point point) {
        float[] coords = {point.x, point.y};
        mInverseMatrix.mapPoints(coords);
        return new Point((int)coords[0], (int)coords[1]);
    }
    
    // 转换区域
    public Rect transformRect(Rect rect) {
        Rect result = new Rect(rect);
        mScaleMatrix.mapRect(result);
        return result;
    }
    
    // 逆转换区域
    public Rect inverseTransformRect(Rect rect) {
        Rect result = new Rect(rect);
        mInverseMatrix.mapRect(result);
        return result;
    }
}
```

### 5.2 触摸事件转换实现

```java
// 触摸事件转换实现
public class TouchEventTransformImpl {
    // 转换矩阵
    private Matrix mTransformMatrix;
    private Matrix mInverseMatrix;
    
    // 转换区域
    private Rect mSourceArea;
    private Rect mTargetArea;
    
    // 初始化转换
    public void initializeTransform(Rect sourceArea, Rect targetArea) {
        mSourceArea = new Rect(sourceArea);
        mTargetArea = new Rect(targetArea);
        
        // 创建转换矩阵
        mTransformMatrix = new Matrix();
        mTransformMatrix.setRectToRect(new RectF(sourceArea), new RectF(targetArea), 
                                      Matrix.ScaleToFit.FILL);
        
        // 创建逆矩阵
        mInverseMatrix = new Matrix();
        mTransformMatrix.invert(mInverseMatrix);
    }
    
    // 转换触摸事件
    public MotionEvent transformTouchEvent(MotionEvent event) {
        // 创建新的事件
        MotionEvent transformedEvent = MotionEvent.obtain(event);
        
        // 转换坐标
        float[] coords = new float[event.getPointerCount() * 2];
        for (int i = 0; i < event.getPointerCount(); i++) {
            coords[i * 2] = event.getX(i);
            coords[i * 2 + 1] = event.getY(i);
        }
        
        // 应用转换矩阵
        mTransformMatrix.mapPoints(coords);
        
        // 设置转换后的坐标
        for (int i = 0; i < event.getPointerCount(); i++) {
            transformedEvent.setLocation(i, coords[i * 2], coords[i * 2 + 1]);
        }
        
        return transformedEvent;
    }
    
    // 逆转换触摸事件
    public MotionEvent inverseTransformTouchEvent(MotionEvent event) {
        // 创建新的事件
        MotionEvent transformedEvent = MotionEvent.obtain(event);
        
        // 转换坐标
        float[] coords = new float[event.getPointerCount() * 2];
        for (int i = 0; i < event.getPointerCount(); i++) {
            coords[i * 2] = event.getX(i);
            coords[i * 2 + 1] = event.getY(i);
        }
        
        // 应用逆转换矩阵
        mInverseMatrix.mapPoints(coords);
        
        // 设置转换后的坐标
        for (int i = 0; i < event.getPointerCount(); i++) {
            transformedEvent.setLocation(i, coords[i * 2], coords[i * 2 + 1]);
        }
        
        return transformedEvent;
    }
    
    // 检查点是否在源区域内
    public boolean isPointInSourceArea(float x, float y) {
        return mSourceArea.contains((int)x, (int)y);
    }
    
    // 检查点是否在目标区域内
    public boolean isPointInTargetArea(float x, float y) {
        return mTargetArea.contains((int)x, (int)y);
    }
```

```java
public class LeftRightModeAnimator {
    // 动画参数
    private long mDefaultDuration = 300;
    private Interpolator mDefaultInterpolator = new AccelerateDecelerateInterpolator();

    private ArrayList<ScaleAnimator> mAnimatorPool;

    // 获取动画器
    public ScaleAnimator getAnimator() {
        ScaleAnimator animator;
        
        // 从池中获取
        if (!mAnimatorPool.isEmpty()) {
            animator = mAnimatorPool.remove(mAnimatorPool.size() - 1);
        } else {
            // 创建新的动画器
            animator = new ScaleAnimator();
        }
        
        return animator;
    }
    
    // 回收动画器
    public void recycleAnimator(ScaleAnimator animator) {
        if (mAnimatorPool.size() < mMaxPoolSize) {
            animator.reset();
            mAnimatorPool.add(animator);
        }
    }
    
    // 批量动画
    public void animateScaleBatch(ArrayList<WindowState> windows, 
                                 LeftRightMode fromMode, LeftRightMode toMode) {
        // 创建批量动画
        ScaleBatchAnimator batchAnimator = new ScaleBatchAnimator();
        
        // 添加窗口到批量动画
        for (WindowState win : windows) {
            WindowScaleParams fromParams = calculateScaleParams(win, fromMode);
            WindowScaleParams toParams = calculateScaleParams(win, toMode);
            batchAnimator.addWindow(win, fromParams, toParams);
        }
        
        // 启动批量动画
        batchAnimator.start();
    }
    
    // 批量动画器
    private class ScaleBatchAnimator {
        private ArrayList<WindowState> mWindows;
        private ArrayList<WindowScaleParams> mFromParams;
        private ArrayList<WindowScaleParams> mToParams;
        private ValueAnimator mAnimator;
        
        public ScaleBatchAnimator() {
            mWindows = new ArrayList<>();
            mFromParams = new ArrayList<>();
            mToParams = new ArrayList<>();
            
            mAnimator = ValueAnimator.ofFloat(0f, 1f);
            mAnimator.setDuration(mDefaultDuration);
            mAnimator.setInterpolator(mDefaultInterpolator);
            mAnimator.addUpdateListener(new ValueAnimator.AnimatorUpdateListener() {
                @Override
                public void onAnimationUpdate(ValueAnimator animation) {
                    float fraction = (Float) animation.getAnimatedValue();
                    updateWindows(fraction);
                }
            });
        }
        
        public void addWindow(WindowState win, WindowScaleParams fromParams, 
                            WindowScaleParams toParams) {
            mWindows.add(win);
            mFromParams.add(fromParams);
            mToParams.add(toParams);
        }
        
        public void start() {
            mAnimator.start();
        }
        
        private void updateWindows(float fraction) {
            for (int i = 0; i < mWindows.size(); i++) {
                WindowState win = mWindows.get(i);
                WindowScaleParams fromParams = mFromParams.get(i);
                WindowScaleParams toParams = mToParams.get(i);
                
                // 计算当前参数
                WindowScaleParams currentParams = interpolateParams(fromParams, toParams, fraction);
                
                // 应用参数
                applyScaleParams(win, currentParams);
            }
        }
        
        private WindowScaleParams interpolateParams(WindowScaleParams from, 
                                                   WindowScaleParams to, float fraction) {
            WindowScaleParams result = new WindowScaleParams();
            
            // 插值缩放因子
            result.setScaleFactor(interpolate(from.getScaleFactor(), to.getScaleFactor(), fraction));
            
            // 插值偏移
            result.setOffsetX((int)interpolate(from.getOffsetX(), to.getOffsetX(), fraction));
            result.setOffsetY((int)interpolate(from.getOffsetY(), to.getOffsetY(), fraction));
            
            // 插值中心点
            result.setPivotX((int)interpolate(from.getPivotX(), to.getPivotX(), fraction));
            result.setPivotY((int)interpolate(from.getPivotY(), to.getPivotY(), fraction));
            
            return result;
        }
        
        private float interpolate(float from, float to, float fraction) {
            return from + (to - from) * fraction;
        }
    }
```
