# 客户端兜底策略

## 1. 兜底策略概述

### 1.1 兜底策略定义与重要性

**兜底策略**是指在应用出现异常或崩溃时，通过预设的机制确保应用能够优雅降级或安全退出，避免对用户体验造成严重影响的技术方案。

**兜底策略的重要性：**
- **用户体验保障**：避免应用崩溃导致用户数据丢失
- **业务连续性**：确保核心功能在异常情况下仍能正常工作
- **问题诊断**：在崩溃前收集关键日志信息，便于问题定位
- **快速恢复**：通过预设机制快速恢复正常状态

### 1.2 兜底策略分类

```
客户端兜底策略
├── 启动阶段兜底
│   ├── Application初始化兜底
│   ├── 主Activity启动兜底
│   └── 关键服务启动兜底
├── 运行时兜底
│   ├── 异常捕获兜底
│   ├── 内存溢出兜底
│   └── 线程异常兜底
└── 崩溃处理兜底
    ├── 崩溃日志收集
    ├── 崩溃现场保护
    └── 安全退出机制
```

## 2. 启动阶段兜底策略

### 2.1 Application初始化兜底

#### 2.1.1 启动早期崩溃日志收集

**问题场景：**
应用在Application.onCreate()阶段发生崩溃，此时日志系统尚未初始化，无法收集崩溃信息。

**解决方案：**
```java
// 启动早期崩溃日志收集器
public class EarlyCrashLogger {
    
    private static final String EARLY_CRASH_LOG_FILE = "early_crash.log";
    private static boolean initialized = false;
    
    // 初始化早期日志系统
    public static void init(Context context) {
        if (initialized) return;
        
        // 设置未捕获异常处理器
        Thread.setDefaultUncaughtExceptionHandler(new EarlyCrashHandler(
            Thread.getDefaultUncaughtExceptionHandler()
        ));
        
        // 初始化文件日志系统
        initFileLogging(context);
        
        initialized = true;
        Log.d("EarlyCrashLogger", "Early crash logger initialized");
    }
    
    // 记录启动阶段关键信息
    public static void logStartupInfo(String info) {
        if (!initialized) return;
        
        writeToFile("STARTUP: " + info + " at " + System.currentTimeMillis());
    }
    
    // 早期崩溃处理器
    private static class EarlyCrashHandler implements Thread.UncaughtExceptionHandler {
        private final Thread.UncaughtExceptionHandler defaultHandler;
        
        public EarlyCrashHandler(Thread.UncaughtExceptionHandler defaultHandler) {
            this.defaultHandler = defaultHandler;
        }
        
        @Override
        public void uncaughtException(Thread thread, Throwable throwable) {
            // 记录崩溃信息到文件
            logCrashInfo(thread, throwable);
            
            // 调用默认处理器
            if (defaultHandler != null) {
                defaultHandler.uncaughtException(thread, throwable);
            }
        }
    }
    
    private static void logCrashInfo(Thread thread, Throwable throwable) {
        try {
            String crashInfo = buildCrashInfo(thread, throwable);
            writeToFile("CRASH: " + crashInfo);
            
            // 强制刷盘，确保日志写入
            forceFlush();
            
        } catch (Exception e) {
            // 兜底：即使日志记录失败也不影响崩溃处理
        }
    }
    
    private static String buildCrashInfo(Thread thread, Throwable throwable) {
        StringBuilder sb = new StringBuilder();
        sb.append("Thread: ").append(thread.getName()).append("\n");
        sb.append("Exception: ").append(throwable.getClass().getName()).append("\n");
        sb.append("Message: ").append(throwable.getMessage()).append("\n");
        
        // 记录堆栈信息（限制长度）
        StackTraceElement[] stackTrace = throwable.getStackTrace();
        for (int i = 0; i < Math.min(stackTrace.length, 20); i++) {
            sb.append("    at ").append(stackTrace[i]).append("\n");
        }
        
        return sb.toString();
    }
    
    private static void writeToFile(String content) {
        // 简化实现，实际应该使用更可靠的文件写入方式
        FileOutputStream fos = null;
        try {
            File file = new File(getLogDir(), EARLY_CRASH_LOG_FILE);
            fos = new FileOutputStream(file, true);
            fos.write((content + "\n").getBytes());
        } catch (Exception e) {
            // 忽略写入异常
        } finally {
            if (fos != null) {
                try { fos.close(); } catch (IOException e) {}
            }
        }
    }
}

// 在Application中使用
public class SafeApplication extends Application {
    
    @Override
    public void onCreate() {
        super.onCreate();
        
        // 第一步：初始化早期崩溃日志
        EarlyCrashLogger.init(this);
        EarlyCrashLogger.logStartupInfo("Application.onCreate started");
        
        try {
            // 第二步：分阶段初始化关键组件
            initCriticalComponents();
            
            // 第三步：初始化非关键组件（可延迟或失败）
            initNonCriticalComponents();
            
            EarlyCrashLogger.logStartupInfo("Application.onCreate completed");
            
        } catch (Throwable t) {
            // 记录初始化异常
            EarlyCrashLogger.logStartupInfo("Application.onCreate failed: " + t.getMessage());
            
            // 根据异常类型决定是否继续启动
            if (isCriticalFailure(t)) {
                // 关键失败，终止启动
                handleCriticalFailure(t);
            } else {
                // 非关键失败，继续启动但记录警告
                Log.w("SafeApplication", "Non-critical initialization failed", t);
            }
        }
    }
    
    private void initCriticalComponents() {
        // 1. 初始化崩溃监控（必须在最前面）
        initCrashMonitoring();
        
        // 2. 初始化基础服务
        initBasicServices();
        
        // 3. 初始化核心业务模块
        initCoreBusinessModules();
    }
    
    private void initNonCriticalComponents() {
        // 这些组件初始化失败不应该影响应用启动
        try {
            initAnalyticsSDK();
        } catch (Exception e) {
            Log.w("SafeApplication", "Analytics SDK init failed", e);
        }
        
        try {
            initPushService();
        } catch (Exception e) {
            Log.w("SafeApplication", "Push service init failed", e);
        }
    }
    
    private boolean isCriticalFailure(Throwable t) {
        // 判断是否为关键失败
        return t instanceof OutOfMemoryError || 
               t instanceof NoClassDefFoundError ||
               t instanceof VerifyError;
    }
    
    private void handleCriticalFailure(Throwable t) {
        // 记录关键失败信息
        EarlyCrashLogger.logStartupInfo("Critical failure handled: " + t.getMessage());
        
        // 可以尝试启动降级版本或显示错误页面
        startFallbackMode();
    }
}
```

#### 2.2.1 Activity启动异常处理

```java
// Activity启动兜底管理器
public class ActivityLaunchFallback {
    
    private static final Map<String, Long> activityLaunchAttempts = new ConcurrentHashMap<>();
    private static final int MAX_LAUNCH_ATTEMPTS = 3;
    private static final long ATTEMPT_RESET_INTERVAL = 5 * 60 * 1000; // 5分钟
    
    public static boolean shouldLaunchActivity(Context context, Class<? extends Activity> activityClass) {
        String activityName = activityClass.getName();
        
        // 检查启动次数限制
        if (exceededLaunchAttempts(activityName)) {
            Log.w("ActivityLaunchFallback", "Launch attempts exceeded for: " + activityName);
            return false;
        }
        
        // 检查Activity是否可用
        if (!isActivityAvailable(context, activityClass)) {
            Log.w("ActivityLaunchFallback", "Activity not available: " + activityName);
            return false;
        }
        
        // 记录启动尝试
        recordLaunchAttempt(activityName);
        return true;
    }
    
    public static void onActivityLaunchSuccess(Class<? extends Activity> activityClass) {
        // 清除启动尝试记录
        activityLaunchAttempts.remove(activityClass.getName());
    }
    
    public static void onActivityLaunchFailure(Class<? extends Activity> activityClass, Exception e) {
        String activityName = activityClass.getName();
        
        // 记录失败信息
        Log.e("ActivityLaunchFallback", "Activity launch failed: " + activityName, e);
        
        // 检查是否需要启用备用Activity
        if (shouldEnableFallbackActivity(activityClass)) {
            enableFallbackActivity(activityClass);
        }
    }
    
    private static boolean exceededLaunchAttempts(String activityName) {
        Long lastAttempt = activityLaunchAttempts.get(activityName);
        if (lastAttempt == null) {
            return false;
        }
        
        // 检查时间间隔
        if (System.currentTimeMillis() - lastAttempt > ATTEMPT_RESET_INTERVAL) {
            activityLaunchAttempts.remove(activityName);
            return false;
        }
        
        // 检查尝试次数
        Integer attemptCount = getAttemptCount(activityName);
        return attemptCount >= MAX_LAUNCH_ATTEMPTS;
    }
    
    private static boolean shouldEnableFallbackActivity(Class<? extends Activity> activityClass) {
        // 根据失败频率和影响范围决定是否启用备用Activity
        int failureCount = getRecentFailureCount(activityClass);
        boolean isCriticalActivity = isCriticalActivity(activityClass);
        
        return failureCount > 2 && isCriticalActivity;
    }
    
    private static void enableFallbackActivity(Class<? extends Activity> originalActivity) {
        // 启用备用Activity
        Class<? extends Activity> fallbackActivity = getFallbackActivity(originalActivity);
        
        if (fallbackActivity != null) {
            // 更新路由配置，将原始Activity路由到备用Activity
            updateActivityRouting(originalActivity, fallbackActivity);
            
            Log.i("ActivityLaunchFallback", "Enabled fallback activity: " + 
                  fallbackActivity.getName() + " for " + originalActivity.getName());
        }
    }
}

// 在BaseActivity中使用
public abstract class BaseActivity extends Activity {
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        // 检查启动条件
        if (!ActivityLaunchFallback.shouldLaunchActivity(this, getClass())) {
            // 启动失败，跳转到安全页面
            startSafetyActivity();
            finish();
            return;
        }
        
        super.onCreate(savedInstanceState);
        
        try {
            // 设置异常处理器
            Thread.currentThread().setUncaughtExceptionHandler(
                new ActivityExceptionHandler()
            );
            
            // 执行Activity初始化
            initializeActivity();
            
            // 标记启动成功
            ActivityLaunchFallback.onActivityLaunchSuccess(getClass());
            
        } catch (Exception e) {
            // 处理初始化异常
            handleInitializationException(e);
            ActivityLaunchFallback.onActivityLaunchFailure(getClass(), e);
        }
    }
    
    private class ActivityExceptionHandler implements Thread.UncaughtExceptionHandler {
        private final Thread.UncaughtExceptionHandler defaultHandler;
        
        public ActivityExceptionHandler() {
            this.defaultHandler = Thread.currentThread().getUncaughtExceptionHandler();
        }
        
        @Override
        public void uncaughtException(Thread thread, Throwable throwable) {
            // 记录Activity级别异常
            logActivityException(throwable);
            
            // 尝试恢复或安全退出
            if (canRecoverFromException(throwable)) {
                attemptRecovery(throwable);
            } else {
                // 无法恢复，安全退出
                safeExit();
            }
            
            // 调用默认处理器
            if (defaultHandler != null) {
                defaultHandler.uncaughtException(thread, throwable);
            }
        }
    }
}
```

### 3.1 内存溢出兜底

```java
// 内存溢出预警和兜底
public class MemoryOverflowProtector {
    
    private static final float MEMORY_THRESHOLD = 0.85f; // 85%内存使用阈值
    private static final long CHECK_INTERVAL = 5000; // 5秒检查间隔
    
    private Handler handler;
    private Runtime runtime;
    private boolean protectionEnabled = false;
    
    public MemoryOverflowProtector() {
        this.handler = new Handler(Looper.getMainLooper());
        this.runtime = Runtime.getRuntime();
        
        startMonitoring();
    }
    
    private void startMonitoring() {
        protectionEnabled = true;
        
        handler.post(new Runnable() {
            @Override
            public void run() {
                if (!protectionEnabled) return;
                
                checkMemoryUsage();
                handler.postDelayed(this, CHECK_INTERVAL);
            }
        });
    }
    
    private void checkMemoryUsage() {
        long maxMemory = runtime.maxMemory();
        long totalMemory = runtime.totalMemory();
        long freeMemory = runtime.freeMemory();
        long usedMemory = totalMemory - freeMemory;
        
        float memoryUsage = (float) usedMemory / maxMemory;
        
        if (memoryUsage > MEMORY_THRESHOLD) {
            // 内存使用超过阈值，触发保护措施
            onMemoryThresholdExceeded(memoryUsage);
        }
        
        // 记录内存使用情况
        recordMemoryUsage(usedMemory, maxMemory, memoryUsage);
    }
    
    private void onMemoryThresholdExceeded(float memoryUsage) {
        Log.w("MemoryOverflowProtector", "Memory usage exceeded threshold: " + 
              String.format("%.1f%%", memoryUsage * 100));
        
        // 分级保护措施
        if (memoryUsage > 0.9f) {
            // 紧急保护：强制清理和GC
            applyEmergencyProtection();
        } else if (memoryUsage > 0.85f) {
            // 高级保护：主动清理非关键资源
            applyAdvancedProtection();
        } else {
            // 基础保护：警告和轻度清理
            applyBasicProtection();
        }
    }
    
    private void applyEmergencyProtection() {
        Log.w("MemoryOverflowProtector", "Applying emergency memory protection");
        
        // 1. 强制清理所有缓存
        clearAllCaches();
        
        // 2. 释放可回收资源
        releaseReleasableResources();
        
        // 3. 触发GC
        System.gc();
        
        // 4. 限制新资源分配
        enableMemoryAllocationLimit();
        
        // 5. 通知用户可能的功能限制
        notifyUserAboutMemoryConstraint();
    }
    
    private void applyAdvancedProtection() {
        Log.i("MemoryOverflowProtector", "Applying advanced memory protection");
        
        // 1. 清理非关键缓存
        clearNonCriticalCaches();
        
        // 2. 压缩内存使用
        compressMemoryUsage();
        
        // 3. 延迟非关键操作
        delayNonCriticalOperations();
    }
    
    private void applyBasicProtection() {
        Log.i("MemoryOverflowProtector", "Applying basic memory protection");
        
        // 1. 记录警告日志
        logMemoryWarning();
        
        // 2. 轻度资源清理
        performLightCleanup();
        
        // 3. 增加监控频率
        increaseMonitoringFrequency();
    }
    
    // 内存分配限制器
    public static class MemoryAllocationLimiter {
        private static boolean allocationLimited = false;
        private static final long ALLOCATION_LIMIT = 10 * 1024 * 1024; // 10MB
        
        public static <T> T monitorAllocation(String allocationType, Supplier<T> allocator) {
            if (allocationLimited) {
                // 检查是否超过分配限制
                if (getCurrentAllocation() > ALLOCATION_LIMIT) {
                    throw new MemoryAllocationLimitExceededException(
                        "Memory allocation limit exceeded for: " + allocationType);
                }
            }
            
            return allocator.get();
        }
        
        public static void enableAllocationLimit() {
            allocationLimited = true;
            Log.w("MemoryAllocationLimiter", "Memory allocation limit enabled");
        }
        
        public static void disableAllocationLimit() {
            allocationLimited = false;
            Log.i("MemoryAllocationLimiter", "Memory allocation limit disabled");
        }
    }
}

// 使用示例
public class ImageCacheManager {
    
    public Bitmap getImage(String url) {
        // 检查内存分配限制
        return MemoryAllocationLimiter.monitorAllocation("image_cache", () -> {
            // 实际的图片获取逻辑
            Bitmap bitmap = memoryCache.get(url);
            if (bitmap == null) {
                bitmap = loadImageFromNetwork(url);
                
                // 只有在内存充足时才缓存
                if (isMemorySufficient()) {
                    memoryCache.put(url, bitmap);
                }
            }
            return bitmap;
        });
    }
    
    private boolean isMemorySufficient() {
        Runtime runtime = Runtime.getRuntime();
        long maxMemory = runtime.maxMemory();
        long usedMemory = runtime.totalMemory() - runtime.freeMemory();
        
        return (float) usedMemory / maxMemory < 0.7f; // 70%阈值
    }
}
```

```java
public class SocialAppMemoryProtector {
    
    public static void protectSocialApp() {

                        break;

                        cleanupNonCriticalResources();
                        break;
                    case HIGH:
                        // 高度警告：限制功能
                        limitMemoryIntensiveFeatures();
                        break;
                    case CRITICAL:
                        // 紧急警告：强制清理
                        forceCleanupAndNotifyUser();
                        break;
                }
            }
        });
        
        warningSystem.startMonitoring();
    }
}
```

```java
            writer.println("Time: " + new Date());
            writer.println("Thread: " + thread.getName());
            writer.println("Exception: " + throwable.getClass().getName());
            writer.println("Message: " + throwable.getMessage());
            writer.println();
            
            // 写入堆栈信息
            writer.println("Stack Trace:");
            throwable.printStackTrace(writer);
            writer.println();
            
            // 写入设备信息
            writer.println("Device Info:");
            writer.println("Model: " + Build.MODEL);
            writer.println("Android: " + Build.VERSION.RELEASE);
            writer.println("SDK: " + Build.VERSION.SDK_INT);
            writer.println();
            
            // 写入应用信息
            writer.println("App Info:");
            writer.println("Version: " + getAppVersion());
            writer.println("Process: " + getProcessName());
            writer.println();
            
            // 写入内存信息
            writer.println("Memory Info:");
            Runtime runtime = Runtime.getRuntime();
            writer.println("Max: " + runtime.maxMemory() / 1024 / 1024 + "MB");
            writer.println("Total: " + runtime.totalMemory() / 1024 / 1024 + "MB");
            writer.println("Free: " + runtime.freeMemory() / 1024 / 1024 + "MB");
            
            writer.flush();
            
            Log.i("CrashLogProtector", "Crash log written: " + logFile.getAbsolutePath());
            
        } catch (Exception e) {
            Log.e("CrashLogProtector", "Failed to write crash log", e);
        }
    }
    
    private static void protectLogFiles() {
        File logDir = getLogDirectory();
        File[] logFiles = logDir.listFiles((dir, name) -> name.endsWith(".log"));
        
        if (logFiles != null) {
            for (File logFile : logFiles) {
                // 设置文件权限，防止被误删
                logFile.setReadable(true, false); // 所有用户可读
                logFile.setWritable(true, false); // 所有用户可写
                
                // 备份重要日志文件
                if (isImportantCrashLog(logFile)) {
                    backupCrashLog(logFile);
                }
            }
        }
    }
    
    private static void checkLogSizeAndCleanup() {
        File logDir = getLogDirectory();
        long totalSize = getDirectorySize(logDir);
        
        if (totalSize > MAX_CRASH_LOG_SIZE) {
            // 日志文件过大，清理旧文件
            cleanupOldLogFiles(logDir);
        }
    }
    
    private static void cleanupOldLogFiles(File logDir) {
        File[] logFiles = logDir.listFiles((dir, name) -> name.endsWith(".log"));
        
        if (logFiles != null && logFiles.length > 10) { // 保留最近10个文件
            // 按修改时间排序
            Arrays.sort(logFiles, (f1, f2) -> Long.compare(f2.lastModified(), f1.lastModified()));
            
            // 删除旧文件（保留前10个）
            for (int i = 10; i < logFiles.length; i++) {
                if (logFiles[i].delete()) {
                    Log.i("CrashLogProtector", "Deleted old log file: " + logFiles[i].getName());
                }
            }
        }
    }
}

public class ComprehensiveCrashHandler implements Thread.UncaughtExceptionHandler {
    
    private final Thread.UncaughtExceptionHandler defaultHandler;
    
    public ComprehensiveCrashHandler(Thread.UncaughtExceptionHandler defaultHandler) {
        this.defaultHandler = defaultHandler;
    }
    
    @Override
    public void uncaughtException(Thread thread, Throwable throwable) {
        // 1. 保护崩溃日志
        CrashLogProtector.protectCrashLog(thread, throwable);
        
        // 2. 保存崩溃现场
        saveCrashScene(thread, throwable);
        
        // 3. 尝试恢复或安全退出
        if (canRecoverFromCrash(throwable)) {
            attemptCrashRecovery(throwable);
        } else {
            performSafeExit(throwable);
        }
        
        // 4. 调用默认处理器
        if (defaultHandler != null) {
            defaultHandler.uncaughtException(thread, throwable);
        }
    }
    
    private boolean canRecoverFromCrash(Throwable throwable) {
        // 判断是否可以恢复
        return !(throwable instanceof OutOfMemoryError) &&
               !(throwable instanceof StackOverflowError) &&
               !(throwable instanceof NoClassDefFoundError);
    }
    
    private void attemptCrashRecovery(Throwable throwable) {
        Log.w("ComprehensiveCrashHandler", "Attempting crash recovery");
        
        try {
            // 1. 清理可能的问题资源
            cleanupProblematicResources();
            
            // 2. 重启关键服务
            restartCriticalServices();
            
            // 3. 恢复用户界面
            restoreUserInterface();
            
            Log.i("ComprehensiveCrashHandler", "Crash recovery completed");
            
        } catch (Exception e) {
            Log.e("ComprehensiveCrashHandler", "Crash recovery failed", e);
            performSafeExit(throwable);
        }
    }
    
    private void performSafeExit(Throwable throwable) {
        Log.e("ComprehensiveCrashHandler", "Performing safe exit due to unrecoverable crash");
        
        // 1. 保存用户数据
        saveUserData();
        
        // 2. 清理临时文件
        cleanupTempFiles();
        
        // 3. 显示友好错误信息
        showFriendlyErrorMessage(throwable);
        
        // 4. 退出应用
        System.exit(1);
    }
}
```

### 5.1 分级兜底策略

```java
public class GradedFallbackStrategy {
    
    public static void applyGradedFallback(Throwable throwable, FallbackContext context) {
        FallbackLevel level = determineFallbackLevel(throwable, context);
        
        switch (level) {
            case LEVEL_1_MINIMAL:
                applyMinimalFallback(throwable, context);
                break;
            case LEVEL_2_BASIC:
                applyBasicFallback(throwable, context);
                break;
            case LEVEL_3_ADVANCED:
                applyAdvancedFallback(throwable, context);
                break;
            case LEVEL_4_EMERGENCY:
                applyEmergencyFallback(throwable, context);
                break;
        }
    }
    
    private static FallbackLevel determineFallbackLevel(Throwable throwable, FallbackContext context) {
        // 根据异常类型和上下文确定兜底级别
        if (throwable instanceof OutOfMemoryError) {
            return FallbackLevel.LEVEL_4_EMERGENCY;
        } else if (throwable instanceof NullPointerException && 
                   context.isCriticalPath()) {
            return FallbackLevel.LEVEL_3_ADVANCED;
        } else if (context.getFailureCount() > 3) {
            return FallbackLevel.LEVEL_2_BASIC;
        } else {
            return FallbackLevel.LEVEL_1_MINIMAL;
        }
    }
}
```

```java
public class FallbackMonitor {
    
    private static final Map<String, FallbackStats> fallbackStats = new ConcurrentHashMap<>();
    
    public static void recordFallbackEvent(String fallbackType, boolean success, long duration) {
        FallbackStats stats = fallbackStats.computeIfAbsent(fallbackType, k -> new FallbackStats());
        
        stats.totalAttempts++;
        if (success) {
            stats.successfulAttempts++;
        }
        stats.totalDuration += duration;
        
        // 检查是否需要调整策略
        if (stats.totalAttempts > 10) {
            analyzeAndAdjustStrategy(fallbackType, stats);
        }
    }
    
    private static void analyzeAndAdjustStrategy(String fallbackType, FallbackStats stats) {
        double successRate = (double) stats.successfulAttempts / stats.totalAttempts;
        double avgDuration = (double) stats.totalDuration / stats.totalAttempts;
        
        if (successRate < 0.5 || avgDuration > 5000) { // 成功率低于50%或平均耗时超过5秒
            Log.w("FallbackMonitor", "Fallback strategy needs adjustment: " + fallbackType);
            
            // 触发策略调整
            adjustFallbackStrategy(fallbackType);
        }
    }
}
```
