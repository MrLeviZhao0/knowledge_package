# 慢函数监控

## 1. 慢函数监控概述

### 1.1 慢函数定义与分类

**慢函数是指执行时间超过预期阈值的函数，主要分为以下几类：**

```
慢函数分类
├── I/O密集型函数
│   ├── 文件读写操作
│   ├── 网络请求处理
│   └── 数据库查询操作
├── CPU密集型函数
│   ├── 复杂计算逻辑
│   ├── 图像处理算法
│   └── 数据加密解密
└── 阻塞型函数
    ├── 同步锁等待
    ├── 线程休眠操作
    └── 资源竞争等待
```

### 1.2 监控价值与收益

**慢函数监控的核心价值：**
- **性能瓶颈定位**：快速定位影响性能的关键函数
- **代码优化指导**：为性能优化提供数据支撑
- **用户体验提升**：减少卡顿和响应延迟
- **资源利用优化**：提高CPU和内存使用效率

## 2. 函数执行时间监控

### 2.1 函数执行时间监控原理

通过Hook或代码插桩的方式，在函数入口和出口记录时间戳，计算函数执行耗时。

### 2.2 函数执行时间监控插桩

#### 2.2.1 手动插桩方式

```java
// 手动插桩监控函数执行时间
public class FunctionTimer {
    
    private static final Map<String, FunctionStats> functionStats = new ConcurrentHashMap<>();
    
    // 函数执行开始
    public static TimerContext startTimer(String functionName) {
        return new TimerContext(functionName, System.nanoTime());
    }
    
    // 函数执行结束
    public static void endTimer(TimerContext context) {
        long endTime = System.nanoTime();
        long duration = endTime - context.startTime;
        
        recordFunctionExecution(context.functionName, duration);
    }
    
    // 记录函数执行统计
    private static void recordFunctionExecution(String functionName, long duration) {
        FunctionStats stats = functionStats.computeIfAbsent(functionName, 
            k -> new FunctionStats());
        
        stats.callCount++;
        stats.totalTime += duration;
        stats.maxTime = Math.max(stats.maxTime, duration);
        stats.minTime = Math.min(stats.minTime, duration);
        
        // 检查是否超过阈值
        if (duration > getThresholdForFunction(functionName)) {
            reportSlowFunction(functionName, duration);
        }
    }
    
    // 定时器上下文
    public static class TimerContext {
        public final String functionName;
        public final long startTime;
        
        public TimerContext(String functionName, long startTime) {
            this.functionName = functionName;
            this.startTime = startTime;
        }
    }
    
    // 函数统计信息
    static class FunctionStats {
        long callCount = 0;
        long totalTime = 0;
        long maxTime = Long.MIN_VALUE;
        long minTime = Long.MAX_VALUE;
        
        public double getAverageTime() {
            return callCount > 0 ? (double) totalTime / callCount : 0;
        }
    }
}

// 使用示例
public class BusinessService {
    
    public void processOrder(Order order) {
        FunctionTimer.TimerContext timer = FunctionTimer.startTimer("processOrder");
        
        try {
            // 业务逻辑

            updateInventory(order);
            sendNotification(order);
            
        } finally {

                throw new IllegalArgumentException("Order cannot be null");
            }
            
            // 复杂的验证逻辑...
            Thread.sleep(10); // 模拟耗时操作
            
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        } finally {
            FunctionTimer.endTimer(timer);
        }
    }
```

#### 2.2.2 AOP自动插桩方式

```java
// 使用AOP框架自动插桩
@Aspect
public class FunctionMonitoringAspect {
    
    private static final long SLOW_FUNCTION_THRESHOLD = 100; // 100ms
    
    // 监控所有Service层方法
    @Around("execution(* com.example.service..*(..))")
    public Object monitorServiceMethods(ProceedingJoinPoint joinPoint) throws Throwable {
        String methodName = joinPoint.getSignature().getName();
        String className = joinPoint.getTarget().getClass().getSimpleName();
        String fullMethodName = className + "." + methodName;
        
        long startTime = System.currentTimeMillis();
        
        try {
            Object result = joinPoint.proceed();
            return result;
            
        } finally {
            long duration = System.currentTimeMillis() - startTime;
            
            // 记录方法执行时间
            recordMethodExecution(fullMethodName, duration);
            
            // 检查是否慢函数
            if (duration > SLOW_FUNCTION_THRESHOLD) {
                reportSlowMethod(fullMethodName, duration, joinPoint.getArgs());
            }
        }
    }
    
    // 监控DAO层数据库操作
    @Around("execution(* com.example.dao..*(..))")
    public Object monitorDaoMethods(ProceedingJoinPoint joinPoint) throws Throwable {
        String methodName = joinPoint.getSignature().getName();
        long startTime = System.currentTimeMillis();
        
        try {
            Object result = joinPoint.proceed();
            return result;
            
        } finally {
            long duration = System.currentTimeMillis() - startTime;
            
            // 数据库操作应该有更严格的阈值
            if (duration > 50) { // 50ms阈值
                reportSlowDatabaseOperation(methodName, duration);
            }
        }
    }
    
    private void recordMethodExecution(String methodName, long duration) {
        // 记录到监控系统
        MethodExecutionEvent event = new MethodExecutionEvent();
        event.methodName = methodName;
        event.duration = duration;
        event.timestamp = System.currentTimeMillis();
        event.threadName = Thread.currentThread().getName();
        
        APMReporter.reportMethodExecution(event);
    }
}

    public long duration;
    public long timestamp;
    public String threadName;
    public Object[] methodArgs; // 可选：记录方法参数
    public Object returnValue;   // 可选：记录返回值
}
```

```java
// 电商订单处理慢函数监控
public class OrderProcessingMonitor {
    
    private static final Map<String, PerformanceMetrics> orderMetrics = new ConcurrentHashMap<>();
    
    public static void monitorOrderProcessing(Order order, OrderProcessor processor) {
        String orderId = order.getId();
        long startTime = System.currentTimeMillis();
        
        try {
            // 执行订单处理
            processor.process(order);
            
            long duration = System.currentTimeMillis() - startTime;
            
            // 记录订单处理性能
            recordOrderProcessingMetrics(orderId, duration, true);
            
            if (duration > 3000) { // 3秒阈值
                reportSlowOrderProcessing(orderId, duration);
            }
            
        } catch (Exception e) {
            long duration = System.currentTimeMillis() - startTime;
            
            // 记录失败的处理
            recordOrderProcessingMetrics(orderId, duration, false);
            
            reportOrderProcessingFailure(orderId, duration, e);
            throw e;
        }
    }
    
    // 详细监控订单处理的各个阶段
    public static void monitorOrderProcessingStages(Order order) {
        StageMonitor stageMonitor = new StageMonitor(order.getId());
        
        // 监控验证阶段
        stageMonitor.monitorStage("validation", () -> {
            validateOrder(order);
        });
        
        // 监控价格计算阶段
        stageMonitor.monitorStage("pricing", () -> {
            calculateOrderPrice(order);
        });
        
        // 监控库存检查阶段
        stageMonitor.monitorStage("inventory", () -> {
            checkInventory(order);
        });
        
        // 监控支付处理阶段
        stageMonitor.monitorStage("payment", () -> {
            processPayment(order);
        });
        
        // 获取各阶段耗时分析
        StageAnalysis analysis = stageMonitor.getAnalysis();
        reportStageAnalysis(analysis);
    }

// 阶段监控器
public class StageMonitor {
    
    private final String orderId;
    private final Map<String, StageMetrics> stageMetrics = new LinkedHashMap<>();
    
    public StageMonitor(String orderId) {
        this.orderId = orderId;
    }
    
    public void monitorStage(String stageName, Runnable stageTask) {
        long startTime = System.currentTimeMillis();
        
        try {
            stageTask.run();
            
        } finally {
            long duration = System.currentTimeMillis() - startTime;
            
            StageMetrics metrics = new StageMetrics();
            metrics.stageName = stageName;
            metrics.duration = duration;
            metrics.success = true;
            metrics.timestamp = System.currentTimeMillis();
            
            stageMetrics.put(stageName, metrics);
        }
    }
    
    public StageAnalysis getAnalysis() {
        StageAnalysis analysis = new StageAnalysis();
        analysis.orderId = orderId;
        analysis.stageMetrics = new ArrayList<>(stageMetrics.values());
        analysis.totalDuration = calculateTotalDuration();
        analysis.bottleneckStage = findBottleneckStage();
        
        return analysis;
    }
    
    private long calculateTotalDuration() {
        return stageMetrics.values().stream()
            .mapToLong(metrics -> metrics.duration)
            .sum();
    }
    
    private String findBottleneckStage() {
        return stageMetrics.values().stream()
            .max(Comparator.comparingLong(metrics -> metrics.duration))
            .map(metrics -> metrics.stageName)
            .orElse("unknown");
    }
}
```

## 3. 异步任务监控

### 3.1 异步任务监控原理

监控AsyncTask、线程池、RxJava等异步任务的执行情况，识别慢异步操作。

### 3.2 异步任务监控插桩

```java
// 异步任务监控
public class AsyncTaskMonitor {
    
    private static final Map<String, TaskMetrics> taskMetrics = new ConcurrentHashMap<>();
    
    // 监控AsyncTask
    public static <Params, Progress, Result> AsyncTask<Params, Progress, Result> 
        monitorAsyncTask(String taskName, AsyncTask<Params, Progress, Result> task) {
        
        return new MonitoredAsyncTask<>(taskName, task);
    }
    
    // 监控线程池任务
    public static Runnable monitorRunnable(String taskName, Runnable runnable) {
        return new MonitoredRunnable(taskName, runnable);
    }
    
    // 监控RxJava Observable
    public static <T> Observable<T> monitorObservable(String observableName, Observable<T> observable) {
        return observable
            .doOnSubscribe(disposable -> onTaskStart(observableName))
            .doOnTerminate(() -> onTaskEnd(observableName, true))
            .doOnError(throwable -> onTaskEnd(observableName, false));
    }
    
    // 监控CompletableFuture
    public static <T> CompletableFuture<T> monitorCompletableFuture(
        String futureName, CompletableFuture<T> future) {
        
        long startTime = System.currentTimeMillis();
        
        return future.whenComplete((result, throwable) -> {
            long duration = System.currentTimeMillis() - startTime;
            boolean success = throwable == null;
            
            recordAsyncTaskExecution(futureName, duration, success);
            
            if (duration > getThresholdForAsyncTask(futureName)) {
                reportSlowAsyncTask(futureName, duration);
            }
        });
    }
    
    // 被监控的AsyncTask
    private static class MonitoredAsyncTask<Params, Progress, Result> 
        extends AsyncTask<Params, Progress, Result> {
        
        private final String taskName;
        private final AsyncTask<Params, Progress, Result> originalTask;
        private long startTime;
        
        public MonitoredAsyncTask(String taskName, AsyncTask<Params, Progress, Result> originalTask) {
            this.taskName = taskName;
            this.originalTask = originalTask;
        }
        
        @Override
        protected void onPreExecute() {
            startTime = System.currentTimeMillis();
            originalTask.onPreExecute();
        }
        
        @Override
        protected Result doInBackground(Params... params) {
            return originalTask.doInBackground(params);
        }
        
        @Override
        protected void onPostExecute(Result result) {
            long duration = System.currentTimeMillis() - startTime;
            recordAsyncTaskExecution(taskName, duration, true);
            
            originalTask.onPostExecute(result);
        }
        
        @Override
        protected void onCancelled(Result result) {
            long duration = System.currentTimeMillis() - startTime;
            recordAsyncTaskExecution(taskName, duration, false);
            
            originalTask.onCancelled(result);
        }
    }
    
    // 被监控的Runnable
    private static class MonitoredRunnable implements Runnable {
        private final String taskName;
        private final Runnable originalRunnable;
        
        public MonitoredRunnable(String taskName, Runnable originalRunnable) {
            this.taskName = taskName;
            this.originalRunnable = originalRunnable;
        }
        
        @Override
        public void run() {
            long startTime = System.currentTimeMillis();
            
            try {
                originalRunnable.run();
                
                long duration = System.currentTimeMillis() - startTime;
                recordAsyncTaskExecution(taskName, duration, true);
                
            } catch (Exception e) {
                long duration = System.currentTimeMillis() - startTime;
                recordAsyncTaskExecution(taskName, duration, false);
                throw e;
            }
        }
    }
}

// 使用示例
public class ImageLoader {
    
    private ExecutorService executor = Executors.newFixedThreadPool(4);
    
    public CompletableFuture<Bitmap> loadImageAsync(String imageUrl) {
        String taskName = "loadImage_" + imageUrl.hashCode();
        
        CompletableFuture<Bitmap> future = CompletableFuture.supplyAsync(() -> {
            // 图片加载逻辑
            return loadImageFromNetwork(imageUrl);
        }, executor);
        
        return AsyncTaskMonitor.monitorCompletableFuture(taskName, future);
    }
    
    public void loadImageWithCallback(String imageUrl, ImageLoadCallback callback) {
        String taskName = "loadImageWithCallback_" + imageUrl.hashCode();
        
        Runnable loadTask = AsyncTaskMonitor.monitorRunnable(taskName, () -> {
            Bitmap bitmap = loadImageFromNetwork(imageUrl);
            
            // 回到主线程回调
            new Handler(Looper.getMainLooper()).post(() -> {
                callback.onImageLoaded(bitmap);
            });
        });
        
        executor.execute(loadTask);
    }
}
```

## 4. 数据库操作监控

### 4.1 数据库操作监控原理

监控SQL查询、事务操作、数据库连接等数据库相关操作的执行性能。

### 4.2 数据库操作监控插桩

```java
// 数据库操作监控
public class DatabaseOperationMonitor {
    
    private static final long SLOW_QUERY_THRESHOLD = 100; // 100ms
    private static final long SLOW_TRANSACTION_THRESHOLD = 500; // 500ms
    
    // 监控SQL查询
    public static <T> T monitorQuery(String queryName, Supplier<T> querySupplier) {
        long startTime = System.currentTimeMillis();
        
        try {
            T result = querySupplier.get();
            
            long duration = System.currentTimeMillis() - startTime;
            recordQueryExecution(queryName, duration, true);
            
            if (duration > SLOW_QUERY_THRESHOLD) {
                reportSlowQuery(queryName, duration);
            }
            
            return result;
            
        } catch (Exception e) {
            long duration = System.currentTimeMillis() - startTime;
            recordQueryExecution(queryName, duration, false);
            throw e;
        }
    }
    
    // 监控数据库事务
    public static void monitorTransaction(String transactionName, Runnable transactionTask) {
        long startTime = System.currentTimeMillis();
        
        try {
            transactionTask.run();
            
            long duration = System.currentTimeMillis() - startTime;
            recordTransactionExecution(transactionName, duration, true);
            
            if (duration > SLOW_TRANSACTION_THRESHOLD) {
                reportSlowTransaction(transactionName, duration);
            }
            
        } catch (Exception e) {
            long duration = System.currentTimeMillis() - startTime;
            recordTransactionExecution(transactionName, duration, false);
            throw e;
        }
    }
    
    // 监控Room数据库操作（Android）
    public static class MonitoredRoomDatabase extends RoomDatabase {
        
        private final RoomDatabase originalDatabase;
        
        public MonitoredRoomDatabase(RoomDatabase originalDatabase) {
            this.originalDatabase = originalDatabase;
        }
        
        @Override
        public <T> T createDao(Class<T> daoClass) {
            T originalDao = originalDatabase.createDao(daoClass);
            return createMonitoredDao(originalDao, daoClass);
        }
        
        @SuppressWarnings("unchecked")
        private <T> T createMonitoredDao(T originalDao, Class<T> daoClass) {
            // 使用动态代理监控DAO方法
            return (T) Proxy.newProxyInstance(
                daoClass.getClassLoader(),
                new Class[]{daoClass},
                new DaoInvocationHandler(originalDao)
            );
        }
        
        private class DaoInvocationHandler implements InvocationHandler {
            private final Object originalDao;
            
            public DaoInvocationHandler(Object originalDao) {
                this.originalDao = originalDao;
            }
            
            @Override
            public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
                String methodName = method.getName();
                long startTime = System.currentTimeMillis();
                
                try {
                    Object result = method.invoke(originalDao, args);
                    
                    long duration = System.currentTimeMillis() - startTime;
                    recordDaoMethodExecution(methodName, duration, true);
                    
                    if (duration > SLOW_QUERY_THRESHOLD) {
                        reportSlowDaoMethod(methodName, duration);
                    }
                    
                    return result;
                    
                } catch (InvocationTargetException e) {
                    long duration = System.currentTimeMillis() - startTime;
                    recordDaoMethodExecution(methodName, duration, false);
                    throw e.getTargetException();
                }
            }
        }
    }
}
```
