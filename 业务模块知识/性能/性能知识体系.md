```java
public List<Integer> findDuplicates(int[] nums) {
    List<Integer> result = new ArrayList<>();
    for (int i = 0; i < nums.length; i++) {
        for (int j = i + 1; j < nums.length; j++) {
            if (nums[i] == nums[j]) {
                result.add(nums[i]);
            }
        }
    }
    return result;

// 优化后：O(n)
public List<Integer> findDuplicatesOptimized(int[] nums) {
    List<Integer> result = new ArrayList<>();
    Set<Integer> seen = new HashSet<>();
    
    for (int num : nums) {
        if (seen.contains(num)) {
            result.add(num);
        } else {
            seen.add(num);
        }
    }
    return result;
}
```

```java
// 优化前：使用额外O(n)空间
public void rotateArray(int[] nums, int k) {
    int n = nums.length;
    k = k % n;
    int[] temp = new int[n];
    
    for (int i = 0; i < n; i++) {
        temp[(i + k) % n] = nums[i];
    }
    
    System.arraycopy(temp, 0, nums, 0, n);
}

// 优化后：原地旋转，O(1)空间
public void rotateArrayInPlace(int[] nums, int k) {
    int n = nums.length;
    k = k % n;
    
    reverse(nums, 0, n - 1);
    reverse(nums, 0, k - 1);
    reverse(nums, k, n - 1);
}

private void reverse(int[] nums, int start, int end) {
    while (start < end) {
        int temp = nums[start];
        nums[start] = nums[end];
        nums[end] = temp;
        start++;
        end--;
    }
}
```

#### 3.2.1 对象池技术
```java
public class ObjectPool<T> {
    private final Queue<T> pool;
    private final Supplier<T> creator;
    private final Consumer<T> resetter;
    
    public ObjectPool(int size, Supplier<T> creator, Consumer<T> resetter) {
        this.pool = new ArrayBlockingQueue<>(size);
        this.creator = creator;
        this.resetter = resetter;
        
        // 预创建对象
        for (int i = 0; i < size; i++) {
            pool.offer(creator.get());
        }
    }
    
    public T acquire() {
        T obj = pool.poll();
        if (obj == null) {
            obj = creator.get();
        }
        return obj;
    }
    
    public void release(T obj) {
        if (obj != null) {
            resetter.accept(obj);
            pool.offer(obj);
        }
    }
}
```

```cpp
#include <sched.h>

class ThreadPriorityManager {
public:
    static bool setThreadPriority(pthread_t thread, int priority) {
        struct sched_param param;
        param.sched_priority = priority;
        
        // 尝试实时调度策略
        if (pthread_setschedparam(thread, SCHED_FIFO, &param) == 0) {
            return true;
        }
        
        // 回退到普通优先级
        param.sched_priority = 0;
        return pthread_setschedparam(thread, SCHED_OTHER, &param) == 0;
    }
    
    static bool setThreadAffinity(pthread_t thread, int cpu_id) {
        cpu_set_t cpuset;
        CPU_ZERO(&cpuset);
        CPU_SET(cpu_id, &cpuset);
        
        return pthread_setaffinity_np(thread, sizeof(cpu_set_t), &cpuset) == 0;
    }
};
```

```java
// 锁分段技术
public class StripedLockMap<K, V> {
    private final int stripes;
    private final Lock[] locks;
    private final Map<K, V>[] maps;
    
    @SuppressWarnings("unchecked")
    public StripedLockMap(int stripes) {
        this.stripes = stripes;
        this.locks = new Lock[stripes];
        this.maps = new Map[stripes];
        
        for (int i = 0; i < stripes; i++) {
            locks[i] = new ReentrantLock();
            maps[i] = new HashMap<>();
        }
    }
    
    private int getStripe(K key) {
        return Math.abs(key.hashCode() % stripes);
    }
    
    public void put(K key, V value) {
        int stripe = getStripe(key);
        locks[stripe].lock();
        try {
            maps[stripe].put(key, value);
        } finally {
            locks[stripe].unlock();
        }
    }
    
    public V get(K key) {
        int stripe = getStripe(key);
        locks[stripe].lock();
        try {
            return maps[stripe].get(key);
        } finally {
            locks[stripe].unlock();
        }
    }
}
```

```cpp
class MemoryPool {
private:
    struct Chunk {
        Chunk* next;
    };
    
    Chunk* mFreeList;
    size_t mChunkSize;
    size_t mPoolSize;
    char* mMemory;
    
public:
    MemoryPool(size_t chunkSize, size_t poolSize) {
        mChunkSize = chunkSize;
        mPoolSize = poolSize;
        mFreeList = nullptr;
        
        // 预分配内存
        mMemory = static_cast<char*>(malloc(poolSize));
        size_t numChunks = poolSize / chunkSize;
        
        for (size_t i = 0; i < numChunks; ++i) {
            Chunk* chunk = reinterpret_cast<Chunk*>(mMemory + i * chunkSize);
            chunk->next = mFreeList;
            mFreeList = chunk;
        }
    }
    
    ~MemoryPool() {
        free(mMemory);
    }
    
    void* allocate() {
        if (mFreeList == nullptr) {
            return nullptr;
        }
        
        Chunk* chunk = mFreeList;
        mFreeList = mFreeList->next;
        return chunk;
    }
    
    void deallocate(void* ptr) {
        Chunk* chunk = static_cast<Chunk*>(ptr);
        chunk->next = mFreeList;
        mFreeList = chunk;
    }
};
```

```java
public class MultiLevelCache<K, V> {
    private final Map<K, V> l1Cache; // 内存缓存
    private final Map<K, V> l2Cache; // 磁盘缓存
    private final LoadingCache<K, V> loadingCache; // 加载缓存
    
    public MultiLevelCache() {
        this.l1Cache = new ConcurrentHashMap<>();
        this.l2Cache = createDiskCache();
        
        this.loadingCache = CacheBuilder.newBuilder()
            .maximumSize(1000)
            .expireAfterWrite(10, TimeUnit.MINUTES)
            .build(new CacheLoader<K, V>() {
                @Override
                public V load(K key) throws Exception {
                    return loadFromSource(key);
                }
            });
    }
    
    public V get(K key) {
        // L1缓存查找
        V value = l1Cache.get(key);
        if (value != null) {
            return value;
        }
        
        // L2缓存查找
        value = l2Cache.get(key);
        if (value != null) {
            l1Cache.put(key, value); // 回填L1缓存
            return value;
        }
        
        // 加载缓存
        try {
            value = loadingCache.get(key);
            l1Cache.put(key, value);
            l2Cache.put(key, value);
            return value;
        } catch (ExecutionException e) {
            throw new RuntimeException("Failed to load value for key: " + key, e);
        }
    }
}
```

```java
public class AsyncProcessor {
    private final ExecutorService cpuBoundExecutor;
    private final ExecutorService ioBoundExecutor;
    private final ScheduledExecutorService scheduler;
    
    public AsyncProcessor() {
        this.cpuBoundExecutor = Executors.newFixedThreadPool(
            Runtime.getRuntime().availableProcessors());
        this.ioBoundExecutor = Executors.newCachedThreadPool();
        this.scheduler = Executors.newScheduledThreadPool(1);
    }
    
    public <T> CompletableFuture<T> processCpuBound(Supplier<T> task) {
        return CompletableFuture.supplyAsync(task, cpuBoundExecutor);
    }
    
    public <T> CompletableFuture<T> processIoBound(Supplier<T> task) {
        return CompletableFuture.supplyAsync(task, ioBoundExecutor);
    }
    
    public CompletableFuture<Void> scheduleTask(Runnable task, long delay, TimeUnit unit) {
        CompletableFuture<Void> future = new CompletableFuture<>();
        
        scheduler.schedule(() -> {
            try {
                task.run();
                future.complete(null);
            } catch (Exception e) {
                future.completeExceptionally(e);
            }
        }, delay, unit);
        
        return future;
    }
    
    public <T> CompletableFuture<List<T>> processInBatches(List<T> items, 
                                                          Function<T, CompletableFuture<T>> processor,
                                                          int batchSize) {
        List<CompletableFuture<List<T>>> batchFutures = new ArrayList<>();
        
        for (int i = 0; i < items.size(); i += batchSize) {
            int end = Math.min(i + batchSize, items.size());
            List<T> batch = items.subList(i, end);
            
            CompletableFuture<List<T>> batchFuture = CompletableFuture.supplyAsync(() -> {
                List<CompletableFuture<T>> futures = batch.stream()
                    .map(processor)
                    .collect(Collectors.toList());
                
                return CompletableFuture.allOf(futures.toArray(new CompletableFuture[0]))
                    .thenApply(v -> futures.stream()
                        .map(CompletableFuture::join)
                        .collect(Collectors.toList()));
            }, ioBoundExecutor).thenCompose(future -> future);
            
            batchFutures.add(batchFuture);
        }
        
        return CompletableFuture.allOf(batchFutures.toArray(new CompletableFuture[0]))
            .thenApply(v -> batchFutures.stream()
                .map(CompletableFuture::join)
                .flatMap(List::stream)
                .collect(Collectors.toList()));
    }
}
```
