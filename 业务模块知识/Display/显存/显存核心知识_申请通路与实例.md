## 2. 显存分配器实现

### 2.1 Gralloc分配器架构

#### 2.1.1 Gralloc模块结构
```cpp
class GrallocAllocator {
public:
    // 主要分配方法
    virtual int allocate(uint32_t width, uint32_t height, PixelFormat format,
                        uint32_t layerCount, uint64_t usage,
                        buffer_handle_t* handle, uint32_t* stride,
                        uint64_t graphicBufferId, std::string requestorName) = 0;

};
```

#### 2.1.2 内存类型选择逻辑
```cpp
// 根据usage标志选择内存类型
uint32_t chooseMemoryType(uint64_t usage) {
    if (usage & GRALLOC_USAGE_HW_TEXTURE) {
        return MEMORY_TYPE_GPU;
    }
    if (usage & GRALLOC_USAGE_HW_RENDER) {
        return MEMORY_TYPE_GPU;
    }
    if (usage & GRALLOC_USAGE_SW_READ_OFTEN) {
        return MEMORY_TYPE_SYSTEM;
    }
    if (usage & GRALLOC_USAGE_SW_WRITE_OFTEN) {
        return MEMORY_TYPE_SYSTEM;
    }
    return MEMORY_TYPE_DEFAULT;
}
```

### 2.2 ION内存分配器

#### 2.2.1 ION分配器实现
```cpp
class IonAllocator : public GrallocAllocator {
public:
    int allocate(uint32_t width, uint32_t height, PixelFormat format,
                uint32_t layerCount, uint64_t usage,
                buffer_handle_t* handle, uint32_t* stride,
                uint64_t graphicBufferId, std::string requestorName) override {
        
        // 计算缓冲区大小
        size_t size = calculateBufferSize(width, height, format, layerCount);
        
        // 选择ION堆类型
        int heapMask = chooseIonHeap(usage);
        
        // 通过ION驱动分配内存
        int fd = ion_alloc_fd(ionFd, size, 0, heapMask, 0);
        
        // 创建buffer_handle_t
        native_handle_t* nh = createNativeHandle(fd, size, usage);
        
        *handle = nh;
        *stride = calculateStride(width, format);
        
        return 0;
    }
};
```

#### 2.2.2 ION堆类型选择
```cpp
int chooseIonHeap(uint64_t usage) {
    int heapMask = 0;
    
    if (usage & GRALLOC_USAGE_PROTECTED) {
        // 安全内容使用安全堆
        heapMask |= ION_HEAP(ION_SECURE_HEAP_ID);
    } else if (usage & GRALLOC_USAGE_HW_VIDEO_ENCODER) {
        // 视频编码使用CMA堆
        heapMask |= ION_HEAP(ION_CMA_HEAP_ID);
    } else if (usage & (GRALLOC_USAGE_HW_TEXTURE | GRALLOC_USAGE_HW_RENDER)) {
        // GPU使用使用系统堆
        heapMask |= ION_HEAP(ION_SYSTEM_HEAP_ID);
    } else {
        // 默认使用系统堆
        heapMask |= ION_HEAP(ION_SYSTEM_HEAP_ID);
    }
    
    return heapMask;
}
```

## 3. 实际申请实例

### 3.1 应用层显存申请实例

#### 3.1.1 SurfaceView显存申请
```java
public class MainActivity extends Activity {
    private SurfaceView surfaceView;
    private SurfaceHolder surfaceHolder;
    
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        
        surfaceView = new SurfaceView(this);
        setContentView(surfaceView);
        
        surfaceHolder = surfaceView.getHolder();
        surfaceHolder.addCallback(new SurfaceHolder.Callback() {
            @Override
            public void surfaceCreated(SurfaceHolder holder) {
                // Surface创建时自动申请显存
                // 系统会创建GraphicBuffer用于绘制
                startRendering();
            }
            
            @Override
            public void surfaceChanged(SurfaceHolder holder, int format, int width, int height) {
                // Surface尺寸变化时重新申请显存
                updateBufferSize(width, height);
            }
            
            @Override
            public void surfaceDestroyed(SurfaceHolder holder) {
                // Surface销毁时释放显存
                stopRendering();
            }
        });
    }
    
    private void startRendering() {
        // 开始渲染，使用申请的显存
        Canvas canvas = surfaceHolder.lockCanvas();
        // 绘制操作...
        surfaceHolder.unlockCanvasAndPost(canvas);
    }
}
```

### 3.2 Native层显存申请实例

#### 3.2.1 通过ANativeWindow申请显存
```cpp
#include <android/native_window.h>
#include <android/native_window_jni.h>

class NativeBufferAllocator {
private:
    ANativeWindow* nativeWindow;
    
public:
    NativeBufferAllocator(JNIEnv* env, jobject surface) {
        // 从Java Surface获取ANativeWindow
        nativeWindow = ANativeWindow_fromSurface(env, surface);
        
        // 设置缓冲区格式和尺寸
        ANativeWindow_setBuffersGeometry(nativeWindow, 0, 0, 
                                        AHARDWAREBUFFER_FORMAT_R8G8B8A8_UNORM);
    }
    
    AHardwareBuffer* allocateBuffer(int width, int height) {
        AHardwareBuffer_Desc desc = {};
        desc.width = width;
        desc.height = height;
        desc.layers = 1;
        desc.format = AHARDWAREBUFFER_FORMAT_R8G8B8A8_UNORM;
        desc.usage = AHARDWAREBUFFER_USAGE_GPU_SAMPLED_IMAGE |
                     AHARDWAREBUFFER_USAGE_GPU_COLOR_OUTPUT |
                     AHARDWAREBUFFER_USAGE_CPU_READ_RARELY;
        
        AHardwareBuffer* buffer = nullptr;
        AHardwareBuffer_allocate(&desc, &buffer);
        return buffer;
    }
    
    ~NativeBufferAllocator() {
        if (nativeWindow) {
            ANativeWindow_release(nativeWindow);
        }
    }
};
```

#### 3.2.2 直接使用Gralloc分配器
```cpp
#include <grallocusage/GrallocUsageConversion.h>
#include <system/graphics.h>

class DirectGrallocAllocator {
private:
    sp<GraphicBuffer> graphicBuffer;
    
public:
    bool allocateBuffer(int width, int height, PixelFormat format, uint64_t usage) {
        graphicBuffer = new GraphicBuffer(width, height, format, 
                                         usage | GRALLOC_USAGE_HW_TEXTURE);
        
        if (graphicBuffer->initCheck() != OK) {
            ALOGE("Failed to allocate GraphicBuffer");
            return false;
        }
        
        ALOGI("Allocated buffer: %dx%d, format: %d, usage: 0x%lx",
              width, height, format, usage);
        return true;
    }
    
    void* lockBuffer() {
        if (!graphicBuffer) return nullptr;
        
        void* vaddr = nullptr;
        status_t err = graphicBuffer->lock(GRALLOC_USAGE_SW_WRITE_OFTEN, &vaddr);
        if (err != OK) {
            ALOGE("Failed to lock buffer: %d", err);
            return nullptr;
        }
        return vaddr;
    }
    
    void unlockBuffer() {
        if (graphicBuffer) {
            graphicBuffer->unlock();
        }
    }
};
```

#### 3.3.2 硬件合成器显存管理
```cpp
// HWC中的显存管理
class HWComposer {
private:
    std::vector<HWCLayer*> mLayers;
    
public:
    status_t prepare() {
        for (auto layer : mLayers) {
            // 为每个图层准备显存
            if (layer->requiresNewBuffer()) {
                // 申请新的显存缓冲区
                status_t err = allocateHwcBuffer(layer);
                if (err != NO_ERROR) {
                    return err;
                }
            }
        }
        return NO_ERROR;
    }
    
    status_t allocateHwcBuffer(HWCLayer* layer) {
        // 根据图层属性申请合适的显存
        uint32_t width = layer->getDisplayFrame().getWidth();
        uint32_t height = layer->getDisplayFrame().getHeight();
        PixelFormat format = layer->getBufferFormat();
        
        // 使用HWC特定的内存分配策略
        uint64_t usage = GRALLOC_USAGE_HW_COMPOSER | 
                        GRALLOC_USAGE_HW_RENDER |
                        GRALLOC_USAGE_HW_TEXTURE;
        
        sp<GraphicBuffer> buffer = new GraphicBuffer(width, height, format, usage);
        layer->setBuffer(buffer);
        
        return NO_ERROR;
    }
};
```

## 4. 显存申请优化策略

### 4.1 缓冲区复用策略

#### 4.1.1 BufferQueue缓冲区池
```cpp
class BufferQueue {
private:
    static constexpr int MAX_BUFFER_SLOTS = 64;
    sp<GraphicBuffer> mSlots[MAX_BUFFER_SLOTS];
    
public:
    // 缓冲区复用逻辑
    int findFreeSlot() {
        for (int i = 0; i < MAX_BUFFER_SLOTS; i++) {
            if (mSlots[i] == nullptr) {
                return i;
            }
        }
        // 缓冲区池已满，需要回收
        return recycleOldestSlot();
    }
};
```

#### 4.1.2 内存池管理
```cpp
class MemoryPool {
private:
    std::map<size_t, std::queue<void*>> mPools;
    
public:
    void* allocate(size_t size) {
        auto& pool = mPools[size];
        if (!pool.empty()) {
            // 从池中复用内存
            void* mem = pool.front();
            pool.pop();
            return mem;
        }
        // 池为空，申请新内存
        return malloc(size);
    }
    
    void deallocate(void* mem, size_t size) {
        // 将内存放回池中供复用
        mPools[size].push(mem);
    }
};
```

### 4.2 内存类型优化选择

#### 4.2.1 智能内存类型选择
```cpp
MemoryType chooseOptimalMemoryType(uint64_t usage, int width, int height) {
    size_t size = width * height * 4; // 估算大小
    
    if (usage & GRALLOC_USAGE_HW_VIDEO_ENCODER) {
        // 视频编码：优先使用CMA内存
        return size > CMA_THRESHOLD ? MEMORY_TYPE_CMA : MEMORY_TYPE_SYSTEM;
    }
    
    if (usage & (GRALLOC_USAGE_HW_TEXTURE | GRALLOC_USAGE_HW_RENDER)) {
        // GPU使用：根据大小选择GPU或系统内存
        return size > GPU_MEMORY_THRESHOLD ? MEMORY_TYPE_GPU : MEMORY_TYPE_SYSTEM;
    }
    
    // 默认使用系统内存
    return MEMORY_TYPE_SYSTEM;
}
```
