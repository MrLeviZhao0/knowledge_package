```cpp
class IonMemoryAllocator {
private:
    int mIonFd;
    
public:
    // ION内存分配实现
    int allocateIonMemory(size_t size, unsigned int heapMask, unsigned int flags) {
        struct ion_allocation_data allocationData = {};
        allocationData.len = size;
        allocationData.heap_id_mask = heapMask;
        allocationData.flags = flags;
        
        int ret = ioctl(mIonFd, ION_IOC_ALLOC, &allocationData);
        if (ret < 0) {
            ALOGE("ION allocation failed: %s", strerror(errno));
            return -errno;
        }
        
        return allocationData.fd;
    }
    
    // ION内存映射
    void* mapIonMemory(int fd, size_t size) {
        void* addr = mmap(NULL, size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
        if (addr == MAP_FAILED) {
            ALOGE("ION mmap failed: %s", strerror(errno));
            return nullptr;
        }
        return addr;
    }
};
```

```c
// GPU驱动显存统计
struct kgsl_memstat {
    uint64_t total;           // 总显存
    uint64_t allocated;       // 已分配显存
    uint64_t free;            // 空闲显存
    uint32_t allocation_count; // 分配次数
    uint32_t free_count;      // 释放次数
};

// 获取GPU显存统计
static int kgsl_get_memory_stats(struct kgsl_device *device, 
                                struct kgsl_memstat *stats) {
    spin_lock(&device->mem_lock);
    stats->total = device->gpu_mem_size;
    stats->allocated = device->gpu_mem_allocated;
    stats->free = stats->total - stats->allocated;
    stats->allocation_count = device->allocation_count;
    stats->free_count = device->free_count;
    spin_unlock(&device->mem_lock);
    
    return 0;
}
```

```cpp
class SurfaceFlingerMemoryMonitor {
private:
    std::mutex mMutex;
    std::map<pid_t, MemoryStats> mProcessStats;
    
public:
    // 记录缓冲区分配
    void recordBufferAllocation(pid_t pid, const sp<GraphicBuffer>& buffer) {
        std::lock_guard<std::mutex> lock(mMutex);
        
        size_t size = buffer->getStride() * buffer->getHeight() * 
                     bytesPerPixel(buffer->getPixelFormat());
        
        mProcessStats[pid].allocated += size;
        mProcessStats[pid].allocation_count++;
        
        // 更新峰值使用量
        if (mProcessStats[pid].allocated > mProcessStats[pid].peak_usage) {
            mProcessStats[pid].peak_usage = mProcessStats[pid].allocated;
        }
    }
    
    // 获取监控数据
    void dumpMemoryStats(String8& result) {
        std::lock_guard<std::mutex> lock(mMutex);
        
        result.append("=== SurfaceFlinger Memory Stats ===\n");
        for (const auto& [pid, stats] : mProcessStats) {
            result.appendFormat("PID %d: %zu bytes, Peak: %zu bytes, Allocations: %u\n",
                               pid, stats.allocated, stats.peak_usage, 
                               stats.allocation_count);
        }
    }
};
```

```java
public class GraphicsMemoryMonitor {
    private static final String TAG = "GraphicsMemoryMonitor";
    private static final long MONITOR_INTERVAL = 2000; // 2秒间隔
    
    private Handler mHandler;
    private MemoryStatsCallback mCallback;
    private boolean mIsMonitoring = false;
    
    public interface MemoryStatsCallback {
        void onMemoryStatsUpdate(MemoryStats stats);
        void onMemoryWarning(MemoryLevel level, long currentUsage);
    }
    
    public static class MemoryStats {
        public long totalGraphics;     // 总显存使用
        public long privateGraphics;    // 私有显存
        public long sharedGraphics;     // 共享显存
        public long peakUsage;          // 峰值使用
        public int bufferCount;         // 缓冲区数量
    }
    
    public void startMonitoring(MemoryStatsCallback callback) {
        if (mIsMonitoring) {
            Log.w(TAG, "Monitoring is already running");
            return;
        }
        
        mCallback = callback;
        mIsMonitoring = true;
        
        mHandler = new Handler(Looper.getMainLooper());
        mHandler.post(mMonitorRunnable);
    }
    
    private Runnable mMonitorRunnable = new Runnable() {
        @Override
        public void run() {
            MemoryStats stats = getCurrentMemoryStats();
            
            if (mCallback != null) {
                mCallback.onMemoryStatsUpdate(stats);
            }
            
            // 检查内存警告
            checkMemoryWarnings(stats.totalGraphics);
            
            if (mIsMonitoring) {
                mHandler.postDelayed(this, MONITOR_INTERVAL);
            }
        }
    };
    
    private MemoryStats getCurrentMemoryStats() {
        MemoryStats stats = new MemoryStats();
        
        Debug.MemoryInfo memoryInfo = new Debug.MemoryInfo();
        Debug.getMemoryInfo(memoryInfo);
        
        stats.totalGraphics = memoryInfo.getTotalGraphics();
        stats.privateGraphics = memoryInfo.getTotalPrivateDirty();
        stats.sharedGraphics = memoryInfo.getTotalSharedDirty();
        
        // 获取缓冲区数量（需要反射）
        stats.bufferCount = getBufferCountThroughReflection();
        
        return stats;
    }
}
```

```java
public class SurfaceMemoryTracker {
    private Surface mSurface;
    private long mAllocatedMemory = 0;
    private List<BufferInfo> mBufferInfos = new ArrayList<>();
    
    public SurfaceMemoryTracker(Surface surface) {
        mSurface = surface;
        setupSurfaceCallback();
    }
    
    private void setupSurfaceCallback() {
        try {
            // 通过反射获取Surface内部状态
            Field producerField = Surface.class.getDeclaredField("mProducer");
            producerField.setAccessible(true);
            Object producer = producerField.get(mSurface);
            
            // 监控BufferQueue状态
            monitorBufferQueue(producer);
            
        } catch (Exception e) {
            Log.e(TAG, "Failed to setup surface monitoring", e);
        }
    }
    
    public void trackBufferAllocation(int width, int height, int format) {
        long bufferSize = calculateBufferSize(width, height, format);
        mAllocatedMemory += bufferSize;
        
        BufferInfo info = new BufferInfo(width, height, format, bufferSize);
        mBufferInfos.add(info);
        
        Log.d(TAG, String.format("Buffer allocated: %dx%d, format: %d, size: %d bytes",
                                width, height, format, bufferSize));
    }
    
    public long getTotalAllocatedMemory() {
        return mAllocatedMemory;
    }
    
    public void releaseAllBuffers() {
        mAllocatedMemory = 0;
        mBufferInfos.clear();
        Log.d(TAG, "All buffers released");
    }
}
```

```cpp
#include <jni.h>
#include <android/log.h>

class NativeMemoryMonitorJNI {
private:
    JavaVM* mJvm;
    jobject mJavaMonitor;
    
public:
    NativeMemoryMonitorJNI(JavaVM* jvm, jobject javaMonitor) 
        : mJvm(jvm), mJavaMonitor(javaMonitor) {}
    
    // 监控GraphicBuffer分配
    void onGraphicBufferAllocated(uint32_t width, uint32_t height, 
                                 PixelFormat format, uint64_t usage) {
        JNIEnv* env;
        mJvm->GetEnv((void**)&env, JNI_VERSION_1_6);
        
        jclass clazz = env->GetObjectClass(mJavaMonitor);
        jmethodID method = env->GetMethodID(clazz, "onNativeBufferAllocated", 
                                           "(IIIJ)V");
        
        if (method != nullptr) {
            env->CallVoidMethod(mJavaMonitor, method, 
                              (jint)width, (jint)height, 
                              (jint)format, (jlong)usage);
        }
        
        // 记录本地监控数据
        recordNativeAllocation(width, height, format, usage);
    }
    
private:
    void recordNativeAllocation(uint32_t width, uint32_t height,
                               PixelFormat format, uint64_t usage) {
        size_t size = width * height * getBytesPerPixel(format);
        
        // 使用ATrace记录分配事件
        ATrace_beginSection("Native_Buffer_Allocation");
        ATrace_setCounter("buffer_size", size);
        ATrace_setCounter("buffer_usage", usage);
        ATrace_endSection();
        
        ALOGD("Native buffer allocated: %ux%u, format: %d, size: %zu",
              width, height, format, size);
    }
};

// JNI接口函数
extern "C" JNIEXPORT void JNICALL
```

## 4. 水位管理与优化策略

### 4.1 动态水位调整

#### 4.1.1 基于场景的水位调整
```java
public class AdaptiveMemoryManager {
    
    public enum MemoryScenario {
        NORMAL,          // 正常场景
        GAMING,          // 游戏场景
        VIDEO_PLAYBACK,  // 视频播放
        MULTI_WINDOW,    // 多窗口
        LOW_POWER        // 低电量
    }
    
    public MemoryThreshold getThresholdForScenario(MemoryScenario scenario) {
        switch (scenario) {
            case GAMING:
                return new MemoryThreshold(0.8f, 0.9f, 0.95f); // 宽松阈值
            case VIDEO_PLAYBACK:
                return new MemoryThreshold(0.7f, 0.85f, 0.9f); // 中等阈值
            case MULTI_WINDOW:
                return new MemoryThreshold(0.6f, 0.75f, 0.85f); // 严格阈值
            case LOW_POWER:
                return new MemoryThreshold(0.5f, 0.65f, 0.75f); // 非常严格
            default:
                return new MemoryThreshold(0.7f, 0.85f, 0.95f); // 默认阈值
        }
    }
    
    public void adjustMemoryUsageBasedOnScenario(MemoryScenario scenario) {
        MemoryThreshold threshold = getThresholdForScenario(scenario);
        long currentUsage = getCurrentGraphicsMemoryUsage();
        
        if (currentUsage > threshold.getCriticalThreshold()) {
            // 触发重度优化
            triggerAggressiveOptimization(scenario);
        } else if (currentUsage > threshold.getWarningThreshold()) {
            // 触发轻度优化
            triggerLightOptimization(scenario);
        }
    }
}
```

```java
public class SmartMemoryRecycler {
    
    public void recycleUnusedResources() {
        // 1. 回收长时间未使用的纹理
        recycleUnusedTextures();
        
        // 2. 清理缓存
        clearMemoryCaches();
        
        // 3. 压缩内存
        compressMemoryIfNeeded();
        
        // 4. 通知系统进行垃圾回收
        System.gc();
    }
    
    private void recycleUnusedTextures() {
        long currentTime = System.currentTimeMillis();
        
        for (TextureResource texture : mTexturePool) {
            if (currentTime - texture.getLastUsedTime() > TEXTURE_TIMEOUT) {
                texture.recycle();
                mRecycledTextures.add(texture);
            }
        }
    }
    
    private void compressMemoryIfNeeded() {
        if (isMemoryPressureHigh()) {
            // 使用更压缩的格式
            convertTexturesToCompressedFormat();
            
            // 降低纹理质量
            reduceTextureQuality();
        }
    }
}
```
