## 1. 稳定性问题分类

### 1.1 Java层稳定性问题 (JE - Java Exception)

#### 1.1.1 常见JE问题类型
- **NullPointerException**：空指针异常
- **IndexOutOfBoundsException**：数组越界
- **IllegalArgumentException**：非法参数
- **IllegalStateException**：非法状态
- **SecurityException**：权限异常

#### 1.1.3 JE经典案例
**案例1：Activity内存泄漏**
```java
// 错误示例：静态Context引用
public class AppManager {
    private static Context sContext; // 静态Context导致内存泄漏
    
    public static void init(Context context) {
        sContext = context; // Activity被静态引用，无法回收
    }
}

// 正确示例：使用Application Context
public class AppManager {
    private static Context sAppContext; // Application Context
    
    public static void init(Context context) {
        sAppContext = context.getApplicationContext();
    }
}
```

**案例2：Handler内存泄漏**
```java
// 错误示例：匿名Handler
public class MainActivity extends Activity {
    private Handler mHandler = new Handler() {
        @Override
        public void handleMessage(Message msg) {
            // 隐式持有Activity引用
        }
    };
}

// 正确示例：静态Handler + WeakReference
public class MainActivity extends Activity {
    private static class SafeHandler extends Handler {
        private final WeakReference<MainActivity> mActivity;
        
        public SafeHandler(MainActivity activity) {
            mActivity = new WeakReference<>(activity);
        }
        
        @Override
        public void handleMessage(Message msg) {
            MainActivity activity = mActivity.get();
            if (activity != null && !activity.isFinishing()) {
                activity.handleMessage(msg);
            }
        }
    }
}
```

#### 1.2.3 NE经典案例

**案例1：空指针解引用**
```cpp
// 错误示例
void processData(JNIEnv* env, jobject obj, jbyteArray data) {
    jbyte* buffer = env->GetByteArrayElements(data, NULL);
    
    // 可能为空指针
    if (buffer == NULL) {
        return; // 缺少空指针检查
    }
    
    // 使用buffer
    process_buffer(buffer, env->GetArrayLength(data));
    
    env->ReleaseByteArrayElements(data, buffer, JNI_ABORT);
}

// 正确示例
void processData(JNIEnv* env, jobject obj, jbyteArray data) {
    if (data == NULL) {
        env->ThrowNew(env->FindClass("java/lang/NullPointerException"), 
                     "data is null");
        return;
    }
    
    jbyte* buffer = env->GetByteArrayElements(data, NULL);
    if (buffer == NULL) {
        env->ThrowNew(env->FindClass("java/lang/OutOfMemoryError"), 
                     "Failed to get byte array elements");
        return;
    }
    
    jsize length = env->GetArrayLength(data);
    process_buffer(buffer, length);
    
    env->ReleaseByteArrayElements(data, buffer, JNI_ABORT);
}
```

**案例2：内存越界访问**
```cpp
// 错误示例：数组越界
void copyData(int* dest, int* src, int size) {
    for (int i = 0; i <= size; i++) {  // 错误：应该是 i < size
        dest[i] = src[i];  // 当i==size时越界
    }
}

// 正确示例
void copyData(int* dest, int* src, int size) {
    if (dest == NULL || src == NULL || size <= 0) {
        return;
    }
    
    for (int i = 0; i < size; i++) {
        dest[i] = src[i];
    }
}
```

**崩溃监控方案**：
```java
public class CrashHandler implements Thread.UncaughtExceptionHandler {
    private final Thread.UncaughtExceptionHandler mDefaultHandler;
    
    public CrashHandler() {
        mDefaultHandler = Thread.getDefaultUncaughtExceptionHandler();
    }
    
    @Override
    public void uncaughtException(Thread thread, Throwable ex) {
        // 收集崩溃信息
        collectCrashInfo(ex);
        
        // 上传崩溃日志
        uploadCrashLog(ex);
        
        // 调用默认处理
        if (mDefaultHandler != null) {
            mDefaultHandler.uncaughtException(thread, ex);
        }
    }
    
    private void collectCrashInfo(Throwable ex) {
        StringBuilder sb = new StringBuilder();
        
        // 设备信息
        sb.append("Brand: ").append(Build.BRAND).append("\n");
        sb.append("Model: ").append(Build.MODEL).append("\n");
        sb.append("Android: ").append(Build.VERSION.RELEASE).append("\n");
        
        // 崩溃堆栈
        StringWriter sw = new StringWriter();
        PrintWriter pw = new PrintWriter(sw);
        ex.printStackTrace(pw);
        sb.append("Stack Trace:\n").append(sw.toString());
        
        // 保存到文件
        saveToFile(sb.toString());
    }
}
```

#### 3.1.2 服务端分析
```python
# 崩溃日志分析脚本
import json
import re
from collections import Counter

class CrashAnalyzer:
    def __init__(self, log_file):
        self.log_file = log_file
        self.crash_patterns = {
            'null_pointer': r'NullPointerException',
            'out_of_memory': r'OutOfMemoryError',
            'anr': r'ANR.*keyDispatchingTimedOut',
            'native_crash': r'signal.*SIG'
        }
    
    def analyze_crashes(self):
        crash_stats = Counter()
        
        with open(self.log_file, 'r') as f:
            for line in f:
                for crash_type, pattern in self.crash_patterns.items():
                    if re.search(pattern, line):
                        crash_stats[crash_type] += 1
                        break
        
        return crash_stats
    
    def generate_report(self):
        stats = self.analyze_crashes()
        total_crashes = sum(stats.values())
        
        report = {
            'total_crashes': total_crashes,
            'crash_distribution': dict(stats),
            'crash_rate': total_crashes / self.get_user_count()
        }
        
        return report
```

#### 4.1.2 异常处理策略
```java
public class ExceptionHandler {
    // 可恢复异常
    public boolean retryOperation(int maxRetries) {
        for (int i = 0; i < maxRetries; i++) {
            try {
                performOperation();
                return true;
            } catch (RecoverableException e) {
                Log.w(TAG, "Operation failed, retrying...", e);
                
                // 指数退避
                try {
                    Thread.sleep(1000 * (1 << i));
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    return false;
                }
            }
        }
        return false;
    }
    
    // 不可恢复异常
    public void handleFatalError(Throwable error) {
        // 记录错误信息
        logFatalError(error);
        
        // 优雅退出
        if (Looper.getMainLooper() == Looper.myLooper()) {
            // 在主线程，显示错误对话框
            showErrorDialog(error);
        } else {
            // 在后台线程，终止进程
            System.exit(1);
        }
    }
}
```

#### 4.2.2 自动化监控
```python
# 自动化稳定性测试脚本
import subprocess
import time
import psutil

class StabilityTester:
    def __init__(self, package_name):
        self.package_name = package_name
        
    def run_stability_test(self, duration=3600):  # 1小时
        start_time = time.time()
        crash_count = 0
        
        while time.time() - start_time < duration:
            # 启动应用
            subprocess.run(['adb', 'shell', 'am', 'start', self.package_name])
            
            # 监控应用状态
            time.sleep(10)  # 等待应用启动
            
            # 检查是否崩溃
            if not self.is_app_running():
                crash_count += 1
                self.collect_crash_log()
            
            # 停止应用
            subprocess.run(['adb', 'shell', 'am', 'force-stop', self.package_name])
            time.sleep(2)
        
        return crash_count
    
    def is_app_running(self):
        result = subprocess.run(['adb', 'shell', 'pidof', self.package_name], 
                              capture_output=True)
        return result.returncode == 0
```
