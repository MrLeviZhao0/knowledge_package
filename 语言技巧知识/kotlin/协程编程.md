# Kotlin协程编程

## 概述
Kotlin协程是轻量级的线程，用于简化异步编程。协程通过挂起函数（suspend functions）实现非阻塞的异步操作，避免了回调地狱，提供了更直观的异步代码编写方式。

## 基础概念

### 协程构建器
```kotlin
import kotlinx.coroutines.*

fun main() = runBlocking { // 顶层协程构建器
    // launch - 启动新协程，不返回结果
    val job = launch {
        delay(1000L) // 非阻塞延迟
        println("World!")
    }
    
    println("Hello,")
    job.join() // 等待协程完成
}

// async - 启动新协程，返回Deferred结果
fun main() = runBlocking {
    val deferred: Deferred<Int> = async {
        delay(1000L)
        42 // 返回结果
    }
    
    println("Waiting for result...")
    val result = deferred.await() // 等待并获取结果
    println("Result: $result")
}
```

### 挂起函数
```kotlin
// 挂起函数 - 可以暂停和恢复的执行
suspend fun doWork(): String {
    delay(1000L) // 模拟耗时操作
    return "Work completed"
}

// 组合挂起函数
suspend fun doMultipleWork(): String {
    val result1 = doWork()
    val result2 = doWork()
    return "$result1 and $result2"
}

fun main() = runBlocking {
    val result = doMultipleWork()
    println(result)
}
```

## 协程上下文和调度器

### 调度器类型
```kotlin
fun main() = runBlocking {
    // Dispatchers.Default - CPU密集型任务
    launch(Dispatchers.Default) {
        println("Default: ${Thread.currentThread().name}")
    }
    
    // Dispatchers.IO - I/O密集型任务
    launch(Dispatchers.IO) {
        println("IO: ${Thread.currentThread().name}")
    }
    
    // Dispatchers.Main - Android主线程（需要依赖）
    // launch(Dispatchers.Main) {
    //     println("Main: ${Thread.currentThread().name}")
    // }
    
    // Dispatchers.Unconfined - 不限制线程
    launch(Dispatchers.Unconfined) {
        println("Unconfined: ${Thread.currentThread().name}")
    }
    
    // 自定义线程池
    val customDispatcher = newSingleThreadContext("MyThread")
    launch(customDispatcher) {
        println("Custom: ${Thread.currentThread().name}")
    }
    customDispatcher.close() // 记得关闭
}
```

### 上下文组合
```kotlin
fun main() = runBlocking {
    // 组合上下文元素
    val context = Dispatchers.Default + CoroutineName("MyCoroutine")
    
    launch(context) {
        println("Running in ${coroutineContext[CoroutineName]?.name} on ${Thread.currentThread().name}")
    }
    
    // 修改上下文
    withContext(Dispatchers.IO) {
        println("Now on IO dispatcher: ${Thread.currentThread().name}")
    }
}
```

## 结构化并发

### 作用域构建器
```kotlin
fun main() = runBlocking {
    // coroutineScope - 结构化并发作用域
    coroutineScope {
        launch {
            delay(1000L)
            println("Task 1")
        }
        
        launch {
            delay(500L)
            println("Task 2")
        }
    }
    println("Both tasks completed")
}

// supervisorScope - 子协程失败不影响其他子协程
fun main() = runBlocking {
    supervisorScope {
        val child1 = launch {
            delay(1000L)
            println("Child 1 completed")
        }
        
        val child2 = launch {
            delay(500L)
            throw RuntimeException("Child 2 failed")
        }
        
        child1.join()
        println("Supervisor scope completed")
    }
}
```

### 自定义作用域
```kotlin
class MyActivity {
    private val scope = CoroutineScope(Dispatchers.Main + SupervisorJob())
    
    fun doWork() {
        scope.launch {
            // 在Activity作用域内执行协程
            val data = fetchData()
            updateUI(data)
        }
    }
    
    fun onDestroy() {
        scope.cancel() // 取消所有协程
    }
    
    private suspend fun fetchData(): String {
        return withContext(Dispatchers.IO) {
            delay(1000L)
            "Data from network"
        }
    }
    
    private fun updateUI(data: String) {
        println("Updating UI with: $data")
    }
}
```

## 异步流（Flow）

### 创建和消费流
```kotlin
import kotlinx.coroutines.flow.*

fun simpleFlow(): Flow<Int> = flow {
    for (i in 1..3) {
        delay(100L) // 模拟异步工作
        emit(i) // 发射值
    }
}

fun main() = runBlocking {
    // 收集流
    simpleFlow().collect { value ->
        println(value)
    }
    
    // 转换操作
    simpleFlow()
        .map { it * 2 }
        .collect { println("Doubled: $it") }
    
    // 过滤操作
    simpleFlow()
        .filter { it % 2 == 1 }
        .collect { println("Odd: $it") }
}
```

### 流操作符
```kotlin
fun advancedFlowOperations() = runBlocking {
    // 合并多个流
    val flow1 = flowOf(1, 2, 3)
    val flow2 = flowOf("A", "B", "C")
    
    flow1.zip(flow2) { a, b -> "$a -> $b" }
        .collect { println(it) }
    
    // 扁平化流
    fun requestFlow(i: Int): Flow<String> = flow {
        emit("$i: First")
        delay(500L)
        emit("$i: Second")
    }
    
    (1..3).asFlow()
        .flatMapConcat { requestFlow(it) }
        .collect { println(it) }
    
    // 缓冲和并发
    simpleFlow()
        .buffer() // 缓冲发射值
        .collect { value ->
            delay(300L) // 处理较慢
            println(value)
        }
}
```

## 异常处理

### 协程异常处理
```kotlin
fun exceptionHandling() = runBlocking {
    // try-catch处理异常
    val job = launch {
        try {
            delay(1000L)
            throw RuntimeException("Something went wrong")
        } catch (e: Exception) {
            println("Caught exception: ${e.message}")
        }
    }
    job.join()
    
    // CoroutineExceptionHandler
    val handler = CoroutineExceptionHandler { _, exception ->
        println("CoroutineExceptionHandler got $exception")
    }
    
    val job2 = launch(handler) {
        throw RuntimeException("Test exception")
    }
    job2.join()
    
    // async的异常处理
    val deferred = async {
        throw RuntimeException("Async exception")
    }
    
    try {
        deferred.await()
    } catch (e: Exception) {
        println("Caught async exception: ${e.message}")
    }
}
```

### 流异常处理
```kotlin
fun flowExceptionHandling() = runBlocking {
    // catch操作符
    flow {
        emit(1)
        throw RuntimeException("Flow error")
    }
    .catch { e -> 
        println("Caught exception: $e")
        emit(-1) // 从异常中恢复
    }
    .collect { println(it) }
    
    // retry操作符
    var attempt = 0
    flow {
        attempt++
        if (attempt < 3) {
            throw RuntimeException("Attempt $attempt failed")
        }
        emit("Success on attempt $attempt")
    }
    .retry(2) // 重试2次
    .collect { println(it) }
}
```

```kotlin
class UserRepository {
    private val apiService = ApiService()
    
    // 串行请求
    suspend fun getUserWithPosts(userId: Int): UserWithPosts {
        val user = apiService.getUser(userId)
        val posts = apiService.getPosts(userId)
        return UserWithPosts(user, posts)
    }
    
    // 并行请求
    suspend fun getUserWithPostsParallel(userId: Int): UserWithPosts = coroutineScope {
        val userDeferred = async { apiService.getUser(userId) }
        val postsDeferred = async { apiService.getPosts(userId) }
        
        UserWithPosts(userDeferred.await(), postsDeferred.await())
    }
    
    // 使用流实现实时更新
    fun observeUser(userId: Int): Flow<User> = flow {
        while (true) {
            val user = apiService.getUser(userId)
            emit(user)
            delay(5000L) // 每5秒更新一次
        }
    }
}
```

### 数据库操作
```kotlin
class UserDao {
    private val database = Database()
    
    // 事务操作
    suspend fun updateUserWithTransaction(user: User, newName: String) {
        withContext(Dispatchers.IO) {
            database.beginTransaction()
            try {
                database.updateUser(user.copy(name = newName))
                database.markUserUpdated(user.id)
                database.setTransactionSuccessful()
            } finally {
                database.endTransaction()
            }
        }
    }
    
    // 流式查询
    fun observeUsers(): Flow<List<User>> = flow {
        while (true) {
            val users = database.getUsers()
            emit(users)
            delay(2000L) // 每2秒检查一次变化
        }
    }
    .distinctUntilChanged() // 只在数据变化时发射
}

// 模拟数据库
class Database {
    fun beginTransaction() { println("Begin transaction") }
    fun setTransactionSuccessful() { println("Transaction successful") }
    fun endTransaction() { println("End transaction") }
    fun updateUser(user: User) { println("Update user: $user") }
    fun markUserUpdated(id: Int) { println("Mark user updated: $id") }
    fun getUsers(): List<User> = listOf(User(1, "Alice"), User(2, "Bob"))
}
```

### 文件操作
```kotlin
class FileProcessor {
    // 并行处理多个文件
    suspend fun processFilesConcurrently(files: List<File>): List<ProcessedFile> = coroutineScope {
        files.map { file ->
            async(Dispatchers.IO) {
                processFile(file)
            }
        }.awaitAll()
    }
    
    private suspend fun processFile(file: File): ProcessedFile {
        val content = withContext(Dispatchers.IO) {
            file.readText()
        }
        
        // 在Default调度器上进行CPU密集型处理
        val processedContent = withContext(Dispatchers.Default) {
            content.uppercase() // 模拟处理
        }
        
        return ProcessedFile(file.name, processedContent)
    }
    
    // 流式文件读取
    fun readLargeFile(file: File): Flow<String> = flow {
        file.bufferedReader().use { reader ->
            reader.lineSequence().forEach { line ->
                emit(line)
                delay(10L) // 控制发射速率
            }
        }
    }
}

data class ProcessedFile(val name: String, val content: String)
```

```kotlin
    // CPU密集型任务使用Default

private fun calculateFibonacci(n: Int): Int {
    return if (n <= 1) n else calculateFibonacci(n - 1) + calculateFibonacci(n - 2)
}
```

### 避免阻塞操作
```kotlin
// 错误：在协程中使用阻塞调用
suspend fun badExample() {
    Thread.sleep(1000L) // 阻塞线程！
}

// 正确：使用delay进行非阻塞延迟
suspend fun goodExample() {
    delay(1000L) // 非阻塞挂起
}

// 将阻塞操作包装在适当的调度器中
suspend fun handleBlockingOperation() {
    val result = withContext(Dispatchers.IO) {
        // 将阻塞调用放在IO调度器
        runBlocking { 
            // 只有在必要时使用runBlocking
            Thread.sleep(1000L)
            "Result"
        }
    }
    println(result)
}
```

### 资源管理
```kotlin
class ResourceManager {
    // 使用协程作用域管理资源
    private val scope = CoroutineScope(Dispatchers.IO)
    
    fun startProcessing() {
        scope.launch {
            useResource()
        }
    }
    
    private suspend fun useResource() {
        // 确保资源正确释放
        val resource = acquireResource()
        try {
            // 使用资源
            resource.process()
        } finally {
            resource.release()
        }
    }
    
    fun cleanup() {
        scope.cancel() // 取消所有协程
    }
}

// 模拟资源
class Resource {
    fun process() { println("Processing resource") }
    fun release() { println("Resource released") }
}

fun acquireResource(): Resource {
    println("Resource acquired")
    return Resource()
}
```

### 测试挂起函数
```kotlin
import kotlinx.coroutines.test.*
import org.junit.Test

class CoroutineTest {
    @Test
    fun `test suspend function`() = runTest { // 用于测试的协程构建器
        val result = doWork()
        assert(result == "Work completed")
    }
    
    @Test
    fun `test time control`() = runTest {
        val deferred = async {
            delay(1000L)
            "Result"

    }
}
```

## 总结
Kotlin协程提供了强大而简洁的异步编程解决方案。通过结构化并发、流处理和适当的异常处理，可以编写出既安全又高效的异步代码。在实际应用中，要根据任务类型选择合适的调度器，并注意资源管理和性能优化。
