# 导航和架构

## 概述
导航和架构是构建可维护、可扩展Compose应用的关键。本章详细介绍Navigation组件、MVVM架构模式以及应用架构的最佳实践。

## Navigation组件

### 基础导航设置

### 带参数的导航

```kotlin
@Composable
fun MyAppNavHostWithArgs() {
    val navController = rememberNavController()
    
    NavHost(navController, startDestination = "user_list") {
        composable("user_list") {
            UserListScreen(
                onUserClick = { userId ->
                    navController.navigate("user_detail/$userId")
                }
            )
        }
        
        composable(
            route = "user_detail/{userId}",
            arguments = listOf(
                navArgument("userId") { type = NavType.StringType }
            )
        ) { backStackEntry ->
            val userId = backStackEntry.arguments?.getString("userId") ?: ""
            UserDetailScreen(
                userId = userId,
                onNavigateBack = { navController.popBackStack() }
            )
        }
    }
}

// 使用密封类定义路由
sealed class Screen(val route: String) {
    object Home : Screen("home")
    object Profile : Screen("profile")
    object Settings : Screen("settings")
    object UserDetail : Screen("user_detail/{userId}") {
        fun createRoute(userId: String) = "user_detail/$userId"
    }
}
```

## MVVM架构模式

### 基础MVVM实现

```kotlin
// Model
data class User(
    val id: String,
    val name: String,
    val email: String,
    val avatarUrl: String?
)

// Repository
class UserRepository {
    private val apiService = ApiService()
    private val localDataSource = UserLocalDataSource()
    
    suspend fun getUser(userId: String): User {
        return localDataSource.getUser(userId) ?: 
            apiService.getUser(userId).also { user ->
                localDataSource.saveUser(user)
            }
    }
    
    suspend fun updateUser(user: User): Boolean {
        return try {
            apiService.updateUser(user)
            localDataSource.saveUser(user)
            true
        } catch (e: Exception) {
            false
        }
    }
}

// ViewModel
class UserProfileViewModel(
    private val userRepository: UserRepository
) : ViewModel() {
    private val _userState = MutableStateFlow<UserState>(UserState.Loading)
    val userState: StateFlow<UserState> = _userState.asStateFlow()
    
    fun loadUser(userId: String) {
        viewModelScope.launch {
            _userState.value = UserState.Loading
            try {
                val user = userRepository.getUser(userId)
                _userState.value = UserState.Success(user)
            } catch (e: Exception) {
                _userState.value = UserState.Error(e.message ?: "Unknown error")
            }
        }
    }
    
    fun updateUserName(userId: String, newName: String) {
        viewModelScope.launch {
            val currentState = _userState.value
            if (currentState is UserState.Success) {
                val updatedUser = currentState.user.copy(name = newName)
                val success = userRepository.updateUser(updatedUser)
                if (success) {
                    _userState.value = UserState.Success(updatedUser)
                }
            }
        }
    }
}

// State
sealed class UserState {
    object Loading : UserState()
    data class Success(val user: User) : UserState()
    data class Error(val message: String) : UserState()
}

// View
@Composable
fun UserProfileScreen(
    userId: String,
    viewModel: UserProfileViewModel = viewModel(
        factory = UserProfileViewModelFactory(userId)
    )
) {
    val userState by viewModel.userState.collectAsState()
    
    when (userState) {
        is UserState.Loading -> {
            CircularProgressIndicator()
        }
        is UserState.Success -> {
            val user = (userState as UserState.Success).user
            UserProfileContent(
                user = user,
                onNameUpdate = { newName ->
                    viewModel.updateUserName(userId, newName)
                }
            )
        }
        is UserState.Error -> {
            val error = (userState as UserState.Error).message
            ErrorScreen(error = error)
        }
    }
}
```

```kotlin
class GetUserUseCase(
    private val userRepository: UserRepository
) {
    suspend operator fun invoke(userId: String): Result<User> {
        return try {
            Result.success(userRepository.getUser(userId))
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
}

// 表示模块 - presentation
@Composable

    getUserUseCase: GetUserUseCase = GetUserUseCase(AppModule.userRepository)

    LaunchedEffect(userId) {
        val result = getUserUseCase(userId)
        userState = if (result.isSuccess) {
            UserState.Success(result.getOrThrow())
        } else {
            UserState.Error(result.exceptionOrThrow()?.message ?: "Unknown error")
        }
    }

}
```

### 状态管理架构

```kotlin
// 全局状态管理
class AppStateManager {
    private val _currentUser = MutableStateFlow<User?>(null)
    val currentUser: StateFlow<User?> = _currentUser.asStateFlow()
    
    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()
    
    fun setCurrentUser(user: User?) {
        _currentUser.value = user
    }
    
    fun setLoading(loading: Boolean) {
        _isLoading.value = loading
    }
}

// 在Composable中使用全局状态
@Composable
fun AppContent() {
    val appStateManager = remember { AppStateManager() }
    val currentUser by appStateManager.currentUser.collectAsState()
    val isLoading by appStateManager.isLoading.collectAsState()
    
    if (isLoading) {
        LoadingScreen()
    } else if (currentUser == null) {
        LoginScreen(
            onLoginSuccess = { user ->
                appStateManager.setCurrentUser(user)
            }
        )
    } else {
        MainScreen(
            user = currentUser!!,
            onLogout = {
                appStateManager.setCurrentUser(null)
            }
        )
    }
}
```

## 导航模式

### 底部导航

```kotlin
@Composable
fun BottomNavigationApp() {
    val navController = rememberNavController()
    val currentDestination = navController.currentBackStackEntryAsState().value?.destination
    
    Scaffold(
        bottomBar = {
            BottomNavigation {
                val screens = listOf(
                    Screen.Home,
                    Screen.Profile,
                    Screen.Settings
                )
                
                screens.forEach { screen ->
                    BottomNavigationItem(
                        icon = { Icon(Icons.Default.Home, contentDescription = null) },
                        label = { Text(screen.route) },
                        selected = currentDestination?.route == screen.route,
                        onClick = {
                            navController.navigate(screen.route) {
                                // 弹出到起始目的地，避免回退栈堆积
                                popUpTo(navController.graph.findStartDestination().id) {
                                    saveState = true
                                }
                                // 避免同一目的地的重复点击
                                launchSingleTop = true
                                // 恢复状态
                                restoreState = true
                            }
                        }
                    )
                }
            }
        }
    ) { innerPadding ->
        NavHost(
            navController = navController,
            startDestination = Screen.Home.route,
            modifier = Modifier.padding(innerPadding)
        ) {
            composable(Screen.Home.route) { HomeScreen() }
            composable(Screen.Profile.route) { ProfileScreen() }
            composable(Screen.Settings.route) { SettingsScreen() }
        }
    }
}
```

## 总结
合理的导航和架构设计是构建高质量Compose应用的基础。通过使用Navigation组件、MVVM架构模式以及遵循架构原则，可以创建出可维护、可测试且性能优异的现代Android应用。
