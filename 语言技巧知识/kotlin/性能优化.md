# 性能优化

## 概述
Kotlin性能优化涉及代码效率、内存管理、网络请求和UI渲染等多个方面。本章详细介绍Kotlin在Android开发中的性能优化技巧和最佳实践。

## 内存优化

### 对象创建优化

**避免不必要的对象创建:**
```kotlin
// 不好的做法：每次调用都创建新对象
fun processData(data: List<String>): List<String> {
    return data.map { it.toUpperCase() }
        .filter { it.length > 5 }
        .toList() // 不必要的toList()调用
}

// 优化后的做法
fun processDataOptimized(data: List<String>): List<String> {
    return data.asSequence()
        .map { it.toUpperCase() }
        .filter { it.length > 5 }
        .toList()
}

// 使用内联函数减少函数调用开销
inline fun <T> List<T>.fastFilter(predicate: (T) -> Boolean): List<T> {
    return filter(predicate)
}
```

**字符串操作优化:**
```kotlin
// 不好的做法：字符串拼接
fun buildMessage(parts: List<String>): String {
    var result = ""
    for (part in parts) {
        result += part // 每次拼接都创建新字符串
    }
    return result
}

// 优化后的做法：使用StringBuilder
fun buildMessageOptimized(parts: List<String>): String {
    return StringBuilder().apply {
        for (part in parts) {
            append(part)
        }
    }.toString()
}

// 更好的做法：使用joinToString
fun buildMessageBest(parts: List<String>): String {
    return parts.joinToString("")
}
```

### 集合操作优化

**序列(Sequence)的使用:**
```kotlin
// 不好的做法：链式操作创建中间集合
val result = largeList
    .map { it * 2 }        // 创建中间集合1
    .filter { it > 100 }   // 创建中间集合2
    .take(10)              // 创建中间集合3

// 优化后的做法：使用序列
val resultOptimized = largeList.asSequence()
    .map { it * 2 }        // 惰性计算
    .filter { it > 100 }   // 惰性计算
    .take(10)              // 惰性计算
    .toList()              // 只创建最终集合

// 性能对比
fun benchmark() {
    val largeList = (1..1_000_000).toList()
    
    // 传统方式：约500ms
    val time1 = measureTimeMillis {
        val result1 = largeList.map { it * 2 }.filter { it > 100 }.take(10)
    }
    
    // 序列方式：约50ms
    val time2 = measureTimeMillis {
        val result2 = largeList.asSequence().map { it * 2 }.filter { it > 100 }.take(10).toList()
    }
    
    println("传统方式: ${time1}ms, 序列方式: ${time2}ms")
}
```

## 协程性能优化

### 调度器选择

**合理使用调度器:**
```kotlin
suspend fun processData() {
    // 错误的调度器使用
    withContext(Dispatchers.Main) {
        heavyComputation() // 在主线程执行耗时操作
    }
    
    // 正确的调度器使用
    val result = withContext(Dispatchers.Default) {
        heavyComputation() // 在后台线程执行
    }
    
    // 回到主线程更新UI
    withContext(Dispatchers.Main) {
        updateUI(result)
    }
}

// 优化IO操作
suspend fun fetchData(): String {

    }
```

### 协程取消处理

**正确处理协程取消:**
```kotlin
suspend fun heavyComputation(): Result {
    return withContext(Dispatchers.Default) {
        // 检查协程是否被取消
        ensureActive()
        
        // 执行第一步
        val step1 = performStep1()
        ensureActive() // 再次检查
        
        // 执行第二步
        val step2 = performStep2(step1)
        ensureActive() // 再次检查
        
        // 合并结果
        combineResults(step1, step2)
    }
}

// 使用协程取消回调
fun startWork(scope: CoroutineScope) {
    val job = scope.launch {
        try {
            heavyComputation()
        } finally {
            // 清理资源
            cleanupResources()
        }
    }
    
    // 5秒后取消
    scope.launch {
        delay(5000)
        job.cancel("Timeout")
    }
}
```

```kotlin
@Composable
fun OptimizedImage(
    url: String,
    contentDescription: String?,
    modifier: Modifier = Modifier

    val imageLoader = ImageLoader.Builder(LocalContext.current)

        model = ImageRequest.Builder(LocalContext.current)

            .size(Size.ORIGINAL)
            .build(),
        contentDescription = contentDescription,
        modifier = modifier,
        imageLoader = imageLoader
    )

// 使用占位符和错误处理
@Composable
fun ImageWithPlaceholder(
    url: String,
    contentDescription: String?,
    modifier: Modifier = Modifier
) {
    var imageState by remember { mutableStateOf<ImageState>(ImageState.Loading) }
    
    AsyncImage(
        model = url,
        contentDescription = contentDescription,
        modifier = modifier,
        onLoading = { imageState = ImageState.Loading },
        onSuccess = { imageState = ImageState.Success },
        onError = { imageState = ImageState.Error }
    )
    
    when (imageState) {
        is ImageState.Loading -> {
            Box(modifier = modifier.background(Color.LightGray))
        }
        is ImageState.Error -> {
            Icon(Icons.Default.Error, contentDescription)
        }
        else -> { /* 图片加载成功 */ }
    }
}

sealed class ImageState {
    object Loading : ImageState()
    object Success : ImageState()
    object Error : ImageState()
}
```

## 网络性能优化

### 请求优化

**批量请求和缓存:**
```kotlin
class OptimizedApiService {
    
    private val cache = LruCache<String, Any>(1024) // 1MB缓存
    
    suspend fun getUserWithCache(userId: String): User {
        val cacheKey = "user_$userId"
        
        // 检查缓存
        val cached = cache.get(cacheKey) as? User
        if (cached != null) {
            return cached
        }
        
        // 从网络获取
        val user = apiService.getUser(userId)
        
        // 更新缓存
        cache.put(cacheKey, user)
        
        return user
    }
    
    // 批量请求
    suspend fun getUsersBatch(userIds: List<String>): List<User> {
        return coroutineScope {
            userIds.map { userId ->
                async { getUserWithCache(userId) }
            }.awaitAll()
        }
    }
}
```

### 连接池优化

**OkHttp配置优化:**
```kotlin
fun createOptimizedHttpClient(): OkHttpClient {
    return OkHttpClient.Builder()
        .connectionPool(ConnectionPool(5, 5, TimeUnit.MINUTES)) // 连接池配置
        .connectTimeout(15, TimeUnit.SECONDS) // 连接超时
        .readTimeout(30, TimeUnit.SECONDS)    // 读取超时
        .writeTimeout(30, TimeUnit.SECONDS)   // 写入超时
        .retryOnConnectionFailure(true)      // 连接失败重试
        .addInterceptor(GzipRequestInterceptor()) // Gzip压缩
        .addInterceptor(CacheInterceptor())   // 缓存拦截器
        .cache(createCache())                 // 响应缓存
        .build()
}

// 缓存配置
private fun createCache(): Cache {
    val cacheSize = 10 * 1024 * 1024 // 10MB
    val cacheDir = File(context.cacheDir, "http_cache")
    return Cache(cacheDir, cacheSize.toLong())
}
```

```kotlin
class PerformanceMonitor {
    
    fun startTrace(tag: String) {
        Trace.beginSection(tag)
    }
    
    fun endTrace() {
        Trace.endSection()
    }
    
    // 自动跟踪函数执行
    inline fun <T> trace(tag: String, block: () -> T): T {
        startTrace(tag)
        try {
            return block()
        } finally {
            endTrace()
        }
    }
}
```

## 最佳实践总结

### 编码规范
1. **使用val代替var**：尽可能使用不可变变量
2. **避免使用!!操作符**：使用安全调用或Elvis操作符
3. **使用序列处理大数据集**：减少中间集合创建
4. **合理使用内联函数**：减少函数调用开销

### 内存管理
1. **及时释放资源**：使用use函数或try-with-resources
2. **避免内存泄漏**：正确使用生命周期感知组件
3. **优化图片加载**：使用合适的图片格式和尺寸

### 网络优化
1. **使用缓存策略**：减少重复网络请求
2. **批量处理请求**：减少连接建立开销
3. **压缩数据传输**：使用Gzip等压缩技术

### 数据库优化
1. **使用索引**：加速查询性能
2. **批量操作**：减少事务开销
3. **合理分页**：避免一次性加载大量数据

通过遵循这些优化策略，可以显著提升Kotlin应用的性能和用户体验。
