# Kotlin语法基础

## 变量声明

### 变量类型
```kotlin
// 不可变变量（推荐使用）
val name: String = "Kotlin"
val age = 25  // 类型推断

// 可变变量
var count: Int = 0
var score = 100.0  // 类型推断为Double

// 延迟初始化
lateinit var lateInitVar: String
```

### 空安全机制
```kotlin
// 非空类型（默认）
var nonNullString: String = "hello"
// nonNullString = null  // 编译错误

// 可空类型
var nullableString: String? = null

// 安全调用
val length1 = nullableString?.length  // 返回 Int?
val length2 = nullableString!!.length  // 非空断言，可能抛出异常

// Elvis操作符
val result = nullableString ?: "default"
val length3 = nullableString?.length ?: 0
```

## 数据类型

### 基本数据类型
```kotlin
// 数字类型
val intValue: Int = 42

val isTrue: Boolean = true

val raw = """
    Name: $name

```

## 控制流

### if表达式
```kotlin
// 传统用法
val max: Int
if (a > b) {
    max = a
} else {
    max = b
}

// 作为表达式
val max = if (a > b) a else b

// 多行表达式
val max = if (a > b) {
    println("Choose a")
    a
} else {
    println("Choose b")
    b
}
```

### when表达式
```kotlin
// 替代switch-case
when (x) {
    1 -> println("x == 1")
    2 -> println("x == 2")
    else -> println("x is neither 1 nor 2")
}

// 多个条件
when (x) {
    0, 1 -> println("x == 0 or x == 1")
    else -> println("otherwise")
}

// 范围判断
when (x) {
    in 1..10 -> println("x is in the range")
    !in 10..20 -> println("x is outside the range")
    else -> println("none of the above")
}

// 类型判断
when (obj) {
    is String -> println(obj.length)
    is Int -> println(obj + 1)
    else -> println("Unknown type")
}

// 无参数when
when {
    x.isOdd() -> println("x is odd")
    x.isEven() -> println("x is even")
    else -> println("x is funny")
}
```

### 循环
```kotlin
// for循环
for (i in 1..10) {
    print(i)
}

// 步长
for (i in 1..10 step 2) {
    print(i)
}

// 递减
for (i in 10 downTo 1) {
    print(i)
}

// 遍历集合
val list = listOf("a", "b", "c")
for (item in list) {
    println(item)
}

// 带索引遍历
for ((index, value) in list.withIndex()) {
    println("$index: $value")
}

// while循环
var x = 10
while (x > 0) {
    x--
}

// do-while循环
do {
    x++
} while (x < 10)
```

## 函数

### 函数定义
```kotlin
// 基本函数
fun sayHello(name: String): String {
    return "Hello, $name!"
}

// 单表达式函数
fun double(x: Int): Int = x * 2

// 类型推断的单表达式函数
fun triple(x: Int) = x * 3

// 默认参数
fun greet(name: String, msg: String = "Hello") {
    println("$msg, $name!")
}

// 命名参数
greet(name = "Alice", msg = "Hi")
greet(msg = "Hi", name = "Alice")  // 顺序可以改变

// 可变参数
fun printAll(vararg messages: String) {
    for (m in messages) println(m)
}

printAll("Hello", "Hallo", "Salut", "Hola", "你好")
```

### 高阶函数
```kotlin
// 函数类型参数
fun calculate(x: Int, y: Int, operation: (Int, Int) -> Int): Int {
    return operation(x, y)
}

// 使用
val result1 = calculate(10, 5) { a, b -> a + b }
val result2 = calculate(10, 5) { a, b -> a * b }

// 返回函数的函数
fun operationFactory(operation: String): (Int, Int) -> Int {
    return when (operation) {
        "add" -> { a, b -> a + b }
        "multiply" -> { a, b -> a * b }
        else -> { a, b -> a - b }
    }
}

val addFunction = operationFactory("add")
println(addFunction(5, 3))  // 输出 8
```

## 类与对象

### 类定义
```kotlin
// 简单类
class Person(val name: String, var age: Int)

// 带有方法的类
class Rectangle(val width: Int, val height: Int) {
    val area: Int
        get() = width * height
    
    fun isSquare() = width == height
}

// 主构造函数和初始化块
class InitOrderDemo(name: String) {
    val firstProperty = "First property: $name"
    
    init {
        println("First initializer block")
    }
    
    val secondProperty = "Second property: ${name.length}"
    
    init {
        println("Second initializer block")
    }
}
```

### 数据类
```kotlin
// 自动生成equals(), hashCode(), toString(), copy()
data class User(val name: String, val age: Int)

// 使用
val user1 = User("Alice", 30)
val user2 = User("Alice", 30)
println(user1 == user2)  // true
println(user1)  // User(name=Alice, age=30)

// 复制
val user3 = user1.copy(age = 31)
println(user3)  // User(name=Alice, age=31)

// 解构
val (name, age) = user1
println("$name is $age years old")
```

### 伴生对象
```kotlin
class MyClass {
    companion object {
        fun create(): MyClass = MyClass()
        const val CONSTANT = "constant"
    }
}

// 使用
val instance = MyClass.create()
println(MyClass.CONSTANT)
```

## 集合操作

### 集合创建
```kotlin
// 列表
val list = listOf("a", "b", "c")
val mutableList = mutableListOf("a", "b", "c")

// 集合
val set = setOf(1, 2, 3)
val mutableSet = mutableSetOf(1, 2, 3)

// 映射
val map = mapOf("a" to 1, "b" to 2, "c" to 3)
val mutableMap = mutableMapOf("a" to 1, "b" to 2)
```

### 集合操作
```kotlin
val numbers = listOf(1, 2, 3, 4, 5)

// 过滤
val evens = numbers.filter { it % 2 == 0 }

// 映射
val squares = numbers.map { it * it }

// 排序
val sorted = numbers.sorted()
val sortedDesc = numbers.sortedDescending()

// 分组
val grouped = numbers.groupBy { if (it % 2 == 0) "even" else "odd" }

// 折叠
val sum = numbers.fold(0) { acc, i -> acc + i }

// 链式操作
val result = numbers
    .filter { it > 2 }
    .map { it * 2 }
    .take(2)
    .joinToString(", ")
```

## 扩展函数

```kotlin
// 为String添加扩展函数
fun String.addExclamation(): String = this + "!"

// 使用
val greeting = "Hello".addExclamation()  // "Hello!"

// 扩展属性
val String.lastChar: Char
    get() = this[length - 1]

println("Kotlin".lastChar)  // n

// 可空类型的扩展
fun String?.isNullOrEmpty(): Boolean = this == null || this.isEmpty()

val nullableString: String? = null
println(nullableString.isNullOrEmpty())  // true
```

## 异常处理

```kotlin
// 基本异常处理
try {
    val number = "abc".toInt()
} catch (e: NumberFormatException) {
    println("Invalid number")
} finally {
    println("This block always executes")
}

// try作为表达式
val number = try {
    "123".toInt()
} catch (e: NumberFormatException) {
    -1
}

println(number)  // 123 或 -1
```

## 总结

Kotlin语法设计强调简洁性和安全性：
- **空安全**：编译时防止空指针异常
- **类型推断**：减少样板代码
- **表达式导向**：if、when、try都是表达式
- **扩展函数**：为现有类添加新功能
- **函数式特性**：Lambda、高阶函数、集合操作

这些特性使得Kotlin代码既安全又表达力强，是Android开发的理想选择。
