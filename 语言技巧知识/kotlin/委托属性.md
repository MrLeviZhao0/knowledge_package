# Kotlin委托属性

## 概述
委托属性是Kotlin的强大特性，允许将属性的getter和setter委托给另一个对象。这种机制可以简化代码、实现重用逻辑，并支持各种高级模式如惰性初始化、观察者模式等。

## 基础语法

### 属性委托语法
```kotlin
import kotlin.properties.ReadWriteProperty
import kotlin.reflect.KProperty

// 自定义委托类
class ExampleDelegate : ReadWriteProperty<Any?, String> {
    private var storedValue: String = "Default"
    
    override fun getValue(thisRef: Any?, property: KProperty<*>): String {
        println("Getting value: $storedValue")
        return storedValue
    }
    
    override fun setValue(thisRef: Any?, property: KProperty<*>, value: String) {
        println("Setting value: $value")
        storedValue = value
    }
}

class Example {
    var delegatedProperty: String by ExampleDelegate()
}

fun main() {
    val example = Example()
    
    example.delegatedProperty = "Hello"  // 输出: "Setting value: Hello"
    println(example.delegatedProperty)    // 输出: "Getting value: Hello" 然后 "Hello"
}
```

## 标准委托

### lazy 惰性初始化
```kotlin
class HeavyObject {
    init {
        println("HeavyObject created")
    }
    
    fun doWork() {
        println("Working...")
    }
}

class LazyExample {
    // 默认线程安全（LazyThreadSafetyMode.SYNCHRONIZED）
    val heavyObject1: HeavyObject by lazy {
        println("Initializing heavyObject1")
        HeavyObject()
    }
    
    // 非线程安全（用于单线程环境）
    val heavyObject2: HeavyObject by lazy(LazyThreadSafetyMode.NONE) {
        println("Initializing heavyObject2")
        HeavyObject()
    }
    
    // 使用自定义锁
    val heavyObject3: HeavyObject by lazy(mode = LazyThreadSafetyMode.PUBLICATION) {
        println("Initializing heavyObject3")
        HeavyObject()
    }
}

fun main() {
    val example = LazyExample()
    println("LazyExample created")
    
    // 第一次访问时初始化
    example.heavyObject1.doWork() // 输出初始化信息
    example.heavyObject1.doWork() // 不再初始化
}
```

### observable 可观察属性
```kotlin
import kotlin.properties.Delegates

class ObservableExample {
    var name: String by Delegates.observable("<no name>") { prop, old, new ->
        println("Property ${prop.name} changed from '$old' to '$new'")
    }
    
    var age: Int by Delegates.observable(0) { _, old, new ->
        println("Age changed from $old to $new")
    }
}

fun main() {
    val example = ObservableExample()
    
    example.name = "Alice"    // 输出: "Property name changed from '<no name>' to 'Alice'"
    example.age = 25          // 输出: "Age changed from 0 to 25"
    example.name = "Bob"      // 输出: "Property name changed from 'Alice' to 'Bob'"
}
```

### vetoable 可否决的属性
```kotlin
class VetoableExample {
    var positiveNumber: Int by Delegates.vetoable(0) { _, old, new ->
        println("Trying to change from $old to $new")
        new >= 0 // 只有新值非负时才允许更改
    }
    
    var evenNumber: Int by Delegates.vetoable(0) { _, _, new ->
        new % 2 == 0 // 只有偶数才允许
    }
}

fun main() {
    val example = VetoableExample()
    
    example.positiveNumber = 10  // 允许更改
    println("positiveNumber: ${example.positiveNumber}") // 10
    
    example.positiveNumber = -5  // 否决更改
    println("positiveNumber: ${example.positiveNumber}") // 10（保持不变）
    
    example.evenNumber = 4       // 允许更改
    example.evenNumber = 3       // 否决更改
    println("evenNumber: ${example.evenNumber}") // 4
}
```

## 自定义委托模式

### 缓存委托
```kotlin
class CacheDelegate<T>(private val compute: () -> T) : ReadOnlyProperty<Any?, T> {
    private var cachedValue: T? = null
    
    override fun getValue(thisRef: Any?, property: KProperty<*>): T {
        return cachedValue ?: compute().also { cachedValue = it }
    }
}

class CacheExample {
    val expensiveComputation: String by CacheDelegate {
        println("Performing expensive computation...")
        Thread.sleep(1000) // 模拟耗时计算
        "Computed Result"
    }
}

fun main() {
    val example = CacheExample()
    
    println("First access:")
    println(example.expensiveComputation) // 执行计算并缓存
    
    println("Second access:")
    println(example.expensiveComputation) // 直接返回缓存值
}
```

### 验证委托
```kotlin
class ValidatedDelegate<T>(private val validator: (T) -> Boolean) : ReadWriteProperty<Any?, T> {
    private var value: T? = null
    
    override fun getValue(thisRef: Any?, property: KProperty<*>): T {
        return value ?: throw IllegalStateException("Property ${property.name} not initialized")
    }
    
    override fun setValue(thisRef: Any?, property: KProperty<*>, value: T) {
        if (!validator(value)) {
            throw IllegalArgumentException("Invalid value for ${property.name}: $value")
        }
        this.value = value
    }
}

class User {
    var name: String by ValidatedDelegate { it.isNotBlank() }
    var age: Int by ValidatedDelegate { it in 0..150 }
    var email: String by ValidatedDelegate { it.contains("@") }
}

fun main() {
    val user = User()
    
    user.name = "Alice"      // 有效
    user.age = 25            // 有效
    user.email = "alice@example.com" // 有效
    
    try {
        user.name = ""       // 抛出异常
    } catch (e: IllegalArgumentException) {
        println(e.message)
    }
    
    try {
        user.age = 200       // 抛出异常
    } catch (e: IllegalArgumentException) {
        println(e.message)
    }
}
```

### 映射委托
```kotlin
class User(val map: Map<String, Any?>) {
    val name: String by map
    val age: Int by map
    val email: String by map
}

class MutableUser(val map: MutableMap<String, Any?>) {
    var name: String by map
    var age: Int by map
    var email: String by map
}

fun main() {
    // 只读映射委托
    val user = User(mapOf(
        "name" to "Alice",
        "age" to 25,
        "email" to "alice@example.com"
    ))
    
    println("${user.name}, ${user.age}, ${user.email}")
    
    // 可变映射委托
    val mutableMap = mutableMapOf(
        "name" to "Bob",
        "age" to 30
    )
    
    val mutableUser = MutableUser(mutableMap)
    println("Before: ${mutableUser.name}, ${mutableUser.age}")
    
    mutableUser.name = "Charlie"
    mutableUser.age = 35
    
    println("After: ${mutableUser.name}, ${mutableUser.age}")
    println("Map: $mutableMap") // 映射也被更新
}
```

### 属性委托组合
```kotlin
class CombinedDelegate<T>(
    private var value: T,
    private val validators: List<(T) -> Boolean> = emptyList(),
    private val onChange: ((T, T) -> Unit)? = null
) : ReadWriteProperty<Any?, T> {
    
    override fun getValue(thisRef: Any?, property: KProperty<*>): T = value
    
    override fun setValue(thisRef: Any?, property: KProperty<*>, newValue: T) {
        val oldValue = value
        
        // 验证
        validators.forEach { validator ->
            if (!validator(newValue)) {
                throw IllegalArgumentException("Validation failed for ${property.name}")
            }
        }
        
        value = newValue
        onChange?.invoke(oldValue, newValue)
    }
}

class AdvancedExample {
    var validatedNumber: Int by CombinedDelegate(
        initialValue = 0,
        validators = listOf(
            { it >= 0 },
            { it <= 100 }
        ),
        onChange = { old, new ->
            println("Number changed from $old to $new")
        }
    )
    
    var nonEmptyString: String by CombinedDelegate(
        initialValue = "",
        validators = listOf { it.isNotBlank() }
    )
}

fun main() {
    val example = AdvancedExample()
    
    example.validatedNumber = 50  // 有效更改
    example.nonEmptyString = "Hello" // 有效更改
    
    try {
        example.validatedNumber = -5 // 验证失败
    } catch (e: IllegalArgumentException) {
        println(e.message)
    }
    
    try {
        example.nonEmptyString = "" // 验证失败
    } catch (e: IllegalArgumentException) {
        println(e.message)
    }
}
```

## 实际应用场景

### Android开发中的委托
```kotlin
// 简化SharedPreferences访问
class PreferenceDelegate<T>(
    private val preferences: SharedPreferences,
    private val key: String,
    private val defaultValue: T,
    private val serializer: (T) -> String = { it.toString() },
    private val deserializer: (String) -> T = { it as T }
) : ReadWriteProperty<Any?, T> {
    
    override fun getValue(thisRef: Any?, property: KProperty<*>): T {
        val stringValue = preferences.getString(key, null)
        return stringValue?.let { deserializer(it) } ?: defaultValue
    }
    
    override fun setValue(thisRef: Any?, property: KProperty<*>, value: T) {
        preferences.edit().putString(key, serializer(value)).apply()
    }
}

// 扩展函数简化使用
fun <T> SharedPreferences.delegate(
    key: String,
    defaultValue: T,
    serializer: (T) -> String = { it.toString() },
    deserializer: (String) -> T = { it as T }
): ReadWriteProperty<Any?, T> {
    return PreferenceDelegate(this, key, defaultValue, serializer, deserializer)
}

// 在Activity中使用
class SettingsActivity : AppCompatActivity() {
    private val prefs by lazy {
        getSharedPreferences("settings", Context.MODE_PRIVATE)
    }
    
    var userName: String by prefs.delegate("user_name", "Guest")
    var notificationsEnabled: Boolean by prefs.delegate("notifications", true)
    var themeColor: Int by prefs.delegate("theme_color", 0xFF0000.toInt())
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        // 自动从SharedPreferences读取
        println("User: $userName")
        println("Notifications: $notificationsEnabled")
        
        // 自动保存到SharedPreferences
        userName = "Alice"
        notificationsEnabled = false
    }
}
```

### 视图绑定委托
```kotlin
// 简化View查找
class ViewDelegate<T : View>(
    private val activity: Activity,
    private val viewId: Int
) : ReadOnlyProperty<Activity, T> {
    
    private var view: T? = null
    
    override fun getValue(thisRef: Activity, property: KProperty<*>): T {
        return view ?: thisRef.findViewById<T>(viewId).also { view = it }
    }
}

// 扩展函数
fun <T : View> Activity.viewDelegate(viewId: Int): ReadOnlyProperty<Activity, T> {
    return ViewDelegate(this, viewId)
}

// 在Activity中使用
class MainActivity : AppCompatActivity() {
    private val button: Button by viewDelegate(R.id.button)
    private val textView: TextView by viewDelegate(R.id.text_view)
    private val recyclerView: RecyclerView by viewDelegate(R.id.recycler_view)
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        
        // 自动查找视图
        button.setOnClickListener {
            textView.text = "Button clicked!"
        }
    }
}
```

### 资源访问委托

// 简化资源访问
class ResourceDelegate<T>(
    private val context: Context,
    private val resourceFunc: Context.() -> T
) : ReadOnlyProperty<Any?, T> {
    
    private var value: T? = null
    
    override fun getValue(thisRef: Any?, property: KProperty<*>): T {
        return value ?: context.resourceFunc().also { value = it }
    }
}

// 扩展函数
fun <T> Context.resourceDelegate(resourceFunc: Context.() -> T): ReadOnlyProperty<Any?, T> {
    return ResourceDelegate(this, resourceFunc)
}

// 在Activity中使用
class ResourceActivity : AppCompatActivity() {
    private val appName: String by resourceDelegate { getString(R.string.app_name) }
    private val primaryColor: Int by resourceDelegate { getColor(R.color.primary) }
    private val appIcon: Drawable by resourceDelegate { getDrawable(R.drawable.ic_launcher)!! }
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        println("App: $appName")
        println("Color: #${primaryColor.toHexString()}")
    }
    
    private fun Int.toHexString(): String {
        return "%06X".format(this and 0xFFFFFF)
    }
}

## 测试委托属性

### 单元测试
```kotlin
import org.junit.Test
import kotlin.test.assertEquals
import kotlin.test.assertFailsWith

class DelegatePropertiesTest {
    
    @Test
    fun `test observable delegate`() {
        var changeCount = 0
        var lastChange: Pair<String, String>? = null
        
        class TestClass {
            var observedProperty: String by Delegates.observable("initial") { _, old, new ->
                changeCount++
                lastChange = old to new
            }
        }
        
        val instance = TestClass()
        
        instance.observedProperty = "first"
        assertEquals(1, changeCount)
        assertEquals("initial" to "first", lastChange)
        
        instance.observedProperty = "second"
        assertEquals(2, changeCount)
        assertEquals("first" to "second", lastChange)
    }
    
    @Test
    fun `test vetoable delegate`() {
        class TestClass {
            var onlyPositive: Int by Delegates.vetoable(0) { _, _, new ->
                new >= 0
            }
        }
        
        val instance = TestClass()
        
        instance.onlyPositive = 10
        assertEquals(10, instance.onlyPositive)
        
        instance.onlyPositive = -5
        assertEquals(10, instance.onlyPositive) // 值未改变
    }
    
    @Test
    fun `test custom delegate`() {
        class TestDelegate : ReadWriteProperty<Any?, String> {
            private var value: String = ""
            var getCount = 0
            var setCount = 0
            
            override fun getValue(thisRef: Any?, property: KProperty<*>): String {
                getCount++
                return value
            }
            
            override fun setValue(thisRef: Any?, property: KProperty<*>, value: String) {
                setCount++
                this.value = value
            }
        }
        
        class TestClass {
            var delegatedProperty: String by TestDelegate()
        }
        
        val instance = TestClass()
        val delegate = instance::delegatedProperty.getDelegate() as TestDelegate
        
        instance.delegatedProperty = "test"
        assertEquals("test", instance.delegatedProperty)
        assertEquals(1, delegate.setCount)
        assertEquals(1, delegate.getCount)
    }
}
```

## 总结
Kotlin委托属性提供了强大的机制来：

1. **简化代码**：减少样板代码，提高可读性
2. **实现重用**：通用逻辑可以在多个属性间共享
3. **增强功能**：添加验证、日志、缓存等额外功能
4. **支持模式**：实现观察者、惰性初始化等设计模式

使用委托属性时要注意：
- 选择合适的标准委托或创建自定义委托
- 考虑性能影响，必要时使用内联或缓存
- 保持委托的单一职责和可测试性
- 在Android等特定平台中利用委托简化常见任务
