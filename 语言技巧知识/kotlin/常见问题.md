## 语法和基础问题

### 空安全相关问题

**问题1: 如何处理可能为null的变量？**

```kotlin
// 错误做法：直接使用可能为null的变量
val name: String? = getUserName()
println(name.length) // 编译错误：可能为null

// 正确做法：使用安全调用操作符
val length = name?.length // 如果name为null，返回null

// 使用Elvis操作符提供默认值
val safeLength = name?.length ?: 0

// 使用非空断言（谨慎使用）
val forcedLength = name!!.length // 如果name为null，抛出异常

// 使用let进行空安全处理
name?.let { 
    println("Name length: ${it.length}")
}
```

**问题2: 如何将可空类型转换为非空类型？**

```kotlin
val nullableString: String? = getNullableString()

// 安全转换
val nonNullString: String = nullableString ?: "default"

// 使用requireNotNull或checkNotNull
val requiredString = requireNotNull(nullableString) { "String cannot be null" }

// 在初始化时确保非空
class UserService {
    private val repository: UserRepository by lazy {
        requireNotNull(userRepository) { "Repository must be initialized" }
    }
}
```

### 集合操作问题

**问题3: List和MutableList的区别？**

```kotlin
// List是不可变的
val immutableList: List<String> = listOf("A", "B", "C")
// immutableList.add("D") // 编译错误

// MutableList是可变的
val mutableList: MutableList<String> = mutableListOf("A", "B", "C")
mutableList.add("D") // 正确

// 转换
val convertedList: List<String> = mutableList.toList() // 创建不可变副本
val convertedMutable: MutableList<String> = immutableList.toMutableList()
```

**问题4: 如何避免集合操作中的性能问题？**

```kotlin
// 不好的做法：链式操作创建中间集合
val result = largeList
    .map { it * 2 }        // 中间集合1
    .filter { it > 100 }   // 中间集合2
    .take(10)              // 中间集合3

// 好的做法：使用序列
val optimizedResult = largeList.asSequence()
    .map { it * 2 }        // 惰性计算
    .filter { it > 100 }   // 惰性计算
    .take(10)              // 惰性计算
    .toList()              // 只创建最终集合
```

**问题9: 如何处理协程取消？**

```kotlin
suspend fun processData(): Result {
    return withContext(Dispatchers.IO) {
        // 检查协程是否被取消
        ensureActive()
        
        val step1 = performStep1()
        ensureActive() // 再次检查
        
        val step2 = performStep2(step1)
        ensureActive() // 再次检查
        
        combineResults(step1, step2)
    }
}

// 使用协程取消回调
fun startWork(scope: CoroutineScope) {
    val job = scope.launch {
        try {
            processData()
        } catch (e: CancellationException) {
            // 协程被取消
            Log.d("Coroutine", "Work cancelled")
        } finally {
            // 清理资源
            cleanup()
        }
    }
    
    // 超时取消
    scope.launch {
        delay(5000)
        job.cancel("Timeout")
    }
}
```

### 结构化并发问题

**问题10: 如何避免协程泄漏？**

```kotlin
// 不好的做法：使用GlobalScope
class BadRepository {
    fun loadData() {
        GlobalScope.launch {
            // 协程可能永远不会被取消
            fetchData()
        }
    }
}

// 好的做法：使用viewModelScope或lifecycleScope
class GoodRepository {
    fun loadData(scope: CoroutineScope) {
        scope.launch {
            // 当scope被取消时，协程自动取消
            fetchData()
        }
    }
}

// 在ViewModel中使用
class MyViewModel : ViewModel() {
    fun loadData() {
        viewModelScope.launch {
            // 当ViewModel被清除时自动取消
            repository.loadData(this)
        }
    }
}
```

```kotlin
    private val binding get() = _binding!!
    
    override fun onCreateView(inflater: LayoutInflater, container: ViewGroup?, savedInstanceState: Bundle?): View {
        _binding = FragmentMyBinding.inflate(inflater, container, false)
```

**问题12: 如何处理权限请求？**

```kotlin
class PermissionActivity : AppCompatActivity() {
    
    private val permissionLauncher = registerForActivityResult(
        ActivityResultContracts.RequestPermission()
    ) { isGranted ->
        if (isGranted) {
            // 权限已授予
            startCamera()
        } else {
            // 权限被拒绝
            showPermissionDeniedDialog()
        }
    }
    
    fun requestCameraPermission() {
        when {
            ContextCompat.checkSelfPermission(this, Manifest.permission.CAMERA) == 
                PackageManager.PERMISSION_GRANTED -> {
                // 已有权限
                startCamera()
            }
            ActivityCompat.shouldShowRequestPermissionRationale(this, Manifest.permission.CAMERA) -> {
                // 需要解释为什么需要权限
                showRationaleDialog()
            }
            else -> {
                // 请求权限
                permissionLauncher.launch(Manifest.permission.CAMERA)
            }
        }
    }
    
    private fun showRationaleDialog() {
        AlertDialog.Builder(this)
            .setTitle("需要相机权限")
            .setMessage("此功能需要访问相机才能拍照")
            .setPositiveButton("确定") { _, _ ->
                permissionLauncher.launch(Manifest.permission.CAMERA)
            }
            .setNegativeButton("取消", null)
            .show()
    }
}
```

```kotlin
@Composable
fun OptimizedUserList(users: List<User>) {

fun UserItem(user: User) {
```

```kotlin
fun CounterScreen() {

) {

// 使用ViewModel管理状态
@Composable
fun UserProfileScreen(viewModel: UserViewModel = viewModel()) {
    val userState by viewModel.userState.collectAsState()
    
    when (userState) {
        is UserState.Loading -> LoadingIndicator()
        is UserState.Success -> {
            val user = (userState as UserState.Success).user
            UserProfileContent(user = user)
        }
        is UserState.Error -> {
            val error = (userState as UserState.Error).message
            ErrorScreen(error = error)
        }
    }
}
```

```kotlin
class MainActivity : AppCompatActivity() {
```

```kotlin
fun OptimizedImage(

@Composable
fun ImageWithState(url: String, contentDescription: String?) {
```

**问题18: 如何编写单元测试？**

```kotlin
class CalculatorTest {
    
    private val calculator = Calculator()
    
    @Test
    fun `addition should return correct result`() {
        // Given
        val a = 5
        val b = 3
        
        // When
        val result = calculator.add(a, b)
        
        // Then
        assertEquals(8, result)
    }
    
    @Test
    fun `division by zero should throw exception`() {
        // Given
        val a = 10
        val b = 0
        
        // When & Then
        assertThrows<ArithmeticException> {
            calculator.divide(a, b)
        }
    }
    
    @Test
    fun `async operation should complete successfully`() = runTest {
        // Given
        val expectedResult = "result"
        
        // When
        val result = calculator.asyncOperation()
        
        // Then
        assertEquals(expectedResult, result)
    }
}

class Calculator {
    fun add(a: Int, b: Int) = a + b
    
    fun divide(a: Int, b: Int) = a / b
    
    suspend fun asyncOperation(): String {
        delay(1000) // 模拟异步操作
        return "result"
    }
}
```
