# 主题和动画

## 概述
Compose提供了强大的主题系统和动画框架，可以创建一致、美观且具有动态效果的UI。本章详细介绍Material Design主题定制和动画实现。

## Material Design主题

```kotlin
@Composable
fun ThemeSwitcherApp() {
    var isDarkTheme by remember { mutableStateOf(false) }
    
    MaterialTheme(
        colors = if (isDarkTheme) DarkColorPalette else LightColorPalette

        Column {
```

### 值动画

```kotlin
@Composable
fun AnimatedCounter() {
    var count by remember { mutableStateOf(0) }
    
    // 使用animateIntAsState创建平滑的数字变化
    val animatedCount by animateIntAsState(
        targetValue = count,
        animationSpec = tween(durationMillis = 1000)
    )
    
    Column {
        Text("计数: $animatedCount", style = MaterialTheme.typography.h4)
        
        Button(onClick = { count++ }) {
            Text("增加")
        }
        
        Button(onClick = { count = 0 }) {
            Text("重置")
        }
    }
}
```

## 高级动画

### 过渡动画

```kotlin
@Composable
fun CrossfadeExample() {
    var currentScreen by remember { mutableStateOf("Home") }
    
    Column {
        Row {
            Button(onClick = { currentScreen = "Home" }) {
                Text("首页")
            }
            Button(onClick = { currentScreen = "Profile" }) {
                Text("个人资料")
            }
            Button(onClick = { currentScreen = "Settings" }) {
                Text("设置")
            }
        }
        
        Crossfade(targetState = currentScreen) { screen ->
            when (screen) {
                "Home" -> HomeScreen()
                "Profile" -> ProfileScreen()
                "Settings" -> SettingsScreen()
            }
        }
    }
}
```

### 自定义动画

```kotlin
@Composable
fun CustomAnimationExample() {
    var enabled by remember { mutableStateOf(false) }
    
    // 自定义动画规格
    val animationSpec = spring<Float>(
        dampingRatio = Spring.DampingRatioMediumBouncy,
        stiffness = Spring.StiffnessLow
    )
    
    val animatedScale by animateFloatAsState(
        targetValue = if (enabled) 1.5f else 1f,
        animationSpec = animationSpec
    )
    
    val animatedRotation by animateFloatAsState(
        targetValue = if (enabled) 45f else 0f,
        animationSpec = animationSpec
    )
    
    Column {
        Box(
            modifier = Modifier
                .size(100.dp)
                .background(Color.Blue)
                .scale(animatedScale)
                .rotate(animatedRotation)
        )
        
        Button(onClick = { enabled = !enabled }) {
            Text(if (enabled) "缩小" else "放大")
        }
    }
}
```

## 手势动画

### 拖拽动画

```kotlin
@Composable
fun DraggableBox() {
    var offsetX by remember { mutableStateOf(0f) }
    var offsetY by remember { mutableStateOf(0f) }
    
    Box(
        modifier = Modifier
            .size(100.dp)
            .offset { IntOffset(offsetX.roundToInt(), offsetY.roundToInt()) }
            .background(Color.Red)
            .pointerInput(Unit) {
                detectDragGestures { change, dragAmount ->
                    change.consume()
                    offsetX += dragAmount.x
                    offsetY += dragAmount.y
                }
            }
    )
}
```

### 缩放和旋转手势

```kotlin
@Composable
fun TransformableBox() {
    var scale by remember { mutableStateOf(1f) }
    var rotation by remember { mutableStateOf(0f) }
    var offset by remember { mutableStateOf(Offset.Zero) }
    
    Box(
        modifier = Modifier
            .graphicsLayer(
                scaleX = scale,
                scaleY = scale,
                rotationZ = rotation,
                translationX = offset.x,
                translationY = offset.y
            )
            .size(100.dp)
            .background(Color.Green)
            .transformable(
                state = rememberTransformableState { zoomChange, offsetChange, rotationChange ->
                    scale *= zoomChange
                    rotation += rotationChange
                    offset += offsetChange
                }
            )
    )
}
```

## 复杂动画组合

### 交错动画

```kotlin
@Composable
fun StaggeredAnimation() {
    val items = listOf("Item 1", "Item 2", "Item 3", "Item 4", "Item 5")
    var visible by remember { mutableStateOf(false) }
    
    Column {
        Button(onClick = { visible = !visible }) {
            Text(if (visible) "隐藏" else "显示")
        }
        
        items.forEachIndexed { index, item ->
            AnimatedVisibility(
                visible = visible,
                enter = fadeIn() + slideInVertically(
                    initialOffsetY = { fullHeight -> fullHeight },
                    animationSpec = tween(
                        durationMillis = 300,
                        delayMillis = index * 100
                    )
                ),
                exit = fadeOut() + slideOutVertically(
                    targetOffsetY = { fullHeight -> fullHeight },
                    animationSpec = tween(
                        durationMillis = 300,
                        delayMillis = index * 100
                    )
                )
            ) {
                Text(
                    text = item,
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(8.dp)
                )
            }
        }
    }
}
```

```kotlin
@Composable
fun InfinitePulse() {
    val infiniteTransition = rememberInfiniteTransition()
    
    val pulse by infiniteTransition.animateFloat(
        initialValue = 1f,
        targetValue = 1.2f,
        animationSpec = infiniteRepeatable(
            animation = keyframes {
                durationMillis = 1000
                1.2f at 500
            },
            repeatMode = RepeatMode.Reverse
        )
    )
    
    Box(
        modifier = Modifier
            .size(100.dp)
            .graphicsLayer(scaleX = pulse, scaleY = pulse)
            .background(Color.Magenta)
    )
}
```

## 动画性能优化

### 使用derivedStateOf优化

```kotlin
@Composable
fun OptimizedAnimation() {
    var items by remember { mutableStateOf(listOf<String>()) }
    
    // 使用derivedStateOf避免不必要的动画计算
    val animatedItems by remember(items) {
        derivedStateOf {
            items.map { item ->
                AnimatedItem(item)
            }
        }
    }
    
    Column {
        Button(onClick = { 
            items = items + "Item ${items.size + 1}" 
        }) {
            Text("添加项目")
        }
        
        LazyColumn {
            items(animatedItems) { animatedItem ->
                AnimatedVisibility(
                    visible = true,
                    enter = fadeIn() + slideInVertically()
                ) {
                    Text(animatedItem.content)
                }
            }
        }
    }
}

data class AnimatedItem(val content: String)
```

### 动画规格优化

```kotlin
@Composable
fun OptimizedAnimationSpec() {
    var enabled by remember { mutableStateOf(false) }
    
    // 使用keyframes实现精确控制
    val animationSpec = keyframes<Float> {
        durationMillis = 1000
        0.0f at 0
        0.5f at 250
        1.0f at 500
        0.7f at 750
        1.0f at 1000
    }
    
    val animatedValue by animateFloatAsState(
        targetValue = if (enabled) 1f else 0f,
        animationSpec = animationSpec
    )
    
    Box(
        modifier = Modifier
            .size(100.dp)
            .graphicsLayer(alpha = animatedValue)
            .background(Color.Cyan)
    )
    
    Button(onClick = { enabled = !enabled }) {
        Text("切换动画")
    }
}
```

### 响应式设计

```kotlin
@Composable
fun ResponsiveLayout() {
    val configuration = LocalConfiguration.current
    val screenWidth = configuration.screenWidthDp.dp
    
    // 根据屏幕宽度调整布局
    val columnCount = when {
        screenWidth < 600.dp -> 1
        screenWidth < 840.dp -> 2
        else -> 3
    }
    
    LazyVerticalGrid(
        columns = GridCells.Fixed(columnCount)
    ) {
        items(10) { index ->
            ThemedCard {
                Text("Item $index")
            }
        }
    }
}
```
