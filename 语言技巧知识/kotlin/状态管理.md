# 状态管理

## 概述
状态管理是Compose应用的核心，决定了UI如何响应数据变化。正确的状态管理策略可以提高应用性能、可维护性和可测试性。

## 基础状态管理

### 可变状态

```kotlin
@Composable
fun Counter() {
    // 使用remember保持状态跨重组
    var count by remember { mutableStateOf(0) }
    
    Column {
        Text("Count: $count")
        Button(onClick = { count++ }) {
            Text("Increment")
        }
    }
}
```

### 状态提升
将状态提升到调用者，提高组件的可复用性：

```kotlin
@Composable
fun CounterScreen() {
    var count by remember { mutableStateOf(0) }
    
    Counter(
        count = count,
        onIncrement = { count++ },
        onDecrement = { count-- }
    )
}

@Composable
fun Counter(
    count: Int,
    onIncrement: () -> Unit,
    onDecrement: () -> Unit
) {
    Row {
        Button(onClick = onDecrement) { Text("-") }
        Text(text = "Count: $count")
        Button(onClick = onIncrement) { Text("+") }
    }
}
```

```kotlin
class CounterViewModel : ViewModel() {
```

## 复杂状态管理

### 表单状态管理

```kotlin
data class FormState(
    val username: String = "",
    val email: String = "",
    val password: String = "",
    val confirmPassword: String = "",
    val errors: Map<String, String> = emptyMap()
)

class FormViewModel : ViewModel() {
    private val _formState = MutableStateFlow(FormState())
    val formState: StateFlow<FormState> = _formState.asStateFlow()
    
    fun updateField(field: String, value: String) {
        _formState.update { currentState ->
            currentState.copy(
                when (field) {
                    "username" -> currentState.copy(username = value)
                    "email" -> currentState.copy(email = value)
                    "password" -> currentState.copy(password = value)
                    "confirmPassword" -> currentState.copy(confirmPassword = value)
                    else -> currentState
                }
            )
        }
    }
    
    fun validateForm(): Boolean {
        val errors = mutableMapOf<String, String>()
        val currentState = _formState.value
        
        if (currentState.username.isEmpty()) {
            errors["username"] = "Username is required"
        }
        
        if (currentState.email.isEmpty() || !isValidEmail(currentState.email)) {
            errors["email"] = "Valid email is required"
        }
        
        if (currentState.password.length < 8) {
            errors["password"] = "Password must be at least 8 characters"
        }
        
        if (currentState.password != currentState.confirmPassword) {
            errors["confirmPassword"] = "Passwords do not match"
        }
        
        _formState.update { it.copy(errors = errors) }
        return errors.isEmpty()
    }
}
```

### 列表状态管理

```kotlin
data class TodoItem(
    val id: String = UUID.randomUUID().toString(),
    val text: String,
    val isCompleted: Boolean = false,
    val createdAt: Long = System.currentTimeMillis()
)

class TodoViewModel : ViewModel() {
    private val _todos = MutableStateFlow<List<TodoItem>>(emptyList())
    val todos: StateFlow<List<TodoItem>> = _todos.asStateFlow()
    
    fun addTodo(text: String) {
        if (text.isNotBlank()) {
            val newTodo = TodoItem(text = text)
            _todos.update { currentTodos ->
                currentTodos + newTodo
            }
        }
    }
    
    fun toggleTodo(id: String) {
        _todos.update { currentTodos ->
            currentTodos.map { todo ->
                if (todo.id == id) {
                    todo.copy(isCompleted = !todo.isCompleted)
                } else {
                    todo
                }
            }
        }
    }
    
    fun deleteTodo(id: String) {
        _todos.update { currentTodos ->
            currentTodos.filter { it.id != id }
        }
    }
    
    fun clearCompleted() {
        _todos.update { currentTodos ->
            currentTodos.filter { !it.isCompleted }
        }
    }
}

@Composable
fun TodoScreen(viewModel: TodoViewModel = viewModel()) {
    val todos by viewModel.todos.collectAsState()
    var newTodoText by remember { mutableStateOf("") }
    
    Column {
        // 添加新待办事项
        Row {
            TextField(
                value = newTodoText,
                onValueChange = { newTodoText = it },
                placeholder = { Text("Add a new todo...") },
                modifier = Modifier.weight(1f)
            )
            Button(
                onClick = {
                    viewModel.addTodo(newTodoText)
                    newTodoText = ""
                }
            ) {
                Text("Add")
            }
        }
        
        // 待办事项列表
        LazyColumn {
            items(todos, key = { it.id }) { todo ->
                TodoItemRow(
                    todo = todo,
                    onToggle = { viewModel.toggleTodo(todo.id) },
                    onDelete = { viewModel.deleteTodo(todo.id) }
                )
            }
        }
        
        // 统计信息
        val completedCount = todos.count { it.isCompleted }
        Text("Completed: $completedCount/${todos.size}")
        
        if (completedCount > 0) {
            Button(onClick = { viewModel.clearCompleted() }) {
                Text("Clear Completed")
            }
        }
    }
}
```

## 高级状态管理技术

### 派生状态

```kotlin
@Composable
fun TodoStats(todos: List<TodoItem>) {
    // 使用derivedStateOf避免不必要的重组
    val stats by remember(todos) {
        derivedStateOf {
            val total = todos.size
            val completed = todos.count { it.isCompleted }
            val percentage = if (total > 0) completed.toFloat() / total else 0f
            
            Stats(total, completed, percentage)
        }
    }
    
    Column {
        Text("Total: ${stats.total}")
        Text("Completed: ${stats.completed}")
        Text("Progress: ${(stats.percentage * 100).toInt()}%")
        
        LinearProgressIndicator(
            progress = stats.percentage,
            modifier = Modifier.fillMaxWidth()
        )
    }
}

data class Stats(val total: Int, val completed: Int, val percentage: Float)
```

```kotlin
@Composable
fun ComplexStateWithSaver() {
    val complexState = rememberSaveable(
        saver = listSaver<
            save = { listOf(it.value1, it.value2) },
            restore = { ComplexState(it[0] as Int, it[1] as String) }
        >
    ) { mutableStateOf(ComplexState(0, "")) }
    
    // 使用complexState.value
}
```

## 架构模式

### MVVM模式

```kotlin
// Model
data class User(
    val id: String,
    val name: String,
    val email: String,
    val avatarUrl: String?
)

// Repository
class UserRepository {
    suspend fun getUser(id: String): User {
        // 从网络或数据库获取用户数据
        return User(id, "John Doe", "john@example.com", null)
    }
    
    suspend fun updateUser(user: User): Boolean {
        // 更新用户数据
        return true
    }
}

// ViewModel
class UserProfileViewModel(
    private val userRepository: UserRepository
) : ViewModel() {
    private val _userState = MutableStateFlow<UserState>(UserState.Loading)
    val userState: StateFlow<UserState> = _userState.asStateFlow()
    
    fun loadUser(userId: String) {
        viewModelScope.launch {
            _userState.value = UserState.Loading
            try {
                val user = userRepository.getUser(userId)
                _userState.value = UserState.Success(user)
            } catch (e: Exception) {
                _userState.value = UserState.Error(e.message ?: "Unknown error")
            }
        }
    }
    
    fun updateUserName(userId: String, newName: String) {
        viewModelScope.launch {
            val currentState = _userState.value
            if (currentState is UserState.Success) {
                val updatedUser = currentState.user.copy(name = newName)
                val success = userRepository.updateUser(updatedUser)
                if (success) {
                    _userState.value = UserState.Success(updatedUser)
                }
            }
        }
    }
}

// State
sealed class UserState {
    object Loading : UserState()
    data class Success(val user: User) : UserState()
    data class Error(val message: String) : UserState()
}

// View
@Composable
fun UserProfileScreen(
    userId: String,
    viewModel: UserProfileViewModel = viewModel(
        factory = UserProfileViewModelFactory(userId)
    )
) {
    val userState by viewModel.userState.collectAsState()
    
    when (userState) {
        is UserState.Loading -> {
            CircularProgressIndicator()
        }
        is UserState.Success -> {
            val user = (userState as UserState.Success).user
            UserProfileContent(
                user = user,
                onNameUpdate = { newName ->
                    viewModel.updateUserName(userId, newName)
                }
            )
        }
        is UserState.Error -> {
            val error = (userState as UserState.Error).message
            Text("Error: $error")
        }
    }
}
```

## 性能优化

### 避免不必要的重组

```kotlin
@Composable
fun OptimizedList(items: List<String>) {
    LazyColumn {
        items(
            items = items,
            key = { it }  // 使用key标识项目
        ) { item ->
            // 使用LaunchedEffect避免重复计算
            val processedItem by remember(item) {
                derivedStateOf { processItem(item) }
            }
            
            Text(text = processedItem)
        }
    }
}
```
