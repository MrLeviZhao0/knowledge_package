# AArch64内存模型

## 概述
AArch64架构采用加载-存储模型，所有数据处理操作都在寄存器中完成，内存访问通过专门的加载和存储指令实现。理解内存模型对于优化性能和进行逆向分析至关重要。

## 内存访问基础

### 地址空间
```assembly
; AArch64使用64位虚拟地址空间
; 实际可用的地址空间取决于实现

; 典型的地址空间布局
0x0000_0000_0000_0000 - 0x0000_FFFF_FFFF_FFFF : 用户空间
0xFFFF_0000_0000_0000 - 0xFFFF_FFFF_FFFF_FFFF : 内核空间

; 页大小通常为4KB、16KB或64KB
```

### 对齐要求
```assembly
; 不同数据类型的对齐要求
; 字节: 1字节对齐
; 半字: 2字节对齐  
; 字: 4字节对齐
; 双字: 8字节对齐
; 四字: 16字节对齐（SIMD寄存器）

; 未对齐访问可能导致性能下降或异常
LDR X0, [X1]        ; 要求X1 8字节对齐
LDR W0, [X1]        ; 要求X1 4字节对齐
LDRH W0, [X1]       ; 要求X1 2字节对齐
```

## 内存访问指令

### 基本加载存储
```assembly
; 加载指令
LDR X0, [X1]        ; 从X1地址加载64位到X0
LDR W0, [X1]        ; 加载32位到W0
LDRH W0, [X1]       ; 加载16位半字
LDRB W0, [X1]       ; 加载8位字节

; 存储指令
STR X0, [X1]        ; 存储X0到X1地址
STR W0, [X1]        ; 存储32位
STRH W0, [X1]       ; 存储16位
STRB W0, [X1]       ; 存储8位

; 有符号加载
LDRSW X0, [X1]      ; 加载有符号32位到64位寄存器
LDRSH W0, [X1]      ; 加载有符号16位到32位寄存器
LDRSB W0, [X1]      ; 加载有符号8位
```

### 地址模式
```assembly
; 立即数偏移
LDR X0, [X1, #8]    ; 地址 = X1 + 8
STR X0, [X1, #-16]  ; 地址 = X1 - 16

; 寄存器偏移
LDR X0, [X1, X2]    ; 地址 = X1 + X2
LDR X0, [X1, X2, LSL #3] ; 地址 = X1 + (X2 << 3)

; 前变址
LDR X0, [X1, #8]!   ; 先计算X1+8，然后加载，最后X1=X1+8

; 后变址
LDR X0, [X1], #8    ; 从X1加载，然后X1=X1+8

; 对变址
LDP X0, X1, [X2], #16 ; 加载X0=[X2], X1=[X2+8], 然后X2=X2+16
STP X0, X1, [X2, #-16]! ; 先X2=X2-16, 然后存储X0到[X2], X1到[X2+8]
```

## 内存屏障

### 数据同步屏障
```assembly
; 确保内存访问顺序
DSB SY              ; 全系统数据同步屏障
DSB ISH             ; 内共享域同步
DSB NSH             ; 非共享域同步
DSB OSH             ; 外共享域同步

; 使用场景：在修改页表后确保所有CPU看到更改
MSR TTBR0_EL1, X0   ; 设置页表基址
DSB SY              ; 等待所有内存访问完成
ISB                 ; 清空流水线
```

### 指令同步屏障
```assembly
; 确保后续指令从新地址获取
ISB                 ; 指令同步屏障

; 使用场景：修改代码后执行新代码
; 自修改代码或JIT编译后
STR W0, [X1]        ; 写入新指令
DMB SY              ; 数据内存屏障
ISB                 ; 清空流水线
BR X2               ; 跳转到新代码
```

## 缓存管理

### 缓存操作指令
```assembly
; 缓存行操作
DC CVAU, X0         ; 数据缓存清理到PoU
DC CVAC, X0         ; 数据缓存清理到PoC
DC CIVAC, X0        ; 数据缓存无效化并清理

; 使用场景：DMA操作前后
; DMA读取前：清理缓存确保内存数据最新
; DMA写入后：无效化缓存确保CPU读取新数据

; TLB操作
TLBI VAE1, X0       ; 按虚拟地址无效化TLB
TLBI VMALLE1        ; 无效化所有TLB条目
TLBI ASIDE1, X0     ; 按ASID无效化TLB
```

### 缓存一致性
```assembly
; 多核缓存一致性协议
; AArch64使用MOESI协议：
; M - Modified (修改)
; O - Owned (拥有)
; E - Exclusive (独占)
; S - Shared (共享)
; I - Invalid (无效)

; 确保多核间数据一致性
DMB ISH             ; 内共享域内存屏障
; 确保核间数据可见性
```

## 虚拟内存系统

### 页表结构
```assembly
; AArch64支持多级页表
; 通常使用4级页表（48位虚拟地址）

; 页表项格式
; Bit 0: Valid (有效位)
; Bit 1: Table/Block (表项/块项)
; Bits 2-47: 物理地址
; Bits 48-58: 保留
; Bits 59-61: 属性

; 页属性
; AP[2:1]: 访问权限 (00=无, 01=只读, 10=读写, 11=只读)
; AP[0]: 用户访问权限
; XN: 执行从不
; PXN: 特权执行从不
```

### 地址转换
```assembly
; 虚拟地址到物理地址转换过程
; VA[47:39] -> L0索引 -> L1页表
; VA[38:30] -> L1索引 -> L2页表  
; VA[29:21] -> L2索引 -> L3页表
; VA[20:12] -> L3索引 -> 4KB页
; VA[11:0] -> 页内偏移

; 使用AT指令进行地址转换
AT S1E1R, X0        ; 阶段1EL1读地址转换
AT S1E1W, X0        ; 阶段1EL1写地址转换
AT S1E0R, X0        ; 阶段1EL0读地址转换
```

## 内存类型

### 内存属性
```assembly
; 内存类型定义
; Normal Memory: 可缓存，可预测
; Device Memory: 不可缓存，有副作用

; 设备内存类型
; Device-nGnRnE: 不聚集，不重排序，不早期写确认
; Device-nGnRE: 不聚集，不重排序，早期写确认
; Device-GRE: 聚集，重排序，早期写确认

; 设置内存属性
; 通过页表或内存类型范围寄存器(MAIR_EL1)

; MAIR_EL1配置示例
MOV X0, #0xFF00000000000000    ; 设备内存属性
ORR X0, X0, #0x0000000000000044 ; 普通内存属性
MSR MAIR_EL1, X0               ; 设置内存属性
```

## 性能优化

### 内存访问模式优化
```assembly
; 顺序访问优化
; 坏模式：随机访问
LDR X0, [X1]
LDR X1, [X10] 
LDR X2, [X5]
LDR X3, [X15]

; 好模式：顺序访问
LDR X0, [X1]
LDR X1, [X1, #8]
LDR X2, [X1, #16]
LDR X3, [X1, #24]

; 使用LDP/STP减少指令数
LDP X0, X1, [X2]     ; 一次加载两个寄存器
LDP X2, X3, [X2, #16]
```

### 缓存友好代码
```assembly
; 缓存行大小通常为64字节
; 优化数据布局

; 结构体对齐优化
.struct 0
field1: .skip 8      ; 8字节对齐
field2: .skip 16     ; 16字节对齐（缓存行友好）
field3: .skip 8
.struct_size:

; 循环优化：步长为缓存行大小
MOV X0, base_addr
MOV X1, #64          ; 缓存行大小
MOV X2, #0
loop:
LDR X3, [X0, X2]    ; 每次加载一个缓存行
ADD X2, X2, X1
CMP X2, #4096
B.LT loop
```

## 逆向分析技巧

### 识别内存访问模式
```assembly
; 识别数组访问
LDR W0, [X1, W2, SXTW #2]  ; 数组索引：base + index*4

; 识别结构体访问
LDR W0, [X1, #16]          ; 结构体字段偏移16字节

; 识别链表遍历
LDR X1, [X1, #8]           ; 链表next指针在偏移8
CBZ X1, end
B loop
```

### 内存布局分析
```assembly
; 分析栈帧布局
STP X29, X30, [SP, #-32]!  ; 保存帧指针和返回地址
MOV X29, SP                 ; 设置新帧指针
STR X19, [SP, #16]         ; 保存被调用者保存寄存器

; 分析堆分配模式
BL malloc                   ; 调用内存分配
MOV X19, X0                 ; 保存返回的指针
```

## 实战示例

### 内存拷贝优化
```assembly
; 优化的内存拷贝函数
MEMCPY_OPTIMIZED:
    ; X0: 目标地址, X1: 源地址, X2: 大小
    
    ; 检查对齐
    AND X3, X0, #7
    AND X4, X1, #7
    CBNZ X3, unaligned_copy
    CBNZ X4, unaligned_copy
    
    ; 对齐拷贝：使用LDP/STP
    LSr X3, X2, #4          ; 计算四字数量
    CBZ X3, small_copy
    
aligned_loop:
    LDP X4, X5, [X1], #16   ; 加载两个四字
    STP X4, X5, [X0], #16   ; 存储两个四字
    SUBS X3, X3, #1
    B.NE aligned_loop
    
small_copy:
    ; 处理剩余字节
    AND X2, X2, #15
    CBZ X2, copy_done
    
byte_loop:
    LDRB W3, [X1], #1
    STRB W3, [X0], #1
    SUBS X2, X2, #1
    B.NE byte_loop
    
copy_done:
    RET
    
unaligned_copy:
    ; 非对齐处理
    B byte_loop
```

### 内存安全检查
```assembly
; 边界检查函数
CHECK_BOUNDS:
    ; X0: 指针, X1: 大小, X2: 边界
    
    ; 检查空指针
    CBZ X0, bounds_error
    
    ; 检查大小
    CMP X1, X2
    B.GT bounds_error
    
    ; 检查地址对齐（可选）
    AND X3, X0, #7
    CBNZ X3, bounds_warning
    
    ; 检查地址范围
    MOV X3, #0x0000FFFFFFFFFFFF
    CMP X0, X3
    B.HI bounds_error
    
    RET
    
bounds_error:
    MOV X0, #-1
    RET
    
bounds_warning:
    ; 非对齐警告，但继续执行
    RET
```

## 总结
AArch64内存模型提供了灵活而强大的内存管理能力，从基础的加载存储指令到复杂的虚拟内存系统。理解内存模型对于编写高性能代码和进行有效的逆向分析都至关重要。通过优化内存访问模式、合理使用缓存和内存屏障，可以显著提升应用性能。
