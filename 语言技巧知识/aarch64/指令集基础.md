# AArch64指令集基础

## 指令集概述

AArch64指令集采用精简指令集设计，具有固定的32位指令长度。指令集分为多个功能类别，支持丰富的操作能力。

## 数据处理指令

### 算术运算指令

#### 基本算术运算
```assembly
; 加法运算
ADD  X0, X1, X2           ; X0 = X1 + X2
ADD  X0, X1, #100         ; X0 = X1 + 100
ADD  X0, X1, X2, LSL #2   ; X0 = X1 + (X2 << 2)

; 减法运算
SUB  X0, X1, X2           ; X0 = X1 - X2
SUB  X0, X1, #50          ; X0 = X1 - 50

; 乘法运算
MUL  X0, X1, X2           ; X0 = X1 * X2
SMULH X0, X1, X2          ; 有符号高位乘法
UMULH X0, X1, X2          ; 无符号高位乘法

; 除法运算
SDIV X0, X1, X2           ; 有符号除法
UDIV X0, X1, X2           ; 无符号除法
```

#### 扩展算术运算
```assembly
; 带进位的加法
ADCS X0, X1, X2           ; X0 = X1 + X2 + C (设置标志)

; 带借位的减法
SBCS X0, X1, X2           ; X0 = X1 - X2 - (1 - C)

; 比较指令（减法但不存储结果）
CMP  X1, X2               ; 设置标志：X1 - X2
CMN  X1, X2               ; 设置标志：X1 + X2 (比较负值)

; 测试位
TST  X1, X2               ; 设置标志：X1 & X2
```

### 逻辑运算指令

#### 位运算
```assembly
; 逻辑与
AND  X0, X1, X2           ; X0 = X1 & X2
AND  X0, X1, #0xFF        ; X0 = X1 & 0xFF

; 逻辑或
ORR  X0, X1, X2           ; X0 = X1 | X2
ORR  X0, X1, #0x100       ; X0 = X1 | 0x100

; 逻辑异或
EOR  X0, X1, X2           ; X0 = X1 ^ X2
EON  X0, X1, X2           ; X0 = X1 ^ ~X2

; 位清除
BIC  X0, X1, X2           ; X0 = X1 & ~X2
```

#### 移位运算
```assembly
; 逻辑左移
LSL  X0, X1, #5           ; X0 = X1 << 5
LSL  X0, X1, X2           ; X0 = X1 << X2

; 逻辑右移
LSR  X0, X1, #3           ; X0 = X1 >>> 3 (无符号)
LSR  X0, X1, X2           ; X0 = X1 >>> X2

; 算术右移
ASR  X0, X1, #4           ; X0 = X1 >> 4 (有符号)
ASR  X0, X1, X2           ; X0 = X1 >> X2

; 循环右移
ROR  X0, X1, #8           ; X0 = (X1 >> 8) | (X1 << (64-8))
ROR  X0, X1, X2           ; X0 = (X1 >> X2) | (X1 << (64-X2))
```

### 移动指令

#### 寄存器移动
```assembly
; 基本移动
MOV  X0, X1               ; X0 = X1
MOV  X0, #0x1234          ; X0 = 0x1234

; 位域移动
MOVK X0, #0x5678, LSL #16 ; 将0x5678移动到X0的16-31位
MOVZ X0, #0xABCD, LSL #32 ; 零扩展移动
MOVN X0, #0x1234          ; 移动取反值

; 符号扩展移动
SXTB X0, W1               ; 字节符号扩展到64位
SXTH X0, W1               ; 半字符号扩展到64位
SXTW X0, W1               ; 字符号扩展到64位

; 零扩展移动
UXTB X0, W1               ; 字节零扩展到64位
UXTH X0, W1               ; 半字零扩展到64位
UXTW X0, W1               ; 字零扩展到64位
```

## 加载存储指令

### 基本加载存储

#### 单寄存器访问
```assembly
; 加载指令
LDR  X0, [X1]             ; X0 = [X1] (64位加载)
LDR  W0, [X1]             ; W0 = [X1] (32位加载)
LDRB W0, [X1]             ; 字节加载（零扩展）
LDRH W0, [X1]             ; 半字加载（零扩展）
LDRSB X0, [X1]            ; 有符号字节加载
LDRSH X0, [X1]            ; 有符号半字加载
LDRSW X0, [X1]            ; 有符号字加载

; 存储指令
STR  X0, [X1]             ; [X1] = X0 (64位存储)
STR  W0, [X1]             ; [X1] = W0 (32位存储)
STRB W0, [X1]             ; 字节存储
STRH W0, [X1]             ; 半字存储
```

#### 带偏移的访问
```assembly
; 前变址
LDR  X0, [X1, #8]         ; X0 = [X1 + 8]
STR  X0, [X1, #-16]!      ; 先X1=X1-16，然后[X1]=X0

; 后变址
LDR  X0, [X1], #8         ; X0 = [X1]，然后X1=X1+8
STR  X0, [X1], #-16       ; [X1] = X0，然后X1=X1-16

; 寄存器偏移
LDR  X0, [X1, X2]         ; X0 = [X1 + X2]
LDR  X0, [X1, X2, LSL #3] ; X0 = [X1 + (X2 << 3)]
```

### 多寄存器加载存储

#### 成对加载存储
```assembly
; 加载一对寄存器
LDP  X0, X1, [X2]         ; X0 = [X2], X1 = [X2 + 8]
LDP  X0, X1, [X2, #16]    ; X0 = [X2 + 16], X1 = [X2 + 24]
LDP  X0, X1, [X2], #16    ; 加载后X2 = X2 + 16
LDP  X0, X1, [X2, #16]!   ; 先X2 = X2 + 16，再加载

; 存储一对寄存器
STP  X0, X1, [X2]         ; [X2] = X0, [X2 + 8] = X1
STP  X0, X1, [X2, #-16]!  ; 先X2 = X2 - 16，再存储
```

#### 批量加载存储
```assembly
; 加载多个寄存器
LDM  X1, {X0, X2, X4}     ; 批量加载（实际使用LDP组合）

; 存储多个寄存器
STM  X1, {X0, X2, X4}     ; 批量存储

; 在实际代码中，通常使用多个LDP/STP指令
; 而不是真正的LDM/STM指令
```

## 控制流指令

### 无条件跳转

#### 直接跳转
```assembly
; 相对跳转
B    label                ; 跳转到label
B    .L1                  ; 跳转到本地标签

; 函数调用
BL   function_name        ; 跳转并链接（保存返回地址到X30）
BLR  X0                   ; 跳转到X0地址并链接

; 返回
RET  X30                  ; 返回到X30地址（默认）
RET                       ; 等同于RET X30
BR    X0                  ; 跳转到X0地址（不链接）
```

#### 间接跳转
```assembly
; 通过寄存器跳转
BR   X0                   ; 跳转到X0
BLR  X1                   ; 调用X1指向的函数

; 返回指令的变体
RET  X5                   ; 返回到X5地址
ERET                      ; 异常返回（从异常级别返回）
DRET                      ; 调试异常返回
```

### 条件跳转

#### 基于标志的条件跳转
```assembly
; 相等/不等
B.EQ label                ; 如果Z=1（相等）跳转
B.NE label                ; 如果Z=0（不等）跳转

; 无符号比较
B.HI label                ; 如果C=1且Z=0（高于）跳转
B.HS label                ; 如果C=1（高于或等于）跳转
B.LO label                ; 如果C=0（低于）跳转
B.LS label                ; 如果C=0或Z=1（低于或等于）跳转

; 有符号比较
B.GT label                ; 如果Z=0且N=V（大于）跳转
B.GE label                ; 如果N=V（大于或等于）跳转
B.LT label                ; 如果N≠V（小于）跳转
B.LE label                ; 如果Z=1或N≠V（小于或等于）跳转

; 其他条件
B.MI label                ; 如果N=1（负）跳转
B.PL label                ; 如果N=0（正）跳转
B.VS label                ; 如果V=1（溢出）跳转
B.VC label                ; 如果V=0（无溢出）跳转
```

#### 条件执行
```assembly
; 条件选择
CSEL X0, X1, X2, EQ      ; 如果相等，X0=X1，否则X0=X2
CSINC X0, X1, X2, NE     ; 如果不相等，X0=X1，否则X0=X2+1
CSINV X0, X1, X2, GT     ; 如果大于，X0=X1，否则X0=~X2
CSNEG X0, X1, X2, LE     ; 如果小于等于，X0=X1，否则X0=-X2

; 条件设置
CSET X0, EQ              ; 如果相等，X0=1，否则X0=0
CSETM X0, NE             ; 如果不相等，X0=-1，否则X0=0
CINC X0, X1, GT          ; 如果大于，X0=X1+1，否则X0=X1
CINV X0, X1, LE          ; 如果小于等于，X0=~X1，否则X0=X1
CNEG X0, X1, MI          ; 如果负，X0=-X1，否则X0=X1
```

### 循环和分支预测

#### 循环结构
```assembly
; 简单循环
MOV  X0, #0              ; 初始化计数器
MOV  X1, #100            ; 循环上限

loop_start:
    CMP  X0, X1          ; 比较计数器和上限
    B.GE loop_end        ; 如果X0 >= X1，跳出循环
    
    ; 循环体
    ADD  X2, X2, X0      ; 累加
    
    ADD  X0, X0, #1      ; 计数器加1
    B    loop_start      ; 继续循环
    
loop_end:
    ; 循环结束后的代码
```

#### 分支预测提示
```assembly
; 静态分支预测
B.PT label               ; 预测跳转发生
B.PF label               ; 预测跳转不发生

; 在实际代码中较少使用，编译器通常自动处理
```

## 系统指令

### 系统寄存器访问
```assembly
; 读写系统寄存器
MRS  X0, SCTLR_EL1       ; 读取系统控制寄存器
MSR  SCTLR_EL1, X0       ; 写入系统控制寄存器

; 特殊寄存器访问
MRS  X0, NZCV            ; 读取条件标志
MSR  NZCV, X0            ; 写入条件标志

MRS  X0, FPCR            ; 读取浮点控制寄存器
MSR  FPCR, X0            ; 写入浮点控制寄存器
```

### 系统调用
```assembly
; 系统调用指令
SVC  #0                  ; 超级visor调用（系统调用）
HVC  #0                  ; 超级visor调用（虚拟化）
SMC  #0                  ; 安全监控调用

; 系统调用示例
MOV  X8, #93             ; exit系统调用号
MOV  X0, #0              ; 退出码
SVC  #0                  ; 执行系统调用
```

### 内存屏障
```assembly
; 数据内存屏障
DMB  SY                  ; 数据内存屏障（全系统）
DMB  ISH                 ; 内共享域屏障
DMB  NSH                 ; 非共享域屏障

; 指令内存屏障
ISB  SY                  ; 指令同步屏障

; 数据同步屏障
DSB  SY                  ; 数据同步屏障
```

## SIMD和浮点指令

### 基本浮点运算
```assembly
; 浮点加载存储
LDR  S0, [X1]            ; 加载单精度浮点数
LDR  D0, [X1]            ; 加载双精度浮点数
STR  S0, [X1]            ; 存储单精度浮点数
STR  D0, [X1]            ; 存储双精度浮点数

; 浮点算术
FADD S0, S1, S2          ; 单精度加法
FSUB D0, D1, D2          ; 双精度减法
FMUL S0, S1, S2          ; 单精度乘法
FDIV D0, D1, D2          ; 双精度除法

; 浮点比较
FCMP S1, S2              ; 比较单精度浮点数
FCMP D1, D2              ; 比较双精度浮点数
FCMP S1, #0.0            ; 与零比较
```

### SIMD向量运算
```assembly
; 向量加载存储
LD1  {V0.4S}, [X1]       ; 加载4个单精度浮点数
LD1  {V0.2D, V1.2D}, [X1] ; 加载4个双精度浮点数
ST1  {V0.8H}, [X1]       ; 存储8个半字

; 向量算术
ADD  V0.4S, V1.4S, V2.4S ; 4个单精度浮点数相加
SUB  V0.8H, V1.8H, V2.8H ; 8个半字相减
MUL  V0.2D, V1.2D, V2.2D ; 2个双精度浮点数相乘

; 向量比较
CMEQ V0.4S, V1.4S, V2.4S ; 向量相等比较
CMGT V0.8H, V1.8H, V2.8H ; 向量大于比较
```

## 实际代码示例

### 函数实现示例
```assembly
; C函数: int64_t factorial(int64_t n)
FACTORIAL:
    ; 参数: X0 = n
    ; 返回值: X0 = n!
    
    CMP  X0, #1          ; 比较 n <= 1
    B.LE base_case       ; 如果是，跳转到基本情况
    
    ; 保存寄存器
    STP  X29, X30, [SP, #-16]!
    MOV  X29, SP
    
    ; 保存n
    MOV  X1, X0          ; X1 = n
    
    ; 递归调用: factorial(n-1)
    SUB  X0, X0, #1      ; X0 = n - 1
    BL   FACTORIAL       ; 递归调用
    
    ; 计算 n * factorial(n-1)
    MUL  X0, X1, X0      ; X0 = n * factorial(n-1)
    
    ; 恢复寄存器并返回
    LDP  X29, X30, [SP], #16
    RET
    
base_case:
    MOV  X0, #1          ; 0! = 1, 1! = 1
    RET
```

### 数组处理示例
```assembly
; C代码: void array_sum(int64_t *arr, int64_t n, int64_t *result)
ARRAY_SUM:
    ; 参数: X0 = arr, X1 = n, X2 = result
    
    MOV  X3, XZR         ; sum = 0
    MOV  X4, XZR         ; i = 0
    
loop:
    CMP  X4, X1          ; 比较 i < n
    B.GE done
    
    LDR  X5, [X0, X4, LSL #3]  ; X5 = arr[i] (64位，左移3位)
    ADD  X3, X3, X5      ; sum += arr[i]
    
    ADD  X4, X4, #1      ; i++
    B    loop
    
done:
    STR  X3, [X2]        ; *result = sum
    RET
```

### 字符串处理示例
```assembly
; C代码: int64_t strlen(const char *str)
STRLEN:
    ; 参数: X0 = str
    ; 返回值: X0 = 字符串长度
    
    MOV  X1, XZR         ; len = 0
    
loop:
    LDRB W2, [X0, X1]    ; 加载字符
    CBZ  W2, done        ; 如果字符为0，结束
    
    ADD  X1, X1, #1      ; len++
    B    loop
    
done:
    MOV  X0, X1          ; 返回长度
    RET
```

## 性能优化技巧

### 指令选择优化
```assembly
; 使用更高效的指令
; 不好的代码
MOV  X0, #0              ; 清零寄存器

; 好的代码
EOR  X0, X0, X0          ; 异或自身清零

; 或者使用零寄存器
MOV  X0, XZR             ; 使用零寄存器
```

### 减少数据依赖
```assembly
; 不好的代码（数据依赖）
LDR  X0, [X1]           ; 加载数据
ADD  X0, X0, #1         ; 依赖上一条指令
STR  X0, [X1]           ; 存储结果

; 好的代码（减少依赖）
LDR  X0, [X1]           ; 加载数据
LDR  X2, [X3]           ; 加载其他数据
ADD  X0, X0, #1         ; 此时X0已经就绪
STR  X0, [X1]           ; 存储结果
```

### 循环优化
```assembly
; 循环展开
; 原始循环
MOV  X2, #100
loop:
    LDR  X0, [X1], #8
    ADD  X3, X3, X0
    SUBS X2, X2, #1
    B.NE loop

; 展开4次
MOV  X2, #25
loop_unrolled:
    LDR  X0, [X1], #8
    ADD  X3, X3, X0
    LDR  X0, [X1], #8
    ADD  X3, X3, X0
    LDR  X0, [X1], #8
    ADD  X3, X3, X0
    LDR  X0, [X1], #8
    ADD  X3, X3, X0
    SUBS X2, X2, #1
    B.NE loop_unrolled
```

## 总结

AArch64指令集提供了丰富的操作能力：
- **数据处理指令**支持各种算术和逻辑运算
- **加载存储指令**提供灵活的内存访问模式
- **控制流指令**实现复杂的程序流程
- **系统指令**用于底层系统操作
- **SIMD指令**支持高性能向量运算

掌握这些指令是进行AArch64汇编编程和性能优化的基础。
