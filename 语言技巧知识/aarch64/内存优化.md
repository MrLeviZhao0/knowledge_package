# AArch64内存优化

## 概述
内存优化是提升AArch64应用性能的关键技术。通过合理的数据布局、缓存优化和内存访问模式改进，可以显著减少内存延迟，提高程序执行效率。

## 缓存优化

### 缓存层次结构
```assembly
; AArch64典型缓存层次
; L1缓存: 分离的指令缓存(I$)和数据缓存(D$)，通常32-64KB
; L2缓存: 统一的缓存，通常256KB-1MB
; L3缓存: 共享缓存，通常1-8MB

; 缓存行大小通常为64字节
; 了解缓存参数有助于优化
MRS X0, CTR_EL0              ; 缓存类型寄存器
UBFX X1, X0, #0, #4          ; IminLine: 指令缓存行大小对数
UBFX X2, X0, #16, #4         ; DminLine: 数据缓存行大小对数

; 计算缓存行大小
MOV X3, #4
LSL X1, X3, X1               ; 指令缓存行大小
LSL X2, X3, X2               ; 数据缓存行大小
```

### 缓存友好代码
```assembly
; 顺序访问优化
OPTIMIZED_LOOP:
    MOV X0, base_addr
    MOV X1, #cache_line_size  ; 64字节
    MOV X2, #0
loop:
    LDR X3, [X0, X2]          ; 顺序访问
    ; 处理数据
    ADD X2, X2, X1            ; 步长为缓存行大小
    CMP X2, total_size
    B.LT loop
    
; 避免缓存抖动
; 坏模式：频繁跳转访问不同内存区域
BAD_PATTERN:
    LDR X0, [MEM_AREA1]
    LDR X1, [MEM_AREA2]       ; 可能引起缓存冲突
    LDR X2, [MEM_AREA3]
    LDR X3, [MEM_AREA1]       ; 再次访问，但可能已被逐出
```

## 数据布局优化

### 结构体对齐
```assembly
; 优化结构体布局减少缓存行占用
; 原始布局（可能跨缓存行）
.struct 0
field1: .skip 4      ; 4字节
field2: .skip 8      ; 8字节 - 可能跨行
field3: .skip 4      ; 4字节
.struct_size:        ; 总大小16字节

; 优化布局（缓存行对齐）
.struct 0
field2: .skip 8      ; 8字节 - 对齐
field1: .skip 4      ; 4字节
field3: .skip 4      ; 4字节
.padding: .skip 8    ; 填充到32字节对齐
.struct_size:
```

### 数组访问优化
```assembly
; 多维数组访问模式优化
; C代码: int array[100][100];
; 访问array[i][j]

; 非优化访问（列优先，缓存不友好）
MOV X0, array_base
MOV X1, #100                 ; 列数
MUL X2, row, X1              ; row * 100
ADD X2, X2, column           ; + column
LSL X2, X2, #2               ; * 4 (int大小)
LDR W3, [X0, X2]             ; 加载array[row][column]

; 优化访问（行优先，缓存友好）
; 重新组织数据为行优先存储
MOV X0, array_base
MOV X1, #100                 ; 行数
MUL X2, column, X1           ; column * 100
ADD X2, X2, row              ; + row
LSL X2, X2, #2               ; * 4
LDR W3, [X0, X2]             ; 加载array[row][column]
```

## 预取优化

### 硬件预取
```assembly
; AArch64支持自动硬件预取
; 基于访问模式的预取策略

; 顺序访问模式触发预取
SEQUENTIAL_ACCESS:
    MOV X0, base_addr
    MOV X1, #0
    MOV X2, element_size
loop:
    LDR X3, [X0, X1]          ; 顺序加载
    ; 硬件自动预取后续数据
    ADD X1, X1, X2
    CMP X1, total_size
    B.LT loop
```

### 软件预取
```assembly
; 使用PRFM指令进行软件预取
OPTIMIZED_LOOP_WITH_PREFETCH:
    MOV X0, base_addr
    MOV X1, #0
    MOV X2, cache_line_size
    MOV X3, prefetch_ahead    ; 预取提前量
    
prefetch_loop:
    ; 预取未来数据
    ADD X4, X1, X3
    CMP X4, total_size
    B.GE no_prefetch
    PRFM PLDL1KEEP, [X0, X4]  ; 预取到L1缓存
    
no_prefetch:
    LDR X5, [X0, X1]          ; 加载当前数据
    ; 处理数据
    ADD X1, X1, X2
    CMP X1, total_size
    B.LT prefetch_loop

; 预取策略
; PLDL1KEEP: 预取到L1，保留在缓存
; PLDL2KEEP: 预取到L2，保留在缓存
; PLDL3KEEP: 预取到L3，保留在缓存
; PLDL1STRM: 预取到L1，流式访问
```

## 内存访问指令优化

### 批量加载存储
```assembly
; 使用LDP/STP减少指令数
; 非优化版本
LDR X0, [X1]
LDR X2, [X1, #8]
LDR X3, [X1, #16]
LDR X4, [X1, #24]

; 优化版本
LDP X0, X2, [X1]            ; 加载X0=[X1], X2=[X1+8]
LDP X3, X4, [X1, #16]       ; 加载X3=[X1+16], X4=[X1+24]

; 存储优化
STP X0, X1, [X2]            ; 存储两个寄存器
STP X2, X3, [X2, #16]
```

### 非临时存储
```assembly
; 使用非临时存储避免污染缓存
; 适用于只写一次的数据

MOV X0, data_ptr
MOV X1, #0x12345678

; 普通存储（会进入缓存）
STR X1, [X0]

; 非临时存储（绕过缓存）
; 需要特定系统支持
; STNP X1, X2, [X0]         ; 非临时存储配对
```

## 内存屏障优化

### 屏障使用策略
```assembly
; 最小化内存屏障使用
; 只在必要时使用屏障

; 数据依赖屏障（最轻量）
; 依赖后续指令于前导指令的结果时使用

; 写入屏障
DMB ST                     ; 存储屏障（等待存储完成）

; 全屏障
DMB SY                     ; 全系统屏障

; 避免过度使用屏障
OVERUSE_BARRIER:
    STR X0, [X1]
    DMB SY                  ; 可能不必要的全屏障
    LDR X2, [X3]           ; 不依赖X1的读取
    
OPTIMIZED_BARRIER:
    STR X0, [X1]
    DMB ST                  ; 仅存储屏障
    ; 无依赖的读取可以并行执行
    LDR X2, [X3]
```

### 矩阵乘法优化
```assembly
; 缓存块优化的矩阵乘法
MATRIX_MULTIPLY_BLOCKED:
    ; X0: A矩阵, X1: B矩阵, X2: C矩阵, X3: N
    MOV X4, #block_size      ; 缓存块大小
    MOV X5, #0               ; i循环

i_loop:
    MOV X6, #0               ; j循环
j_loop:
    MOV X7, #0               ; k循环
    
    ; 处理block_size x block_size块
    MOV X8, X5               ; i_block = i
    MOV X9, X6               ; j_block = j
    
    ; 内层循环优化为顺序访问
    MOV X10, X8
    MUL X10, X10, X3         ; i * N
    ADD X10, X10, X9         ; i*N + j
    LSL X10, X10, #2         ; *4 (int)
    
    ; 使用SIMD进行块计算
    ; ... SIMD计算代码
    
    ADD X7, X7, #block_size
    CMP X7, X3
    B.LT k_loop
    
    ADD X6, X6, #block_size
    CMP X6, X3
    B.LT j_loop
    
    ADD X5, X5, #block_size
    CMP X5, X3
    B.LT i_loop
    
    RET
```

### 内存池优化
```assembly
; 自定义内存池减少分配开销
MEMORY_POOL:
    .struct 0
pool_base: .skip 8           ; 内存池基址
pool_size: .skip 8           ; 总大小
free_ptr: .skip 8            ; 空闲指针
pool_end: .skip 8            ; 结束地址
.struct_size:

; 初始化内存池
init_pool:
    STR X0, [X1, #pool_base]
    STR X2, [X1, #pool_size]
    STR X0, [X1, #free_ptr]
    ADD X3, X0, X2
    STR X3, [X1, #pool_end]
    RET

; 快速分配（无系统调用）
pool_alloc:
    LDR X2, [X0, #free_ptr]
    LDR X3, [X0, #pool_end]
    ADD X4, X2, X1           ; 新指针
    CMP X4, X3
    B.GT alloc_failed
    
    STR X4, [X0, #free_ptr]  ; 更新空闲指针
    MOV X0, X2               ; 返回分配地址
    RET

alloc_failed:
    MOV X0, #0               ; 返回NULL
    RET
```

## 总结
AArch64内存优化涉及缓存利用、数据布局、预取策略和内存访问模式等多个方面。通过理解硬件特性和应用需求，可以制定有效的优化策略。在逆向分析中，识别这些优化模式有助于理解代码意图和性能特征。
