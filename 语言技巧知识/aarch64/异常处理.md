# AArch64异常处理

## 概述
AArch64架构提供了完善的异常处理机制，包括中断、陷阱、中止和系统调用等异常类型。理解异常处理对于系统编程、内核开发和逆向分析都至关重要。

## 异常级别

### 异常级别定义
```assembly
; AArch64有4个异常级别（EL0-EL3），权限从低到高
; EL0: 用户模式 - 应用程序运行级别
; EL1: 操作系统内核模式
; EL2: 虚拟机监控程序模式  
; EL3: 安全监控模式（TrustZone）

; 异常级别转换
; 应用程序(EL0) -> 系统调用 -> 内核(EL1)
; 内核(EL1) -> 中断 -> 监控程序(EL2)
; 监控程序(EL2) -> 安全调用 -> 安全监控(EL3)
```

### 当前异常级别检查
```assembly
; 读取当前异常级别
MRS X0, CurrentEL

; CurrentEL寄存器格式
; Bits [3:2]: 异常级别 (00=EL0, 01=EL1, 10=EL2, 11=EL3)
; 其他位保留为0

; 检查是否在EL1
MRS X0, CurrentEL
AND X0, X0, #0xC        ; 掩码保留级别位
CMP X0, #0x4            ; EL1 = 0x4
B.EQ at_el1
```

## 异常向量表

### 向量表结构
```assembly
; 每个异常级别有自己的向量表，包含16个条目
; 每个条目对应特定的异常类型和状态

; 向量表偏移计算
; VBAR_ELn + (exception_type * 0x80) + (offset_within_entry)

; 异常类型分类
; 同步异常（指令执行产生的异常）
; 异步异常（外部事件产生的异常）
;    - IRQ: 普通中断
;    - FIQ: 快速中断
;    - SError: 系统错误

; 向量表条目布局（每个0x80字节）
.align 11                ; 2KB对齐要求
vector_table_el1:
    .word 0              ; 当前EL，使用SP0
    .word 0              ; 当前EL，使用SPx  
    .word 0              ; 低EL，AArch64
    .word 0              ; 低EL，AArch32
    ; 重复4次对应4种情况
```

### 设置向量表
```assembly
; 设置EL1向量表
ADR X0, vector_table_el1
MSR VBAR_EL1, X0
ISB                      ; 指令同步屏障

; 设置EL2向量表（如果使用虚拟化）
ADR X0, vector_table_el2
MSR VBAR_EL2, X0
ISB

; 设置EL3向量表（TrustZone）
ADR X0, vector_table_el3
MSR VBAR_EL3, X0
ISB
```

## 同步异常处理

### 异常原因寄存器
```assembly
; ESR_ELn: 异常综合征寄存器，包含异常原因信息

; 处理同步异常
exception_handler_sync:
    MRS X0, ESR_EL1           ; 读取异常原因
    UBFX X1, X0, #26, #6      ; 提取EC字段（异常类）
    UBFX X2, X0, #0, #16      ; 提取ISS字段（指令特定综合征）
    
    ; 根据EC字段处理不同异常
    CMP X1, #0x15             ; SVC指令（系统调用）
    B.EQ handle_svc
    
    CMP X1, #0x24             ; 指令中止（较低EL）
    B.EQ handle_instruction_abort
    
    CMP X1, #0x25             ; 数据中止（较低EL）
    B.EQ handle_data_abort
    
    ; 其他异常处理...
    B unknown_exception
```

### 系统调用处理
```assembly
; SVC指令处理
handle_svc:
    ; 从ESR获取系统调用号
    MRS X0, ESR_EL1
    AND X0, X0, #0xFFFF        ; ISS字段包含立即数
    
    ; 保存用户上下文
    STP X0, X1, [SP, #-16]!
    STP X2, X3, [SP, #-16]!
    ; ... 保存所有需要保存的寄存器
    
    ; 根据系统调用号分发
    CMP X0, #0
    B.EQ syscall_read
    CMP X0, #1  
    B.EQ syscall_write
    ; ... 其他系统调用
    
    ; 恢复上下文并返回
    LDP X2, X3, [SP], #16
    LDP X0, X1, [SP], #16
    ERET                       ; 异常返回
```

## 异步异常处理

### 中断处理
```assembly
; IRQ中断处理
irq_handler:
    ; 保存被中断的上下文
    STP X0, X1, [SP, #-16]!
    STP X2, X3, [SP, #-16]!
    STP X29, X30, [SP, #-16]!
    
    ; 读取中断控制器状态
    ; 这里假设使用GICv3
    MRS X0, ICC_IAR1_EL1      ; 读取中断确认寄存器
    
    ; 根据中断ID处理
    AND X1, X0, #0xFFFFFF      ; 提取中断ID
    CMP X1, #TIMER_IRQ
    B.EQ handle_timer_irq
    CMP X1, #UART_IRQ
    B.EQ handle_uart_irq
    
    ; 默认处理
    B default_irq_handler
    
irq_handler_done:
    ; 结束中断处理
    MSR ICC_EOIR1_EL1, X0      ; 写结束中断寄存器
    
    ; 恢复上下文
    LDP X29, X30, [SP], #16
    LDP X2, X3, [SP], #16
    LDP X0, X1, [SP], #16
    ERET
```

### 定时器中断
```assembly
; 定时器中断处理
handle_timer_irq:
    ; 读取定时器值
    MRS X0, CNTPCT_EL0         ; 物理计数器
    MRS X1, CNTFRQ_EL0         ; 计数器频率
    
    ; 设置下一次中断
    ADD X0, X0, X1             ; 1秒后
    MSR CNTP_TVAL_EL0, X1       ; 设置定时器值
    
    ; 处理定时器任务
    BL timer_tick_handler
    
    B irq_handler_done
```

## 异常返回

### 异常返回指令
```assembly
; ERET: 从异常返回，恢复PC和PSTATE
; 使用ELR_ELn保存返回地址，SPSR_ELn保存处理器状态

; 异常进入时自动保存
; ELR_ELn = 返回地址
; SPSR_ELn = PSTATE（包括NZCV标志）

; 异常返回示例
return_from_exception:
    ; 可选：修改返回地址（如系统调用处理）
    ; MSR ELR_EL1, X30
    
    ; 恢复处理器状态
    ; MSR SPSR_EL1, X29
    
    ; 返回
    ERET
```

### 状态保存和恢复
```assembly
; 完整的异常上下文保存/恢复
save_context:
    ; 保存通用寄存器
    STP X0, X1, [SP, #-16]!
    STP X2, X3, [SP, #-16]!
    ; ... 保存X0-X30
    
    ; 保存特殊寄存器
    MRS X0, SP_EL0
    STR X0, [SP, #-8]!
    MRS X0, ELR_EL1
    STR X0, [SP, #-8]!
    MRS X0, SPSR_EL1
    STR X0, [SP, #-8]!
    
    RET

restore_context:
    ; 恢复特殊寄存器
    LDR X0, [SP], #8
    MSR SPSR_EL1, X0
    LDR X0, [SP], #8
    MSR ELR_EL1, X0
    LDR X0, [SP], #8
    MSR SP_EL0, X0
    
    ; 恢复通用寄存器
    LDP X30, X29, [SP], #16
    ; ... 恢复X30-X0
    
    RET
```

## 系统调用机制

### 用户态系统调用
```assembly
; 用户态发起系统调用
user_syscall:
    ; 准备系统调用参数
    MOV X8, #SYS_CALL_NUMBER   ; 系统调用号
    MOV X0, ARG1               ; 第一个参数
    MOV X1, ARG2               ; 第二个参数
    
    ; 执行SVC指令
    SVC #0
    
    ; 返回值在X0中
    RET
```

### 内核态系统调用分发
```assembly
; 内核系统调用表
syscall_table:
    .quad sys_read              ; 系统调用0
    .quad sys_write             ; 系统调用1
    .quad sys_open              ; 系统调用2
    ; ...

; 系统调用分发器
syscall_dispatcher:
    ; X8包含系统调用号
    CMP X8, #NR_SYSCALLS        ; 检查系统调用号有效性
    B.HS invalid_syscall
    
    ; 从系统调用表跳转
    ADR X9, syscall_table
    LDR X10, [X9, X8, LSL #3]   ; 每个条目8字节
    BR X10

invalid_syscall:
    MOV X0, #-ENOSYS           ; 返回错误
    RET
```

## 安全扩展

### TrustZone异常处理
```assembly
; 安全监控调用（SMC）
handle_smc:
    ; 从ESR获取SMC功能号
    MRS X0, ESR_EL3
    AND X0, X0, #0xFFFF
    
    ; 根据功能号处理
    CMP X0, #SMC_GET_VERSION
    B.EQ smc_get_version
    CMP X0, #SMC_SERVICE_CALL
    B.EQ smc_service_call
    
    ; 无效SMC调用
    MOV X0, #-1
    ERET

; 安全世界切换
switch_to_secure_world:
    ; 配置安全世界入口
    ADR X0, secure_world_entry
    MSR SCR_EL3, X0
    
    ; 执行SMC进入安全世界
    SMC #0
    
    RET
```

## 逆向分析技巧

### 识别异常处理代码
```assembly
; 异常向量表特征
vector_table:
    B reset_handler          ; 复位异常
    B undefined_handler      ; 未定义指令
    B svc_handler            ; SVC调用
    B prefetch_abort_handler ; 指令预取中止
    B data_abort_handler     ; 数据中止
    B irq_handler            ; IRQ中断
    B fiq_handler            ; FIQ中断

; 异常处理函数特征
irq_handler:
    STP X29, X30, [SP, #-16]!  ; 保存寄存器
    MRS X0, ESR_EL1            ; 读取异常原因
    ; ... 异常处理逻辑
    LDP X29, X30, [SP], #16    ; 恢复寄存器
    ERET                       ; 异常返回
```

## 实战示例

### 最小异常处理框架
```assembly
; 简单的异常处理框架
.section .vectors, "ax"
.align 11
vectors:
    // 当前EL，使用SP0
    B . // 同步异常
    B . // IRQ
    B . // FIQ
    B . // SError
    
    // 当前EL，使用SPx
    B . // 同步异常
    B irq_handler // IRQ
    B . // FIQ
    B . // SError
    
    // 低EL，AArch64
    B . // 同步异常
    B . // IRQ
    B . // FIQ
    B . // SError
    
    // 低EL，AArch32
    B . // 同步异常
    B . // IRQ
    B . // FIQ
    B . // SError

// IRQ处理函数
irq_handler:
    // 简单处理：确认中断并返回
    MRS X0, ICC_IAR1_EL1
    MSR ICC_EOIR1_EL1, X0
    ERET
```

### 系统调用包装器
```assembly
; 系统调用包装函数
.macro syscall num, fn
.global \fn
\fn:
    MOV X8, #\num
    SVC #0
    RET
.endm

// 定义系统调用
syscall 0, sys_read
syscall 1, sys_write
syscall 2, sys_open
```

## 总结
AArch64异常处理机制提供了从应用程序到安全监控程序的完整保护层级。通过理解异常级别、向量表、系统调用和中断处理，可以开发健壮的系统软件和进行深入的逆向分析。异常处理是AArch64系统编程的核心内容，对于操作系统开发和嵌入式系统都至关重要。
