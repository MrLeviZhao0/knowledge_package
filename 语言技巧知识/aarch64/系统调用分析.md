# AArch64系统调用分析

## 系统调用概述

系统调用是用户空间程序与操作系统内核交互的接口。在AArch64架构中，系统调用通过`SVC`（Supervisor Call）指令实现，遵循特定的参数传递和返回值约定。

## 系统调用机制

### 基本系统调用流程

#### 系统调用指令
```assembly
; 基本系统调用模式
MOV X8, #syscall_number        ; 系统调用号
MOV X0, arg1                   ; 第一个参数
MOV X1, arg2                   ; 第二个参数
MOV X2, arg3                   ; 第三个参数
MOV X3, arg4                   ; 第四个参数
MOV X4, arg5                   ; 第五个参数
MOV X5, arg6                   ; 第六个参数
SVC #0                         ; 执行系统调用
; 返回值在X0中

; 实际示例：exit系统调用
MOV X0, #0                     ; 退出码
MOV X8, #93                    ; exit系统调用号
SVC #0                         ; 执行系统调用
```

#### 系统调用包装函数
```assembly
; 系统调用包装函数模板
SYSCALL_WRAPPER:
    MOV X8, #syscall_number    ; 系统调用号
    SVC #0
    RET

; 实际使用示例
; write系统调用包装
WRITE_SYSCALL:
    ; 参数：X0=fd, X1=buf, X2=count
    MOV X8, #64                ; write系统调用号
    SVC #0
    RET

; read系统调用包装
READ_SYSCALL:
    ; 参数：X0=fd, X1=buf, X2=count
    MOV X8, #63                ; read系统调用号
    SVC #0
    RET
```

## 系统调用参数分析

### 参数传递规则

#### 基本参数传递
```assembly
; 系统调用参数传递规则
; X0-X5: 前6个参数
; 超过6个参数时，使用栈传递

; 示例：open系统调用（3个参数）
OPEN_SYSCALL:
    ; X0: 文件名指针
    ; X1: 打开标志
    ; X2: 文件模式
    MOV X8, #56                ; open系统调用号
    SVC #0
    RET

; 示例：mmap系统调用（6个参数）
MMAP_SYSCALL:
    ; X0: 地址
    ; X1: 长度
    ; X2: 保护标志
    ; X3: 映射标志
    ; X4: 文件描述符
    ; X5: 偏移量
    MOV X8, #222               ; mmap系统调用号
    SVC #0
    RET
```

#### 复杂参数处理
```assembly
; 处理结构体参数
; 示例：stat系统调用（结构体参数）
STAT_SYSCALL:
    ; X0: 文件路径
    ; X1: stat结构体指针
    MOV X8, #106               ; stat系统调用号
    SVC #0
    RET

; stat结构体在内存中的布局需要正确设置

```

### 返回值处理

#### 系统调用返回值
```assembly
; 系统调用返回值约定
; 成功：返回非负值或具体结果
; 失败：返回负的错误码（-errno）

; 检查系统调用返回值
CHECK_SYSCALL_RESULT:
    MOV X8, #syscall_number
    SVC #0
    
    ; 检查是否成功
    CMP X0, #0
    B.GE success
    
    ; 处理错误
    NEG X1, X0                 ; 获取错误码（正数）
    ; 错误处理逻辑...
    
success:
    ; 成功处理逻辑
    RET
```

## 实际系统调用分析

### 文件操作分析

#### 文件读写操作
```assembly
; 完整的文件读写示例
FILE_OPERATIONS:
    STP X29, X30, [SP, #-32]!
    MOV X29, SP
    
    ; 打开文件
    ADR X0, filename           ; 文件名
    MOV X1, #0101              ; O_CREAT | O_WRONLY
    MOV X2, #0644              ; 文件模式
    MOV X8, #56                ; open系统调用
    SVC #0
    
    CMP X0, #0
    B.LT open_error
    
    STR X0, [X29, #-8]         ; 保存文件描述符
    
    ; 写入文件
    ADR X1, data               ; 数据缓冲区
    MOV X2, #data_len          ; 数据长度
    MOV X8, #64                ; write系统调用
    SVC #0
    
    CMP X0, #0
    B.LT write_error
    
    ; 关闭文件
    LDR X0, [X29, #-8]         ; 恢复文件描述符
    MOV X8, #57                ; close系统调用
    SVC #0
    
    B success

open_error:
    ; 打开文件错误处理
    B exit

write_error:
    ; 写入文件错误处理
    B exit

success:
    MOV X0, #0                 ; 成功返回
    
exit:
    LDP X29, X30, [SP], #32
    RET

.data
filename:
    .asciz "test.txt"
data:
    .asciz "Hello, World!"
data_len = . - data
```

#### 文件描述符操作
```assembly
; 标准输入输出操作
STDIO_OPERATIONS:
    ; 写入标准输出
    ADR X1, message
    MOV X2, #message_len
    MOV X0, #1                 ; stdout文件描述符
    MOV X8, #64                ; write系统调用
    SVC #0
    
    ; 从标准输入读取
    ADR X1, buffer
    MOV X2, #buffer_size
    MOV X0, #0                 ; stdin文件描述符
    MOV X8, #63                ; read系统调用
    SVC #0
    
    ; X0包含读取的字节数
    RET

.data
message:
    .asciz "Enter something: "
message_len = . - message
buffer:
    .space 256
buffer_size = 256
```

### 进程管理分析

#### 进程创建和控制
```assembly
; 进程创建示例
PROCESS_MANAGEMENT:
    ; fork创建子进程
    MOV X8, #107               ; fork系统调用
    SVC #0
    
    CMP X0, #0
    B.GT parent_process
    B.EQ child_process
    B fork_error

parent_process:
    ; 父进程代码
    ; X0包含子进程PID
    
    ; 等待子进程
    MOV X1, #0                 ; 状态指针
    MOV X2, #0                 ; 选项
    MOV X3, #0                 ; 资源使用
    MOV X8, #260               ; wait4系统调用
    SVC #0
    
    B exit

child_process:
    ; 子进程代码
    ; 执行新程序
    ADR X0, program_path       ; 程序路径
    ADR X1, argv               ; 参数数组
    MOV X2, #0                 ; 环境变量
    MOV X8, #221               ; execve系统调用
    SVC #0
    
    ; 如果execve失败，子进程继续执行
    MOV X0, #1                 ; 错误退出码
    MOV X8, #93                ; exit系统调用
    SVC #0

fork_error:
    ; fork错误处理
    MOV X0, #-1
    RET

.data
program_path:
    .asciz "/bin/ls"
argv:
    .quad program_path
    .quad 0
```

#### 进程信息获取
```assembly
; 获取进程信息

    RET
```

### 内存管理分析

#### 内存分配和映射
```assembly
; 内存管理操作
MEMORY_MANAGEMENT:
    ; brk系统调用 - 调整数据段
    MOV X0, #0                 ; 获取当前break
    MOV X8, #214               ; brk系统调用
    SVC #0
    ; X0包含当前break地址
    
    ; 分配内存（增加break）
    ADD X0, X0, #4096          ; 增加4KB
    MOV X8, #214               ; brk系统调用
    SVC #0
    
    ; mmap系统调用 - 内存映射
    MOV X0, #0                 ; 地址（由内核选择）
    MOV X1, #4096              ; 长度
    MOV X2, #3                 ; PROT_READ | PROT_WRITE
    MOV X3, #0x22              ; MAP_PRIVATE | MAP_ANONYMOUS
    MOV X4, #-1                ; 文件描述符（-1表示匿名映射）
    MOV X5, #0                 ; 偏移量
    MOV X8, #222               ; mmap系统调用
    SVC #0
    
    ; X0包含映射区域的地址
    
    RET
```

#### 内存保护
```assembly
; 内存保护操作
MEMORY_PROTECTION:
    ; 假设X0包含内存区域地址
    ; X1包含区域长度
    
    ; 修改内存保护
    MOV X2, #5                 ; PROT_READ | PROT_EXEC
    MOV X8, #226               ; mprotect系统调用
    SVC #0
    
    CMP X0, #0
    B.NE protection_error
    
    ; 保护修改成功
    RET

protection_error:
    ; 保护修改失败
    RET
```

## 系统调用跟踪和分析

### 使用strace跟踪

#### strace基本使用
```bash
# 跟踪程序执行的所有系统调用
strace ./aarch64_program

# 跟踪特定系统调用
strace -e trace=open,read,write ./program

# 输出到文件
strace -o trace.log ./program

# 跟踪子进程
strace -f ./program

# 统计系统调用
strace -c ./program
```

#### strace输出分析
```bash
# 典型strace输出示例
open("test.txt", O_RDONLY) = 3
read(3, "Hello, World!", 13) = 13
write(1, "Hello, World!", 13) = 13
close(3) = 0

# 分析：
# 1. 打开文件test.txt，返回文件描述符3
# 2. 从文件描述符3读取13字节
# 3. 向标准输出（文件描述符1）写入13字节
# 4. 关闭文件描述符3
```

```assembly
    LDP X0, X1, [SP], #16
    
    ; 返回系统调用结果
    MOV X0, X9
    RET
```

## 总结

AArch64系统调用分析的关键要点：

1. **调用机制**：通过`SVC`指令和`X8`寄存器传递系统调用号
2. **参数传递**：X0-X5用于前6个参数，栈用于额外参数
3. **返回值**：成功返回非负值，失败返回负的错误码
4. **常见模式**：文件操作、进程管理、内存管理、网络通信
5. **分析工具**：strace、GDB、自定义监控
6. **安全应用**：恶意行为检测、漏洞分析

掌握系统调用分析对于理解程序行为、进行安全评估和性能优化都至关重要。
