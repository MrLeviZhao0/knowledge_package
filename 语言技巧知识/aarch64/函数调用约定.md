# AArch64函数调用约定（AAPCS64）

## AAPCS64标准概述

AAPCS64（ARM Architecture Procedure Call Standard for 64-bit）是AArch64架构的函数调用标准，定义了参数传递、寄存器使用、栈帧管理等规则。遵循这些约定确保不同编译单元之间的正确交互。

## 基本调用约定

### 参数传递规则

#### 整数参数传递
```assembly
; 参数传递：X0-X7用于前8个整数参数
; 函数调用示例
MOV X0, #1                    ; 第一个参数
MOV X1, #2                    ; 第二个参数
MOV X2, #3                    ; 第三个参数
MOV X3, #4                    ; 第四个参数
MOV X4, #5                    ; 第五个参数
MOV X5, #6                    ; 第六个参数
MOV X6, #7                    ; 第七个参数
MOV X7, #8                    ; 第八个参数
BL my_function                ; 调用函数
; 返回值在X0中

; 函数实现
MY_FUNCTION:
    ; 参数使用：
    ; X0 = 参数1, X1 = 参数2, ..., X7 = 参数8
    ADD X0, X0, X1            ; X0 = 参数1 + 参数2
    ADD X0, X0, X2            ; X0 += 参数3
    ; ... 继续使用其他参数
    RET
```

#### 浮点参数传递
```assembly
; 浮点参数：V0-V7用于前8个浮点参数
FMOV D0, 1.0                  ; 第一个浮点参数
FMOV D1, 2.0                  ; 第二个浮点参数
FMOV D2, 3.0                  ; 第三个浮点参数
FMOV D3, 4.0                  ; 第四个浮点参数
BL float_function             ; 调用浮点函数

; 混合参数传递
MOV X0, #100                  ; 第一个整数参数
FMOV D0, 1.5                  ; 第一个浮点参数
MOV X1, #200                  ; 第二个整数参数
FMOV D1, 2.5                  ; 第二个浮点参数
BL mixed_function             ; 调用混合函数
```

### 返回值规则

#### 基本返回值
```assembly
; 整数返回值：X0
; 浮点返回值：V0
; 64位返回值示例
MY_FUNCTION:
    MOV X0, #42               ; 返回值42
    RET

; 32位返回值示例
MY_FUNCTION32:
    MOV W0, #42               ; 32位返回值
    RET

; 浮点返回值示例
FLOAT_FUNCTION:
    FMOV D0, 3.14             ; 双精度返回值
    RET

; 单精度浮点返回值
FLOAT_FUNCTION32:
    FMOV S0, 3.14             ; 单精度返回值
    RET
```

#### 大结构体返回值
```assembly
; 大结构体（>16字节）通过X8间接返回
; C函数：struct LargeStruct get_large_struct();

; 调用方
MOV X8, SP                    ; X8指向栈上的返回位置
SUB SP, SP, #64               ; 为结构体分配空间
BL get_large_struct           ; 调用函数
ADD SP, SP, #64               ; 恢复栈指针

; 被调用方
GET_LARGE_STRUCT:
    ; X8包含返回结构的地址
    ; 将数据写入[X8]开始的地址
    STR X0, [X8]              ; 写入第一个字段
    STR X1, [X8, #8]          ; 写入第二个字段
    ; ... 继续写入其他字段
    RET
```

## 寄存器使用规则

### 调用者保存寄存器
```assembly
; X9-X15, X16-X17：调用者保存（临时寄存器）
; 调用者需要在使用前保存这些寄存器

CALLER_FUNCTION:
    ; 保存临时寄存器
    STP X9, X10, [SP, #-16]!
    STP X11, X12, [SP, #-16]!
    
    ; 使用临时寄存器
    MOV X9, #100
    MOV X10, #200
    ADD X0, X9, X10
    
    ; 调用其他函数
    BL CALLEE_FUNCTION
    
    ; 恢复临时寄存器
    LDP X11, X12, [SP], #16
    LDP X9, X10, [SP], #16
    
    RET
```

### 被调用者保存寄存器
```assembly
; X19-X28：被调用者保存（非临时寄存器）
; 被调用者需要在使用前保存并在返回前恢复

CALLEE_FUNCTION:
    ; 序言：保存被调用者保存寄存器
    STP X29, X30, [SP, #-48]!  ; 保存帧指针和返回地址
    STP X19, X20, [SP, #16]    ; 保存X19,X20
    STP X21, X22, [SP, #32]    ; 保存X21,X22
    MOV X29, SP                 ; 设置帧指针
    
    ; 函数体：使用保存的寄存器
    MOV X19, X0                ; 使用X19
    MOV X20, #100
    ADD X0, X19, X20
    
    ; 尾声：恢复寄存器
    LDP X21, X22, [SP, #32]    ; 恢复X21,X22
    LDP X19, X20, [SP, #16]    ; 恢复X19,X20
    LDP X29, X30, [SP], #48    ; 恢复帧指针和返回地址
    RET
```

### 特殊寄存器
```assembly
; X8：间接结果位置寄存器
; X16-X17：内部过程调用临时寄存器
; X18：平台寄存器（TLS等）
; X29：帧指针（FP）
; X30：链接寄存器（LR）
; XZR：零寄存器

; 特殊寄存器使用示例
SPECIAL_REGISTERS:
    MOV X16, #0x1000           ; 使用IP0临时寄存器
    MOV X17, #0x2000           ; 使用IP1临时寄存器
    
    ; 使用零寄存器
    CMP X0, XZR                ; 比较X0是否等于0
    ADD X1, X2, XZR            ; X1 = X2 + 0
    
    RET
```

## 栈帧管理

### 标准栈帧布局
```assembly
; 标准栈帧结构
STANDARD_FRAME:
    ; 序言
    STP X29, X30, [SP, #-16]!  ; 保存帧指针和返回地址
    MOV X29, SP                 ; 设置新帧指针
    
    ; 可选：分配局部变量空间
    SUB SP, SP, #32             ; 分配32字节栈空间
    
    ; 函数体
    ; 局部变量在[SP]到[SP+31]范围内
    STR X0, [SP]                ; 保存参数到局部变量
    LDR X1, [SP]                ; 从局部变量加载
    
    ; 尾声
    MOV SP, X29                 ; 恢复栈指针
    LDP X29, X30, [SP], #16     ; 恢复帧指针和返回地址
    RET
```

### 复杂栈帧示例
```assembly
; 包含多个寄存器和局部变量的栈帧
COMPLEX_FRAME:
    ; 序言：保存多个寄存器
    STP X29, X30, [SP, #-64]!   ; 分配64字节栈空间
    STP X19, X20, [SP, #16]     ; 保存X19,X20
    STP X21, X22, [SP, #32]     ; 保存X21,X22
    STP X23, X24, [SP, #48]     ; 保存X23,X24
    MOV X29, SP
    
    ; 函数体
    ; 使用保存的寄存器
    MOV X19, X0                 ; 参数备份到X19
    MOV X20, X1                 ; 参数备份到X20
    
    ; 使用局部变量空间
    STR X19, [SP, #8]           ; 保存到局部变量
    STR X20, [SP, #16]          ; 保存到局部变量
    
    ; 复杂的计算...
    
    ; 尾声：恢复所有寄存器
    LDP X23, X24, [SP, #48]     ; 恢复X23,X24
    LDP X21, X22, [SP, #32]     ; 恢复X21,X22
    LDP X19, X20, [SP, #16]     ; 恢复X19,X20
    LDP X29, X30, [SP], #64     ; 恢复帧指针和返回地址
    RET
```

### 栈对齐要求
```assembly
; SP必须16字节对齐
ALIGNED_FRAME:
    ; 正确的栈对齐
    STP X29, X30, [SP, #-16]!   ; 16字节对齐
    MOV X29, SP
    
    ; 分配空间时保持对齐
    SUB SP, SP, #32             ; 32字节，保持16字节对齐
    
    ; 错误的栈对齐（会导致崩溃）
    ; SUB SP, SP, #24            ; 24字节，不满足16字节对齐
    
    RET
```

## 高级调用约定

### 可变参数函数
```assembly
; 可变参数函数处理
VARIADIC_FUNCTION:
    ; 固定参数在X0-X7
    ; 可变参数在栈上传递
    
    ; 保存可变参数寄存器到栈上
    STP X0, X1, [SP, #-16]!     ; 保存X0,X1
    STP X2, X3, [SP, #-16]!     ; 保存X2,X3
    STP X4, X5, [SP, #-16]!     ; 保存X4,X5
    STP X6, X7, [SP, #-16]!     ; 保存X6,X7
    
    ; 现在所有参数都在栈上连续存储
    ; 可以通过栈指针访问所有参数
    
    ; 处理参数...
    
    ; 恢复栈
    LDP X6, X7, [SP], #16
    LDP X4, X5, [SP], #16
    LDP X2, X3, [SP], #16
    LDP X0, X1, [SP], #16
    
    RET
```

### 函数指针调用
```assembly
; 通过函数指针调用
FUNCTION_POINTER_CALL:
    ; X0包含函数指针
    ; 设置参数
    MOV X1, #100                ; 第一个参数
    MOV X2, #200                ; 第二个参数
    
    ; 通过函数指针调用
    BLR X0                      ; 分支并链接到X0
    
    ; 返回值在X0中
    RET

; 间接尾调用优化
TAIL_CALL:
    ; 设置参数
    MOV X0, #42
    MOV X1, #84
    
    ; 尾调用：直接跳转，不保存返回地址
    B ANOTHER_FUNCTION          ; 尾调用优化
```

## 实际应用示例

### C函数调用示例
```assembly
; C函数：int64_t add_three(int64_t a, int64_t b, int64_t c);

; 调用代码
MOV X0, #10                    ; a = 10
MOV X1, #20                    ; b = 20
MOV X2, #30                    ; c = 30
BL add_three                   ; 调用函数
; X0包含结果 (10+20+30=60)

; 函数实现
ADD_THREE:
    ADD X0, X0, X1             ; X0 = a + b
    ADD X0, X0, X2             ; X0 += c
    RET
```

### 递归函数示例
```assembly
; 递归计算阶乘
; C函数：int64_t factorial(int64_t n);

FACTORIAL:
    ; 参数：X0 = n
    
    ; 基本情况：n <= 1
    CMP X0, #1
    B.LE BASE_CASE
    
    ; 递归情况：保存寄存器
    STP X29, X30, [SP, #-16]!
    MOV X29, SP
    
    ; 保存n
    MOV X19, X0                 ; X19 = n
    
    ; 递归调用：factorial(n-1)
    SUB X0, X0, #1
    BL FACTORIAL
    
    ; 计算 n * factorial(n-1)
    MUL X0, X19, X0
    
    ; 恢复寄存器
    LDP X29, X30, [SP], #16
    RET
    
BASE_CASE:
    MOV X0, #1                  ; 0! = 1, 1! = 1
    RET
```

### 系统调用示例
```assembly
; 系统调用包装函数
; C函数：int64_t sys_write(int fd, const char* buf, size_t count);

SYS_WRITE:
    ; 参数：X0=fd, X1=buf, X2=count
    
    ; 系统调用号：write = 64
    MOV X8, #64                 ; 系统调用号
    SVC #0                      ; 执行系统调用
    
    ; 返回值在X0中
    RET

; 使用示例
MOV X0, #1                      ; stdout文件描述符
ADR X1, message                 ; 消息地址
MOV X2, #14                     ; 消息长度
BL SYS_WRITE                    ; 调用write系统调用

.data
message:
    .asciz "Hello, World!\n"
```

## 调试和优化

### 栈帧调试技巧
```assembly
; 调试信息：查看栈帧
DEBUG_FRAME:
    STP X29, X30, [SP, #-16]!
    MOV X29, SP
    
    ; 调试：打印帧指针和返回地址
    ; 当前帧指针：X29
    ; 返回地址：[X29, #8]
    ; 上一帧指针：[X29]
    
    ; 栈回溯示例
    MOV X0, X29                 ; 当前帧指针
    
backtrace_loop:
    CBZ X0, backtrace_done      ; 如果帧指针为0，结束
    
    LDR X1, [X0, #8]            ; 返回地址
    LDR X0, [X0]                ; 上一帧指针
    
    ; 处理返回地址X1...
    
    B backtrace_loop
    
backtrace_done:
    LDP X29, X30, [SP], #16
    RET
```

### 性能优化考虑
```assembly
; 减少栈操作
OPTIMIZED_FUNCTION:
    ; 避免不必要的栈操作
    ; 使用临时寄存器而不是栈
    
    ; 不好的代码：频繁栈操作
    ; STR X0, [SP, #-8]!
    ; ... 操作
    ; LDR X0, [SP], #8
    
    ; 好的代码：使用寄存器
    MOV X9, X0                  ; 使用临时寄存器X9
    ; ... 操作使用X9
    MOV X0, X9                  ; 结果放回X0
    
    RET
```

## 兼容性考虑

### 与32位代码交互
```assembly
; 32位和64位代码交互
; 32位调用64位函数
; 参数在W0-W7中传递

; 64位函数处理32位调用
MIXED_FUNCTION:
    ; 32位参数在W0-W7的低32位
    ; 需要符号扩展到64位
    
    SXTW X0, W0                 ; 符号扩展到64位
    SXTW X1, W1                 ; 符号扩展到64位
    
    ; 正常处理64位参数
    ADD X0, X0, X1
    
    ; 返回值在W0中（低32位）
    RET
```

### 不同编译器兼容性
```assembly
; GCC和Clang兼容性
COMPATIBLE_FUNCTION:
    ; 遵循标准AAPCS64
    ; 避免编译器特定的扩展
    
    ; 标准的寄存器使用
    STP X29, X30, [SP, #-16]!
    MOV X29, SP
    
    ; 避免非标准指令
    ; 使用标准指令集
    
    LDP X29, X30, [SP], #16
    RET
```

## 总结

AAPCS64调用约定的核心要点：
- **参数传递**：X0-X7用于整数，V0-V7用于浮点
- **返回值**：X0用于整数，V0用于浮点
- **寄存器保存**：调用者保存X9-X15，被调用者保存X19-X28
- **栈对齐**：SP必须16字节对齐
- **栈帧管理**：标准序言和尾声模式

遵循这些约定确保代码的互操作性和稳定性。
