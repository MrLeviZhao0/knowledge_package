# AArch64寄存器系统详解

## 寄存器概述

AArch64是ARMv8-A架构的64位指令集，拥有31个通用寄存器和丰富的特殊寄存器。理解寄存器系统是掌握AArch64汇编的关键。

## 通用寄存器

### 64位寄存器 (X0-X30)
```assembly
; 64位通用寄存器，用于存储地址和数据
MOV X0, #0x123456789ABCDEF0  ; 64位立即数
MOV X1, X0                   ; 寄存器间移动
ADD X2, X0, X1               ; 64位加法
```

### 32位寄存器 (W0-W30)
```assembly
; 32位视图，访问寄存器的低32位
MOV W0, #0x12345678          ; 32位立即数
ADD W1, W0, #1               ; 32位加法

; 64位和32位寄存器关系
MOV X0, #0x123456789ABCDEF0  ; X0 = 0x123456789ABCDEF0
MOV W1, W0                    ; W1 = 0x9ABCDEF0 (低32位)
```

## 特殊寄存器

### 栈指针寄存器 (SP)
```assembly
; 栈指针，指向当前栈顶
MOV X0, SP                    ; 读取栈指针
MOV SP, X0                    ; 设置栈指针

; 栈操作
STR X0, [SP, #-16]!           ; 压栈：SP先减16再存储
LDR X1, [SP], #16             ; 出栈：先加载后SP加16
```

### 零寄存器 (XZR/WZR)
```assembly
; 读取始终返回0，写入被忽略
MOV X0, XZR                   ; X0 = 0
CMP X1, XZR                    ; 比较X1是否等于0
ADD X2, X1, XZR               ; X2 = X1 + 0

; 32位零寄存器
MOV W0, WZR                    ; W0 = 0
```

### 程序计数器 (PC)
```assembly
; 不能直接访问，但可通过分支指令间接使用
B label                       ; 无条件跳转
BL function                   ; 跳转并链接（函数调用）
BR X0                         ; 跳转到X0指定的地址

; 相对跳转
ADR X0, label                 ; 将label的地址加载到X0
```

## 寄存器功能分配 (AAPCS64标准)

### 参数传递寄存器
```assembly
; X0-X7: 用于参数传递和整数返回值
; 函数调用示例
MOV X0, #1                    ; 第一个参数
MOV X1, #2                    ; 第二个参数
MOV X2, #3                    ; 第三个参数
MOV X3, #4                    ; 第四个参数
MOV X4, #5                    ; 第五个参数
MOV X5, #6                    ; 第六个参数
MOV X6, #7                    ; 第七个参数
MOV X7, #8                    ; 第八个参数
BL my_function                ; 调用函数
; 返回值在X0中
```

### 临时寄存器
```assembly
; X9-X15: 临时寄存器（调用者保存）
; 在函数调用前需要保存，调用后可能被修改

MY_FUNCTION:
    ; 保存临时寄存器（如果需要）
    STP X9, X10, [SP, #-16]!
    
    ; 使用临时寄存器
    MOV X9, X0                 ; 参数备份
    MOV X10, #100
    ADD X0, X9, X10
    
    ; 恢复临时寄存器
    LDP X9, X10, [SP], #16
    RET
```

### 被调用者保存寄存器
```assembly
; X19-X28: 被调用者保存寄存器
; 如果函数要使用这些寄存器，必须保存并在返回前恢复

MY_FUNCTION:
    ; 序言：保存被调用者保存寄存器
    STP X29, X30, [SP, #-32]!  ; 保存帧指针和返回地址
    STP X19, X20, [SP, #16]    ; 保存X19,X20
    MOV X29, SP                 ; 设置新帧指针
    
    ; 函数体
    MOV X19, X0                ; 使用X19
    MOV X20, #200
    ADD X0, X19, X20
    
    ; 尾声：恢复寄存器
    LDP X19, X20, [SP, #16]    ; 恢复X19,X20
    LDP X29, X30, [SP], #32    ; 恢复帧指针和返回地址
    RET
```

### 特殊用途寄存器
```assembly
; X8: 间接结果位置寄存器
; 用于返回大型结构体

; X16-X17: 内部过程调用临时寄存器
; 用于链接器和动态链接

; X18: 平台寄存器
; 平台特定用途，如TLS（线程本地存储）

; X29: 帧指针 (FP)
; X30: 链接寄存器 (LR)
```

## 浮点寄存器

### SIMD和浮点寄存器 (V0-V31)
```assembly
; 128位寄存器，支持多种数据视图
; 可以作为标量或向量使用

; 标量视图
FMOV S0, 1.0                  ; 单精度浮点数 (32位)
FMOV D0, 1.0                  ; 双精度浮点数 (64位)

; 向量视图 (NEON)
MOVI V0.4S, #0x3F800000       ; 4个单精度浮点数 (1.0)
MOVI V1.2D, #0x3FF0000000000000 ; 2个双精度浮点数 (1.0)
```

### 数据视图
```assembly
; 不同位宽的视图
; V0.16B: 16个字节 (8位)
; V0.8H: 8个半字 (16位)  
; V0.4S: 4个字 (32位)
; V0.2D: 2个双字 (64位)
; V0.1Q: 1个四字 (128位)

; 实际使用
MOV V0.16B, V1.16B            ; 复制128位数据
ADD V0.4S, V1.4S, V2.4S       ; 4个单精度浮点数相加
```

## 栈帧管理

### 标准函数序言和尾声
```assembly
; 标准栈帧布局
MY_FUNCTION:
    ; 序言
    STP X29, X30, [SP, #-16]!  ; 保存帧指针和返回地址
    MOV X29, SP                 ; 设置新帧指针
    
    ; 可选：分配局部变量空间
    SUB SP, SP, #32             ; 分配32字节栈空间
    
    ; 函数体
    STR X0, [X29, #-8]          ; 保存参数到栈帧
    LDR X1, [X29, #-8]          ; 从栈帧加载
    
    ; 尾声
    MOV SP, X29                 ; 恢复栈指针
    LDP X29, X30, [SP], #16     ; 恢复帧指针和返回地址
    RET
```

### 复杂栈帧示例
```assembly
COMPLEX_FUNCTION:
    ; 序言：保存多个寄存器
    STP X29, X30, [SP, #-48]!   ; 分配48字节栈空间
    STP X19, X20, [SP, #16]     ; 保存X19,X20
    STP X21, X22, [SP, #32]     ; 保存X21,X22
    MOV X29, SP
    
    ; 函数体
    ; 使用保存的寄存器
    MOV X19, X0
    MOV X20, X1
    MOV X21, X2
    MOV X22, X3
    
    ; 复杂的计算...
    
    ; 尾声：恢复所有寄存器
    LDP X21, X22, [SP, #32]     ; 恢复X21,X22
    LDP X19, X20, [SP, #16]     ; 恢复X19,X20
    LDP X29, X30, [SP], #48     ; 恢复帧指针和返回地址
    RET
```

## 条件执行和标志寄存器

### 条件标志
AArch64使用NZCV条件标志：
- **N** (Negative): 结果为负
- **Z** (Zero): 结果为零  
- **C** (Carry): 无符号溢出
- **V** (Overflow): 有符号溢出

```assembly
; 比较操作设置条件标志
CMP X0, X1                     ; 设置NZCV标志

; 条件执行
B.EQ label                     ; 如果Z=1（相等）跳转
B.NE label                     ; 如果Z=0（不相等）跳转
B.GT label                     ; 如果Z=0且N=V（大于）跳转
B.LT label                     ; 如果N≠V（小于）跳转

; 条件选择
CSEL X0, X1, X2, EQ           ; 如果相等，X0=X1，否则X0=X2
```

## 实际应用示例

### 函数参数传递分析
```assembly
; C函数: int64_t complex_calc(int64_t a, int64_t b, int64_t c, int64_t d);

; 调用代码
MOV X0, #10                    ; a = 10
MOV X1, #20                    ; b = 20
MOV X2, #30                    ; c = 30
MOV X3, #40                    ; d = 40
BL complex_calc               ; 调用函数
; 返回值在X0中

; 函数实现
COMPLEX_CALC:
    ; 寄存器使用情况：
    ; X0: 参数a，后用作返回值
    ; X1: 参数b
    ; X2: 参数c  
    ; X3: 参数d
    ; X8: 临时计算
    
    ADD X8, X0, X1             ; X8 = a + b
    ADD X8, X8, X2             ; X8 += c
    ADD X0, X8, X3             ; X0 = X8 + d (返回值)
    
    RET
```

### 循环结构分析
```assembly
; C代码: for (int i = 0; i < 100; i++) sum += i;

MOV X0, XZR                    ; sum = 0
MOV X1, XZR                    ; i = 0
MOV X2, #100                   ; 循环上限

LOOP_START:
    CMP X1, X2                 ; 比较 i < 100
    B.GE LOOP_END              ; 如果 i >= 100，跳出循环
    
    ADD X0, X0, X1             ; sum += i
    ADD X1, X1, #1             ; i++
    
    B LOOP_START               ; 继续循环
    
LOOP_END:
    ; X0包含最终结果
```

### 数组访问模式
```assembly
; C代码: int sum = 0; for (int i = 0; i < n; i++) sum += array[i];

; 假设：
; X0: array指针
; X1: n (数组长度)
; X2: sum
; X3: i (索引)
; X4: 临时值

MOV X2, XZR                    ; sum = 0
MOV X3, XZR                    ; i = 0

ARRAY_LOOP:
    CMP X3, X1                 ; 比较 i < n
    B.GE ARRAY_END
    
    LDR W4, [X0, X3, LSL #2]   ; W4 = array[i] (int数组，左移2位)
    ADD X2, X2, X4             ; sum += array[i]
    
    ADD X3, X3, #1             ; i++
    B ARRAY_LOOP
    
ARRAY_END:
    ; X2包含数组元素和
```

## 性能优化考虑

### 寄存器分配策略
```assembly
; 优化寄存器使用
; 1. 优先使用X0-X7（参数寄存器）
; 2. 合理使用X9-X15（临时寄存器）
; 3. 谨慎使用X19-X28（需要保存/恢复）
; 4. 避免不必要的寄存器移动

; 不好的代码
MOV X8, X0
ADD X8, X8, #1
MOV X0, X8

; 好的代码
ADD X0, X0, #1
```

### 流水线优化
```assembly
; 减少数据依赖
; 不好的代码（数据依赖）
LDR X0, [X1]           ; 加载数据
ADD X0, X0, #1         ; 依赖上一条指令
STR X0, [X1]           ; 存储结果

; 好的代码（减少依赖）
LDR X0, [X1]           ; 加载数据
LDR X2, [X3]           ; 加载其他数据（不依赖X0）
ADD X0, X0, #1         ; 此时X0已经就绪
STR X0, [X1]           ; 存储结果
```

## 总结

AArch64寄存器系统的核心要点：
- **31个通用寄存器**提供丰富的计算资源
- **标准调用约定**确保函数间正确交互
- **栈帧管理**维护函数调用上下文
- **条件执行**支持灵活的流程控制
- **SIMD寄存器**提供强大的向量运算能力

深入理解寄存器系统是掌握AArch64汇编和进行性能优化的基础。
