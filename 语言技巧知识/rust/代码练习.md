```rust
fn fibonacci_recursive(n: u64) -> u64 {
    match n {
        0 => 0,
        1 => 1,
        _ => fibonacci_recursive(n - 1) + fibonacci_recursive(n - 2),
    }
}

fn fibonacci_iterative(n: u64) -> u64 {
    if n <= 1 {
        return n;
    }
    
    let (mut a, mut b) = (0, 1);
    for _ in 2..=n {
        let next = a + b;
        a = b;
        b = next;
    }
    b
}

fn fibonacci_memoized(n: u64, memo: &mut HashMap<u64, u64>) -> u64 {
    if let Some(&result) = memo.get(&n) {
        return result;
    }
    
    let result = match n {
        0 => 0,
        1 => 1,
        _ => fibonacci_memoized(n - 1, memo) + fibonacci_memoized(n - 2, memo),
    };
    
    memo.insert(n, result);
    result
}
```

```rust
    let chars: Vec<char> = s.chars().collect();
    let len = chars.len();
```

**参考答案：**
```rust
impl LinkedList {
    fn push(&mut self, data: i32) {
        let new_node = Rc::new(RefCell::new(Node {
            data,
            next: self.head.take(),
        }));
        self.head = Some(new_node);
    }
    
    fn pop(&mut self) -> Option<i32> {
        self.head.take().map(|node| {
            let node_ref = node.borrow();
            self.head = node_ref.next.clone();
            node_ref.data
        })
    }
    
    fn contains(&self, data: i32) -> bool {
        let mut current = self.head.clone();
        
        while let Some(node) = current {
            if node.borrow().data == data {
                return true;
            }
            current = node.borrow().next.clone();
        }
        
        false
    }
    
    fn reverse(&mut self) {
        let mut prev = None;
        let mut current = self.head.take();
        
        while let Some(node) = current {
            let next = node.borrow_mut().next.take();
            node.borrow_mut().next = prev;
            prev = Some(node);
            current = next;
        }
        
        self.head = prev;
    }
    
    fn to_vec(&self) -> Vec<i32> {
        let mut result = Vec::new();
        let mut current = self.head.clone();
        
        while let Some(node) = current {
            result.push(node.borrow().data);
            current = node.borrow().next.clone();
        }
        
        result
    }
}
```

```rust
impl Counter {
```

```rust
impl fmt::Display for FileError {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        match self {
            FileError::Io(e) => write!(f, "IO错误: {}", e),
            FileError::EmptyFile => write!(f, "文件为空"),
            FileError::InvalidEncoding => write!(f, "文件编码无效"),
            FileError::FileTooLarge { size, max_size } => 
                write!(f, "文件太大: {}字节 > {}字节限制", size, max_size),
        }
    }

impl std::error::Error for FileError {
    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
        match self {

impl FileProcessor {
    fn read_file(&self, path: &str) -> Result<String, FileError> {
        let metadata = fs::metadata(path)?;
        
        if metadata.len() > self.max_file_size {
            return Err(FileError::FileTooLarge {
                size: metadata.len(),
                max_size: self.max_file_size,
            });
        }
        
        let content = fs::read_to_string(path)?;
        
        if content.is_empty() {
            return Err(FileError::EmptyFile);
        }
        
        Ok(content)
    }
    
    fn write_file(&self, path: &str, content: &str) -> Result<(), FileError> {
        fs::write(path, content)?;
        Ok(())
    }
    
    fn file_stats(&self, path: &str) -> Result<FileStats, FileError> {
        let content = self.read_file(path)?;
        let metadata = fs::metadata(path)?;
        
        let line_count = content.lines().count();
        let word_count = content.split_whitespace().count();
        let char_count = content.chars().count();
        
        Ok(FileStats {
            size: metadata.len(),
            line_count,
            word_count,
            char_count,
        })
    }
}
```

```rust
// 创建字符串向量的宏
macro_rules! vec_of_strings {
    ($($element:expr),*) => {
        {
            let mut temp_vec = Vec::new();
            $(temp_vec.push($element.to_string());)*
            temp_vec
        }
    };
    () => {
        Vec::<String>::new()
    };
}

// 简单计算宏
macro_rules! calculate {
    ($expression:expr) => {
        $expression
    };
}

// 调试打印宏
macro_rules! debug_print {
    ($($var:ident),*) => {
        println!("[DEBUG] {}", {
            let mut output = String::new();
            $(
                if !output.is_empty() {
                    output.push_str(", ");
                }
                output.push_str(&format!("{} = {:?}", stringify!($var), $var));
            )*
            output
        });
    };
```
