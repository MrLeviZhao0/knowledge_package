```rust
// src/main.rs
use actix_web::{web, App, HttpServer};
use std::env;

mod handlers;
mod models;
mod database;
mod middleware;

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    // 初始化日志
    env_logger::init();
    
    // 读取配置
    dotenv::dotenv().ok();
    let database_url = env::var("DATABASE_URL")
        .expect("DATABASE_URL must be set");
    let port = env::var("PORT")
        .unwrap_or_else(|_| "8080".to_string())
        .parse::<u16>()
        .expect("PORT must be a valid number");
    
    // 初始化数据库连接池
    let pool = database::connection::create_pool(&database_url).await
        .expect("Failed to create database pool");
    
    log::info!("Starting server on port {}", port);
    
    HttpServer::new(move || {
        App::new()
            .app_data(web::Data::new(pool.clone()))
            .wrap(middleware::auth::Authentication)
            .wrap(actix_web::middleware::Logger::default())
            .service(
                web::scope("/api")
                    .service(
                        web::scope("/users")
                            .route("", web::post().to(handlers::user::create_user))
                            .route("/{id}", web::get().to(handlers::user::get_user))
                            .route("/{id}", web::put().to(handlers::user::update_user))
                            .route("/{id}", web::delete().to(handlers::user::delete_user))
                    )
                    .service(
                        web::scope("/products")
                            .route("", web::get().to(handlers::product::list_products))
                            .route("", web::post().to(handlers::product::create_product))
                    )
            )
            .route("/health", web::get().to(|| async { "OK" }))
    })
    .bind(("0.0.0.0", port))?
    .run()
    .await
}
```

```rust
// src/handlers/user.rs
use actix_web::{web, HttpResponse, Result};
use sqlx::PgPool;
use bcrypt::{hash, DEFAULT_COST};
use uuid::Uuid;

use crate::models::user::{User, CreateUserRequest, UserResponse};

pub async fn create_user(
    pool: web::Data<PgPool>,
    user_data: web::Json<CreateUserRequest>,
) -> Result<HttpResponse> {
    // 验证输入
    if user_data.username.is_empty() || user_data.email.is_empty() {
        return Ok(HttpResponse::BadRequest().json("用户名和邮箱不能为空"));
    }
    
    // 哈希密码
    let password_hash = hash(&user_data.password, DEFAULT_COST)
        .map_err(|e| {
            log::error!("密码哈希失败: {}", e);
            actix_web::error::ErrorInternalServerError("服务器错误")
        })?;
    
    // 插入数据库
    let user = sqlx::query_as::<_, User>(
        r#"
        INSERT INTO users (username, email, password_hash)
        VALUES ($1, $2, $3)
        RETURNING id, username, email, password_hash, created_at, updated_at
        "#
    )
    .bind(&user_data.username)
    .bind(&user_data.email)
    .bind(&password_hash)
    .fetch_one(pool.get_ref())
    .await
    .map_err(|e| {
        log::error!("数据库插入失败: {}", e);
        actix_web::error::ErrorInternalServerError("服务器错误")
    })?;
    
    Ok(HttpResponse::Created().json(UserResponse::from(user)))
}

pub async fn get_user(
    pool: web::Data<PgPool>,
    user_id: web::Path<Uuid>,
) -> Result<HttpResponse> {
    let user = sqlx::query_as::<_, User>(
        "SELECT id, username, email, password_hash, created_at, updated_at FROM users WHERE id = $1"
    )
    .bind(*user_id)
    .fetch_optional(pool.get_ref())
    .await
    .map_err(|e| {
        log::error!("数据库查询失败: {}", e);
        actix_web::error::ErrorInternalServerError("服务器错误")
    })?;
    
    match user {
        Some(user) => Ok(HttpResponse::Ok().json(UserResponse::from(user))),
        None => Ok(HttpResponse::NotFound().json("用户不存在")),
    }
}

pub async fn update_user(
    pool: web::Data<PgPool>,
    user_id: web::Path<Uuid>,
    user_data: web::Json<CreateUserRequest>,
) -> Result<HttpResponse> {
    // 实现更新逻辑
    Ok(HttpResponse::Ok().json("用户更新成功"))
}

pub async fn delete_user(
    pool: web::Data<PgPool>,
    user_id: web::Path<Uuid>,
) -> Result<HttpResponse> {
    let result = sqlx::query("DELETE FROM users WHERE id = $1")
        .bind(*user_id)
        .execute(pool.get_ref())
        .await
        .map_err(|e| {
            log::error!("数据库删除失败: {}", e);
            actix_web::error::ErrorInternalServerError("服务器错误")
        })?;
    
    if result.rows_affected() > 0 {
        Ok(HttpResponse::Ok().json("用户删除成功"))
    } else {
        Ok(HttpResponse::NotFound().json("用户不存在"))
    }
}
```

```rust
use clap::Parser;
use std::path::PathBuf;

mod search;
mod config;
mod output;

#[derive(Parser, Debug)]
#[command(name = "rsearch")]
#[command(about = "A fast file search tool written in Rust")]
struct Cli {
    /// Search pattern (regex supported)
    pattern: String,
    
    /// Search directory (default: current directory)
    #[arg(short, long, default_value = ".")]
    directory: PathBuf,
    
    /// Case insensitive search
    #[arg(short, long)]
    ignore_case: bool,
    
    /// Search file names only (not content)
    #[arg(short, long)]
    name_only: bool,
    
    /// Output format (text, json)
    #[arg(short, long, default_value = "text")]
    format: String,
    
    /// Maximum depth to search
    #[arg(short = 'd', long)]
    max_depth: Option<usize>,
    
    /// Include hidden files and directories
    #[arg(long)]
    hidden: bool,
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    env_logger::init();
    
    let args = Cli::parse();
    
    // 验证目录存在
    if !args.directory.exists() {
        eprintln!("错误: 目录不存在: {:?}", args.directory);
        std::process::exit(1);
    }
    
    // 创建搜索配置
    let config = config::SearchConfig {
        pattern: args.pattern,
        directory: args.directory,
        ignore_case: args.ignore_case,
        name_only: args.name_only,
        max_depth: args.max_depth,
        include_hidden: args.hidden,
    };
    
    // 执行搜索
    let results = search::search_files(&config)?;
    
    // 输出结果
    output::print_results(&results, &args.format)?;
    
    Ok(())
}
```

**搜索核心逻辑：**
```rust
// src/search.rs
use crate::config::SearchConfig;
use regex::Regex;
use ignore::WalkBuilder;
use std::path::Path;
use rayon::prelude::*;
use std::fs;
use std::io::{self, BufRead};

#[derive(Debug, serde::Serialize)]
pub struct SearchResult {
    pub file_path: String,
    pub line_number: Option<usize>,
    pub line_content: Option<String>,
    pub match_start: Option<usize>,
    pub match_end: Option<usize>,
}

pub fn search_files(config: &SearchConfig) -> Result<Vec<SearchResult>, Box<dyn std::error::Error>> {
    let pattern = if config.ignore_case {
        format!(r"(?i){}", config.pattern)
    } else {
        config.pattern.clone()
    };
    
    let regex = Regex::new(&pattern)?;
    
    // 构建文件遍历器
    let walker = WalkBuilder::new(&config.directory)
        .hidden(config.include_hidden)
        .max_depth(config.max_depth)
        .build();
    
    let results: Vec<SearchResult> = walker
        .filter_map(|entry| entry.ok())
        .filter(|entry| entry.file_type().map(|ft| ft.is_file()).unwrap_or(false))
        .par_bridge() // 并行处理
        .flat_map(|entry| {
            let path = entry.path();
            
            if config.name_only {
                // 只搜索文件名
                search_filename(path, &regex)
            } else {
                // 搜索文件内容
                search_file_content(path, &regex)
            }
        })
        .collect();
    
    Ok(results)
}

fn search_filename(path: &Path, regex: &Regex) -> Option<SearchResult> {
    let file_name = path.file_name()?.to_string_lossy();
    
    if regex.is_match(&file_name) {
        Some(SearchResult {
            file_path: path.to_string_lossy().into_owned(),
            line_number: None,
            line_content: None,
            match_start: None,
            match_end: None,
        })
    } else {
        None
    }
}

fn search_file_content(path: &Path, regex: &Regex) -> Vec<SearchResult> {
    let file = match fs::File::open(path) {
        Ok(file) => file,
        Err(_) => return Vec::new(),
    };
    
    let reader = io::BufReader::new(file);
    let mut results = Vec::new();
    
    for (line_num, line) in reader.lines().enumerate() {
        let line = match line {
            Ok(line) => line,
            Err(_) => continue,
        };
        
        if let Some(mat) = regex.find(&line) {
            results.push(SearchResult {
                file_path: path.to_string_lossy().into_owned(),
                line_number: Some(line_num + 1),
                line_content: Some(line.clone()),
                match_start: Some(mat.start()),
                match_end: Some(mat.end()),
            });
        }
    }
    
    results
}
```

```rust
// src/allocator.rs
use std::alloc::{GlobalAlloc, Layout, System};
use std::sync::atomic::{AtomicUsize, Ordering};

pub struct TrackingAllocator {
    allocated: AtomicUsize,
    system: System,
}

impl TrackingAllocator {
    pub const fn new() -> Self {
        Self {
            allocated: AtomicUsize::new(0),
            system: System,
        }
    }
    
    pub fn allocated_bytes(&self) -> usize {
        self.allocated.load(Ordering::Relaxed)
    }
}

unsafe impl GlobalAlloc for TrackingAllocator {
    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
        let ptr = self.system.alloc(layout);
        if !ptr.is_null() {
            self.allocated.fetch_add(layout.size(), Ordering::Relaxed);
        }
        ptr
    }
    
    unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) {
        self.system.dealloc(ptr, layout);
        self.allocated.fetch_sub(layout.size(), Ordering::Relaxed);
    }
    
    unsafe fn realloc(&self, ptr: *mut u8, layout: Layout, new_size: usize) -> *mut u8 {
        let new_ptr = self.system.realloc(ptr, layout, new_size);
        if !new_ptr.is_null() {
            self.allocated.fetch_add(new_size, Ordering::Relaxed);
            self.allocated.fetch_sub(layout.size(), Ordering::Relaxed);
        }
        new_ptr
    }
}

#[global_allocator]
static GLOBAL: TrackingAllocator = TrackingAllocator::new();

// 竞技场分配器（Arena Allocator）
pub struct Arena {
    memory: Vec<u8>,
    current: usize,
    alignment: usize,
}

impl Arena {
    pub fn new(capacity: usize, alignment: usize) -> Self {
        Self {
            memory: vec![0; capacity],
            current: 0,
            alignment,
        }
    }
    
    pub fn allocate<T>(&mut self, value: T) -> &mut T {
        let layout = std::alloc::Layout::new::<T>();
        let aligned_offset = self.align_offset(self.current, layout.align());
        
        if aligned_offset + layout.size() > self.memory.len() {
            panic!("Arena out of memory");
        }
        
        unsafe {
            let ptr = self.memory.as_mut_ptr().add(aligned_offset) as *mut T;
            ptr.write(value);
            self.current = aligned_offset + layout.size();
            &mut *ptr
        }
    }
    
    fn align_offset(&self, offset: usize, align: usize) -> usize {
        let remainder = offset % align;
        if remainder == 0 {
            offset
        } else {
            offset + align - remainder
        }
    }
    
    pub fn clear(&mut self) {
        self.current = 0;
    }
}
```

### 缓存友好的数据结构

**优化矩阵乘法：**
```rust
// 缓存友好的矩阵乘法
pub fn matrix_multiply_optimized(a: &[f64], b: &[f64], c: &mut [f64], n: usize) {
    // 分块大小（根据CPU缓存大小调整）
    const BLOCK_SIZE: usize = 64;
    
    for i in (0..n).step_by(BLOCK_SIZE) {
        for j in (0..n).step_by(BLOCK_SIZE) {
            for k in (0..n).step_by(BLOCK_SIZE) {
                // 处理分块
                let i_end = (i + BLOCK_SIZE).min(n);
                let j_end = (j + BLOCK_SIZE).min(n);
                let k_end = (k + BLOCK_SIZE).min(n);
                
                for ii in i..i_end {
                    for kk in k..k_end {
                        let a_val = a[ii * n + kk];
                        for jj in j..j_end {
                            c[ii * n + jj] += a_val * b[kk * n + jj];
                        }
                    }
                }
            }
        }
    }
}

// 使用SIMD优化
#[cfg(target_arch = "x86_64")]
use std::arch::x86_64::*;

#[cfg(target_arch = "x86_64")]
pub unsafe fn matrix_multiply_simd(a: &[f64], b: &[f64], c: &mut [f64], n: usize) {
    for i in 0..n {
        for k in 0..n {
            let a_val = _mm256_set1_pd(a[i * n + k]);
            
            for j in (0..n).step_by(4) {
                let b_val = _mm256_loadu_pd(&b[k * n + j]);
                let c_val = _mm256_loadu_pd(&c[i * n + j]);
                let result = _mm256_fmadd_pd(a_val, b_val, c_val);
                _mm256_storeu_pd(&mut c[i * n + j], result);
            }
        }
    }
}
```
