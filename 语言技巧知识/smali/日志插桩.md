# Smali日志插桩技术

## 日志插桩概述

日志插桩是在Smali代码中插入日志输出语句的技术，用于监控应用运行时的行为、调试问题和分析恶意代码。这是Android逆向工程中最重要的动态分析技术之一。

## 基础插桩技术

## 自动化插桩工具

### 批量插桩脚本
```python
#!/usr/bin/env python3
# Smali批量插桩工具

import os
import re

class SmaliInstrumenter:
    def __init__(self):
        self.instrumented_count = 0
    
    def instrument_method_entry(self, smali_content, class_name, method_name):
        """在方法入口插入日志"""
        
        # 查找方法开始
        method_pattern = r'(\.method\s+.*?\s+' + re.escape(method_name) + r'\(.*?\)[^\n]*\n)(.*?)(\.end method)'
        
        def instrument_match(match):
            method_declaration = match.group(1)
            method_body = match.group(2)
            method_end = match.group(3)
            
            # 分析寄存器使用
            registers_match = re.search(r'\.registers\s+(\d+)', method_body)
            if registers_match:
                register_count = int(registers_match.group(1))
                # 增加寄存器数量
                method_body = re.sub(r'\.registers\s+\d+', f'.registers {register_count + 2}', method_body)
            
            # 插入日志代码
            log_code = f'''
    # 自动插桩 - 方法入口
    const-string v{register_count}, "{class_name}"
    const-string v{register_count + 1}, "{method_name} called"
    invoke-static {{v{register_count}, v{register_count + 1}}}, Landroid/util/Log;->d(Ljava/lang/String;Ljava/lang/String;)I
'''
            
            # 在方法体开始处插入
            lines = method_body.split('\n')
            for i, line in enumerate(lines):
                if '.registers' in line or '.locals' in line or '.param' in line:
                    continue
                if line.strip() and not line.strip().startswith('#'):
                    # 找到第一个非空非注释行
                    lines.insert(i, log_code)
                    break
            
            method_body = '\n'.join(lines)
            self.instrumented_count += 1
            
            return method_declaration + method_body + method_end
        
        return re.sub(method_pattern, instrument_match, smali_content, flags=re.DOTALL)
    
    def instrument_file(self, file_path, output_path):
        """插桩单个Smali文件"""
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        # 提取类名
        class_match = re.search(r'\.class\s+.*?\s+(L[^;]+;)', content)
        if not class_match:
            return
        
        class_name = class_match.group(1).split('/')[-1].replace(';', '')
        
        # 插桩所有公共方法
        method_pattern = r'\.method\s+public\s+(\w+)\([^)]*\)'
        methods = re.findall(method_pattern, content)
        
        for method_name in methods:
            if not method_name.startswith('access$'):  # 跳过访问器方法
                content = self.instrument_method_entry(content, class_name, method_name)
        
        # 写入输出文件
        with open(output_path, 'w', encoding='utf-8') as f:
            f.write(content)

# 使用示例
if __name__ == "__main__":
    instrumenter = SmaliInstrumenter()
    
    # 插桩整个Smali目录
    for root, dirs, files in os.walk("smali_output"):
        for file in files:
            if file.endswith(".smali"):
                file_path = os.path.join(root, file)
                output_path = os.path.join("instrumented_output", os.path.relpath(file_path, "smali_output"))
                os.makedirs(os.path.dirname(output_path), exist_ok=True)
                instrumenter.instrument_file(file_path, output_path)
    
    print(f"插桩完成，共处理 {instrumenter.instrumented_count} 个方法")
```
