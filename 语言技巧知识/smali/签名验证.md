# Smali签名验证技术

## 签名验证概述

APK签名验证是Android安全机制的核心部分，用于确保应用的完整性和来源可信性。理解签名验证机制对于应用安全分析和加固绕过至关重要。

## APK签名机制

### 签名文件结构

#### APK签名块分析
```smali
# APK签名相关文件
META-INF/
├── MANIFEST.MF          # 清单文件哈希
├── CERT.SF              # 签名文件
└── CERT.RSA             # 数字签名和证书

# 签名验证流程
1. 验证MANIFEST.MF中每个文件的哈希
2. 验证CERT.SF对MANIFEST.MF的签名
3. 验证CERT.RSA证书链
```

## 重打包和重新签名

### 使用Apktool重打包

#### 基本重打包流程
```bash
# 1. 反编译APK
apktool d original.apk -o decoded

# 2. 修改Smali代码（绕过签名验证）
# 编辑decoded/smali/... 中的相关文件

# 3. 重新打包
apktool b decoded -o unsigned.apk

# 4. 重新签名
jarsigner -verbose -sigalg SHA1withRSA -digestalg SHA1 \
    -keystore my.keystore unsigned.apk alias_name

# 5. 优化APK（可选）
zipalign -v 4 unsigned.apk final.apk
```

#### 自动化重打包脚本
```python
#!/usr/bin/env python3
# 自动化重打包和签名脚本

import os
import subprocess
import sys

class ApkRepacker:
    def __init__(self, apk_path, keystore_path, keystore_password, alias, alias_password):
        self.apk_path = apk_path
        self.keystore_path = keystore_path
        self.keystore_password = keystore_password
        self.alias = alias
        self.alias_password = alias_password
        
        self.work_dir = "repack_work"
        self.decoded_dir = os.path.join(self.work_dir, "decoded")
        self.unsigned_apk = os.path.join(self.work_dir, "unsigned.apk")
        self.final_apk = os.path.join(self.work_dir, "final.apk")
    
    def decompile(self):
        """反编译APK"""
        print("[*] Decompiling APK...")
        
        cmd = ["apktool", "d", self.apk_path, "-o", self.decoded_dir, "-f"]
        result = subprocess.run(cmd, capture_output=True, text=True)
        
        if result.returncode != 0:
            print(f"[-] Decompilation failed: {result.stderr}")
            return False
        
        print("[+] APK decompiled successfully")
        return True
    
    def patch_signature_checks(self):
        """修补签名验证代码"""
        print("[*] Patching signature checks...")
        
        # 查找并修改签名验证相关的Smali文件
        for root, dirs, files in os.walk(os.path.join(self.decoded_dir, "smali")):
            for file in files:
                if file.endswith(".smali"):
                    file_path = os.path.join(root, file)
                    
                    # 读取文件内容
                    with open(file_path, 'r', encoding='utf-8') as f:
                        content = f.read()
                    
                    # 简单的签名验证绕过
                    # 查找if-nez条件并修改为goto
                    patched_content = content.replace(
                        'if-nez v0, :cond_',
                        'goto :cond_'
                    )
                    
                    # 如果内容被修改，写入文件
                    if content != patched_content:
                        with open(file_path, 'w', encoding='utf-8') as f:
                            f.write(patched_content)
                        print(f"[+] Patched {file_path}")
        
        return True
    
    def repackage(self):
        """重新打包APK"""
        print("[*] Repackaging APK...")
        
        cmd = ["apktool", "b", self.decoded_dir, "-o", self.unsigned_apk]
        result = subprocess.run(cmd, capture_output=True, text=True)
        
        if result.returncode != 0:
            print(f"[-] Repackaging failed: {result.stderr}")
            return False
        
        print("[+] APK repackaged successfully")
        return True
    
    def sign_apk(self):
        """签名APK"""
        print("[*] Signing APK...")
        
        cmd = [
            "jarsigner", "-verbose", "-sigalg", "SHA1withRSA", 
            "-digestalg", "SHA1", "-keystore", self.keystore_path,
            "-storepass", self.keystore_password, self.unsigned_apk, self.alias
        ]
        
        result = subprocess.run(cmd, capture_output=True, text=True)
        
        if result.returncode != 0:
            print(f"[-] Signing failed: {result.stderr}")
            return False
        
        print("[+] APK signed successfully")
        return True
    
    def zipalign(self):
        """优化APK"""
        print("[*] Optimizing APK...")
        
        cmd = ["zipalign", "-v", "4", self.unsigned_apk, self.final_apk]
        result = subprocess.run(cmd, capture_output=True, text=True)
        
        if result.returncode != 0:
            print(f"[-] Zipalign failed: {result.stderr}")
            return False
        
        print("[+] APK optimized successfully")
        return True
    
    def repack(self):
        """执行完整的重打包流程"""
        
        # 创建工作目录
        os.makedirs(self.work_dir, exist_ok=True)
        
        steps = [
            self.decompile,
            self.patch_signature_checks,
            self.repackage,
            self.sign_apk,
            self.zipalign
        ]
        
        for step in steps:
            if not step():
                print("[-] Repacking failed")
                return False
        
        print(f"[+] Repacking completed. Final APK: {self.final_apk}")
        return True

# 使用示例
if __name__ == "__main__":
    if len(sys.argv) != 6:
        print("Usage: python repacker.py <apk_path> <keystore> <storepass> <alias> <aliaspass>")
        sys.exit(1)
    
    apk_path = sys.argv[1]
    keystore = sys.argv[2]
    storepass = sys.argv[3]
    alias = sys.argv[4]
    aliaspass = sys.argv[5]
    
    repacker = ApkRepacker(apk_path, keystore, storepass, alias, aliaspass)
    repacker.repack()
```

## 总结

1. **签名机制理解**：APK签名结构和验证流程
2. **验证代码分析**：PackageManager API和自定义验证逻辑
3. **绕过技术**：静态修改、动态Hook、重打包
