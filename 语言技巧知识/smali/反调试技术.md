# Smali反调试技术

## 反调试技术概述

反调试技术是恶意软件和受保护应用常用的防御手段，用于检测和阻止调试器的附加。理解这些技术对于逆向工程和安全分析至关重要。

## 自动化检测工具

### 反调试模式识别脚本
```python
#!/usr/bin/env python3
# 反调试模式识别工具

import re
import os

class AntiDebugDetector:
    def __init__(self):
        self.patterns = {
            'isDebuggerConnected': r'Landroid/os/Debug;->isDebuggerConnected',
            'tracerPid': r'/proc/self/status',
            'timingCheck': r'Ljava/lang/System;->(?:currentTimeMillis|nanoTime)',
            'ptrace': r'ptrace',
            'debugPort': r':\d+.*connect'
        }
        
        self.detected_patterns = {}
    
    def analyze_smali_file(self, file_path):
        """分析单个Smali文件中的反调试模式"""
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        for pattern_name, pattern in self.patterns.items():
            if re.search(pattern, content):
                if pattern_name not in self.detected_patterns:
                    self.detected_patterns[pattern_name] = []
                self.detected_patterns[pattern_name].append(file_path)
    
    def analyze_directory(self, directory):
        """分析整个目录"""
        for root, dirs, files in os.walk(directory):
            for file in files:
                if file.endswith('.smali'):
                    file_path = os.path.join(root, file)
                    self.analyze_smali_file(file_path)
    
    def generate_report(self):
        """生成分析报告"""
        report = "反调试技术检测报告\n"
        report += "=" * 50 + "\n"
        
        for pattern_name, files in self.detected_patterns.items():
            report += f"\n{pattern_name}:\n"
            report += f"  检测到 {len(files)} 个文件\n"
            for file in files[:5]:  # 只显示前5个文件
                report += f"  - {file}\n"
            if len(files) > 5:
                report += f"  ... 还有 {len(files) - 5} 个文件\n"
        
        return report

# 使用示例
if __name__ == "__main__":
    detector = AntiDebugDetector()
    detector.analyze_directory("smali_output")
    print(detector.generate_report())
```

# 自动反反调试工具

import re

class AntiAntiDebugPatcher:
    def __init__(self):
        self.patches = {
            'isDebuggerConnected': {
                'pattern': r'invoke-static \{\}, Landroid/os/Debug;->isDebuggerConnected\(\)Z',
                'replacement': 'const/4 v0, 0x0\n    # Patched: isDebuggerConnected'
            },
            'tracerPidCheck': {
                'pattern': r'/proc/self/status',
                'replacement': '/proc/self/cmdline'
            }
        }
    
    def patch_file(self, file_path, output_path):
        """修补单个文件"""
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        patched_content = content
        
        for patch_name, patch_info in self.patches.items():
            pattern = patch_info['pattern']
            replacement = patch_info['replacement']
            
            if re.search(pattern, patched_content):
                patched_content = re.sub(pattern, replacement, patched_content)
                print(f"[*] Applied {patch_name} patch to {file_path}")
        
        with open(output_path, 'w', encoding='utf-8') as f:
            f.write(patched_content)

# 使用示例

                file_path = os.path.join(root, file)
                output_path = os.path.join("patched_output", os.path.relpath(file_path, "smali_output"))
                os.makedirs(os.path.dirname(output_path), exist_ok=True)

## 总结

Smali反调试技术涉及多个层面：
1. **基础检测**：进程状态、时间检测、环境检查
2. **高级技术**：代码完整性、内存断点、多层检测
3. **绕过方法**：静态修改、动态Hook、工具辅助
4. **实战分析**：商业加固、恶意软件案例分析

掌握这些技术对于Android安全分析和逆向工程至关重要。
