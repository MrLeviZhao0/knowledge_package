# Smali资源保护技术

## 资源保护概述

资源保护是Android应用安全的重要组成部分，涉及字符串加密、资源文件保护、代码混淆等多个层面。理解资源保护技术对于逆向工程和安全分析至关重要。

#### 静态分析字符串解密
```python
#!/usr/bin/env python3
# 字符串解密分析工具

import re

class StringDecryptor:
    def __init__(self):
        self.decryption_patterns = {
            'xor': r'xor-int/lit8 v\d+, v\d+, 0x',
            'base64': r'Landroid/util/Base64;->decode',
            'string_builder': r'Ljava/lang/StringBuilder;->append'
        }
    
    def analyze_smali_file(self, file_path):
        """分析Smali文件中的字符串解密模式"""
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        findings = {}
        
        for pattern_name, pattern in self.decryption_patterns.items():
            matches = re.findall(pattern, content)
            if matches:
                findings[pattern_name] = matches
        
        # 查找加密字符串
        encrypted_strings = re.findall(r'const-string v\d+, "([^"]+)"', content)
        
        # 分析字符串特征
        suspicious_strings = []
        for s in encrypted_strings:
            if self.is_likely_encrypted(s):
                suspicious_strings.append(s)
        
        if suspicious_strings:
            findings['encrypted_strings'] = suspicious_strings
        
        return findings
    
    def is_likely_encrypted(self, string):
        """判断字符串是否可能被加密"""
        
        # 高熵字符串可能被加密
        if len(string) > 10 and self.calculate_entropy(string) > 4.0:
            return True
        
        # 包含非打印字符
        if any(ord(c) < 32 or ord(c) > 126 for c in string):
            return True
        
        # Base64特征
        if re.match(r'^[A-Za-z0-9+/]+={0,2}$', string) and len(string) % 4 == 0:
            return True
        
        return False
    
    def calculate_entropy(self, string):
        """计算字符串熵"""
        import math
        
        if not string:
            return 0
        
        entropy = 0
        for x in set(string):
            p_x = string.count(x) / len(string)
            entropy += -p_x * math.log(p_x, 2)
        
        return entropy
    
    def attempt_decryption(self, encrypted_string, method='xor'):
        """尝试解密字符串"""
        
        if method == 'xor':
            # 尝试常见的XOR密钥
            for key in range(256):
                decrypted = ''.join(chr(ord(c) ^ key) for c in encrypted_string)
                if self.looks_like_plaintext(decrypted):
                    return decrypted, key
        
        elif method == 'base64':
            import base64
            try:
                decoded = base64.b64decode(encrypted_string)
                return decoded.decode('utf-8', errors='ignore'), 'base64'
            except:
                pass
        
        return None, None
    
    def looks_like_plaintext(self, string):
        """判断字符串是否像明文"""
        
        # 检查是否包含常见单词
        common_words = ['http', 'com', 'org', 'api', 'key', 'password']
        if any(word in string.lower() for word in common_words):
            return True
        
        # 检查可打印字符比例
        printable_count = sum(1 for c in string if 32 <= ord(c) <= 126)
        if printable_count / len(string) > 0.8:
            return True
        
        return False

# 使用示例
if __name__ == "__main__":
    decryptor = StringDecryptor()
    
    # 分析Smali文件
    findings = decryptor.analyze_smali_file("Security.smali")
    
    for pattern, matches in findings.items():
        print(f"{pattern}: {len(matches)} matches")
        
        if pattern == 'encrypted_strings':
            for s in matches[:5]:  # 只显示前5个
                print(f"  Attempting to decrypt: {s}")
                
                # 尝试解密
                result, key = decryptor.attempt_decryption(s)
                if result:
                    print(f"    Decrypted: {result} (key: {key})")
```

## 总结

Smali资源保护技术涉及多个层面：
1. **字符串加密**：XOR、Base64、分段加密、动态构建
2. **资源文件保护**：资源ID加密、assets/raw文件加密
3. **代码保护**：反射调用、动态类加载、Native代码
4. **分析技术**：动态监控、静态分析、自动化工具
5. **商业保护**：360加固、爱加密等商业方案分析

掌握这些技术对于Android应用安全分析和保护都至关重要。
