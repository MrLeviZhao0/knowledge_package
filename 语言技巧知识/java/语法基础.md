# Java语法基础

## 概述
Java语法基础是学习Java编程的起点，涵盖了变量、数据类型、运算符、控制流等核心概念。本章详细介绍了Java的基本语法规则和编程范式。

## 基本语法结构

### Hello World程序
```java
public class HelloWorld {
    public static void main(String[] args) {
        System.out.println("Hello, World!");
    }
}
```

### 程序结构解析
- `public class HelloWorld`：定义公共类
- `public static void main(String[] args)`：程序入口点
- `System.out.println()`：输出语句

```java
public class VariableScope {
    // 实例变量（类级别）
    private int instanceVar = 10;
    
    // 静态变量（类级别）
    private static int staticVar = 20;
    
    public void method() {
        // 局部变量（方法级别）
        int localVar = 30;
        
        if (true) {
            // 块级变量
            int blockVar = 40;
            System.out.println("Block variable: " + blockVar);
        }
        // System.out.println(blockVar); // 错误：blockVar超出作用域

        System.out.println("Instance variable: " + instanceVar);
        System.out.println("Static variable: " + staticVar);
    }
    
    public static void main(String[] args) {
        VariableScope obj = new VariableScope();
        obj.method();
        
        // 直接访问静态变量
        System.out.println("Static var from main: " + staticVar);
    }
}
```

## 运算符

### 算术运算符
```java
public class ArithmeticOperators {
    public static void main(String[] args) {
        int a = 10, b = 3;
        
        System.out.println("a + b = " + (a + b));     // 13
        System.out.println("a - b = " + (a - b));     // 7
        System.out.println("a * b = " + (a * b));     // 30
        System.out.println("a / b = " + (a / b));     // 3（整数除法）
        System.out.println("a % b = " + (a % b));     // 1（取余）
        
        // 浮点数运算
        double x = 10.0, y = 3.0;
        System.out.println("x / y = " + (x / y));     // 3.333...
        
        // 自增自减
        int count = 5;
        System.out.println("count++ = " + count++);   // 5（后置）
        System.out.println("++count = " + ++count);   // 7（前置）
    }
}
```

### 关系运算符
```java
public class RelationalOperators {
    public static void main(String[] args) {
        int a = 10, b = 20;
        
        System.out.println("a == b: " + (a == b));    // false
        System.out.println("a != b: " + (a != b));    // true
        System.out.println("a < b: " + (a < b));      // true
        System.out.println("a > b: " + (a > b));      // false
        System.out.println("a <= b: " + (a <= b));    // true
        System.out.println("a >= b: " + (a >= b));    // false
        
        // 字符串比较
        String str1 = "hello";
        String str2 = "hello";
        String str3 = new String("hello");
        
        System.out.println("str1 == str2: " + (str1 == str2));        // true（字符串常量池）
        System.out.println("str1 == str3: " + (str1 == str3));        // false（不同对象）
        System.out.println("str1.equals(str3): " + str1.equals(str3)); // true（内容相同）
    }
}
```

### 逻辑运算符
```java
public class LogicalOperators {
    public static void main(String[] args) {
        boolean x = true, y = false;
        
        System.out.println("x && y: " + (x && y));    // false
        System.out.println("x || y: " + (x || y));    // true
        System.out.println("!x: " + (!x));            // false
        
        // 短路运算
        int a = 10;
        boolean result = (a > 5) || (a++ > 10);
        System.out.println("Result: " + result);      // true
        System.out.println("a after ||: " + a);       // 10（短路，a++未执行）
        
        result = (a < 5) && (a++ > 10);
        System.out.println("Result: " + result);      // false
        System.out.println("a after &&: " + a);       // 10（短路，a++未执行）
    }
}
```

## 控制流语句

### 条件语句
```java
public class ConditionalStatements {
    public static void main(String[] args) {
        int score = 85;
        
        // if-else语句
        if (score >= 90) {
            System.out.println("优秀");
        } else if (score >= 80) {
            System.out.println("良好");
        } else if (score >= 60) {
            System.out.println("及格");
        } else {
            System.out.println("不及格");
        }
        
        // switch语句（Java 14+增强）
        String grade;
        switch (score / 10) {
            case 10, 9 -> grade = "A";
            case 8 -> grade = "B";
            case 7 -> grade = "C";
            case 6 -> grade = "D";
            default -> grade = "F";
        }
        System.out.println("Grade: " + grade);
        
        // 传统switch语句
        int day = 3;
        switch (day) {
            case 1:
                System.out.println("星期一");
                break;
            case 2:
                System.out.println("星期二");
                break;
            case 3:
                System.out.println("星期三");
                break;
            default:
                System.out.println("其他");
        }
    }
}
```

### 循环语句
```java
public class LoopStatements {
    public static void main(String[] args) {
        // for循环
        System.out.println("=== for循环 ===");
        for (int i = 1; i <= 5; i++) {
            System.out.println("i = " + i);
        }
        
        // 增强for循环（foreach）
        System.out.println("\n=== 增强for循环 ===");
        int[] numbers = {1, 2, 3, 4, 5};
        for (int num : numbers) {
            System.out.println("Number: " + num);
        }
        
        // while循环
        System.out.println("\n=== while循环 ===");
        int count = 1;
        while (count <= 3) {
            System.out.println("Count: " + count);
            count++;
        }
        
        // do-while循环
        System.out.println("\n=== do-while循环 ===");
        int x = 1;
        do {
            System.out.println("x = " + x);
            x++;
        } while (x <= 3);
        
        // 循环控制语句
        System.out.println("\n=== 循环控制 ===");
        for (int i = 1; i <= 10; i++) {
            if (i % 2 == 0) {
                continue; // 跳过偶数
            }
            if (i > 7) {
                break; // 当i>7时退出循环
            }
            System.out.println("Odd number: " + i);
        }
    }
}
```

### 跳转语句
```java
public class JumpStatements {
    public static void main(String[] args) {
        // break语句
        System.out.println("=== break示例 ===");
        for (int i = 1; i <= 5; i++) {
            if (i == 3) {
                break; // 跳出循环
            }
            System.out.println("i = " + i);
        }
        
        // continue语句
        System.out.println("\n=== continue示例 ===");
        for (int i = 1; i <= 5; i++) {
            if (i % 2 == 0) {
                continue; // 跳过本次循环的剩余部分
            }
            System.out.println("Odd: " + i);
        }
        
        // 带标签的break和continue
        System.out.println("\n=== 标签跳转 ===");
        outer:
        for (int i = 1; i <= 3; i++) {
            inner:
            for (int j = 1; j <= 3; j++) {
                if (i * j > 4) {
                    break outer; // 跳出外层循环
                }
                System.out.println(i + " * " + j + " = " + (i * j));
            }
        }
        
        // return语句
        System.out.println("\n=== return示例 ===");
        int result = calculateSum(5);
        System.out.println("Sum: " + result);
    }
    
    public static int calculateSum(int n) {
        if (n <= 0) {
            return 0; // 提前返回
        }
        int sum = 0;
        for (int i = 1; i <= n; i++) {
            sum += i;
        }
        return sum;
    }
}
```

## 方法定义和调用

### 方法基础
```java
public class MethodBasics {
    
    // 无参数无返回值方法
    public static void greet() {
        System.out.println("Hello, welcome to Java!");
    }
    
    // 带参数无返回值方法
    public static void greetPerson(String name) {
        System.out.println("Hello, " + name + "!");
    }
    
    // 带参数有返回值方法
    public static int add(int a, int b) {
        return a + b;
    }
    
    // 方法重载（同名不同参数）
    public static double add(double a, double b) {
        return a + b;
    }
    
    public static int add(int a, int b, int c) {
        return a + b + c;
    }
    
    // 可变参数方法
    public static int sum(int... numbers) {
        int total = 0;
        for (int num : numbers) {
            total += num;
        }
        return total;
    }
    
    public static void main(String[] args) {
        // 方法调用
        greet();
        greetPerson("Alice");
        
        int result1 = add(5, 3);
        double result2 = add(2.5, 3.7);
        int result3 = add(1, 2, 3);
        int result4 = sum(1, 2, 3, 4, 5);
        
        System.out.println("5 + 3 = " + result1);
        System.out.println("2.5 + 3.7 = " + result2);
        System.out.println("1 + 2 + 3 = " + result3);
        System.out.println("Sum of 1-5 = " + result4);
        
        // 递归方法示例
        int factorial = factorial(5);
        System.out.println("5! = " + factorial);
    }
    
    // 递归方法
    public static int factorial(int n) {
        if (n <= 1) {
            return 1;
        }
        return n * factorial(n - 1);
    }
}
```

### 参数传递机制
```java
public class ParameterPassing {
    
    // 基本类型参数传递（值传递）
    public static void modifyPrimitive(int x) {
        x = x * 2;
        System.out.println("Inside method: x = " + x);
    }
    
    // 引用类型参数传递（引用值传递）
    public static void modifyArray(int[] arr) {
        if (arr.length > 0) {
            arr[0] = 999;
        }
    }
    
    // 字符串参数传递（字符串不可变）
    public static void modifyString(String str) {
        str = str + " modified";
        System.out.println("Inside method: str = " + str);
    }
    
    public static void main(String[] args) {
        // 基本类型测试
        int num = 10;
        System.out.println("Before: num = " + num);
        modifyPrimitive(num);
        System.out.println("After: num = " + num);
        
        // 数组测试
        int[] numbers = {1, 2, 3};
        System.out.println("\nBefore: numbers[0] = " + numbers[0]);
        modifyArray(numbers);
        System.out.println("After: numbers[0] = " + numbers[0]);
        
        // 字符串测试
        String text = "original";
        System.out.println("\nBefore: text = " + text);
        modifyString(text);
        System.out.println("After: text = " + text);
    }
}
```

## 输入输出操作

### 控制台输入输出
```java
import java.util.Scanner;

public class ConsoleIO {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        // 读取不同类型的数据
        System.out.print("请输入您的姓名: ");
        String name = scanner.nextLine();
        
        System.out.print("请输入您的年龄: ");
        int age = scanner.nextInt();
        
        System.out.print("请输入您的身高(米): ");
        double height = scanner.nextDouble();
        
        // 清除缓冲区
        scanner.nextLine();
        
        System.out.print("请输入您的职业: ");
        String job = scanner.nextLine();
        
        // 输出结果
        System.out.println("\n=== 个人信息 ===");
        System.out.println("姓名: " + name);
        System.out.println("年龄: " + age);
        System.out.println("身高: " + height + "米");
        System.out.println("职业: " + job);
        
        scanner.close();
    }
}
```

### 格式化输出
```java
public class FormattedOutput {
    public static void main(String[] args) {
        String name = "Alice";
        int age = 25;
        double salary = 75000.50;
        
        // 使用printf格式化输出
        System.out.printf("姓名: %s, 年龄: %d, 薪资: %.2f\n", name, age, salary);
```
