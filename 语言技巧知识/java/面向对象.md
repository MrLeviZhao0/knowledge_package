# Java面向对象编程

## 概述
面向对象编程（OOP）是Java的核心编程范式，通过类、对象、继承、多态、封装等概念来组织代码。本章详细介绍了Java面向对象编程的核心概念和实践。

## 类和对象

```java
public class Student {
```

### 构造方法重载
```java
public class Book {
    private String title;
    private String author;
    private double price;
    private int pages;
    
    // 默认构造方法
    public Book() {
        this.title = "未知";
        this.author = "未知作者";
        this.price = 0.0;
        this.pages = 0;
    }
    
    // 带参数的构造方法
    public Book(String title, String author) {
        this.title = title;
        this.author = author;
        this.price = 0.0;
        this.pages = 0;
    }
    
    // 完整参数的构造方法
    public Book(String title, String author, double price, int pages) {
        this.title = title;
        this.author = author;
        this.price = price;
        this.pages = pages;
    }
    
    // 拷贝构造方法
    public Book(Book other) {
        this.title = other.title;
        this.author = other.author;
        this.price = other.price;
        this.pages = other.pages;
    }
    
    // Getter和Setter方法
    public String getTitle() { return title; }
    public void setTitle(String title) { this.title = title; }
    
    public String getAuthor() { return author; }
    public void setAuthor(String author) { this.author = author; }
    
    public double getPrice() { return price; }
    public void setPrice(double price) { 
        if (price >= 0) this.price = price; 
    }
    
    public int getPages() { return pages; }
    public void setPages(int pages) { 
        if (pages > 0) this.pages = pages; 
    }
    
    @Override
    public String toString() {
        return String.format("《%s》- %s - ￥%.2f - %d页", 
                           title, author, price, pages);
    }
}

public class ConstructorDemo {
    public static void main(String[] args) {
```

### 访问控制修饰符
```java
public class BankAccount {
    // private: 只能在类内部访问
    private String accountNumber;
    private String accountHolder;
    private double balance;
    private String password;
    
    // protected: 同包或子类可访问
    protected String bankName;
    
    // default (包级私有): 同包可访问
    String branch;
    
    // public: 所有地方可访问
    public static final String CURRENCY = "CNY";
    
    public BankAccount(String accountNumber, String accountHolder, double initialBalance, String password) {
        this.accountNumber = accountNumber;
        this.accountHolder = accountHolder;
        this.balance = initialBalance;
        this.password = password;
        this.bankName = "中国银行";
        this.branch = "北京分行";
    }
    
    // 公共方法提供对私有属性的安全访问
    public boolean deposit(double amount, String inputPassword) {
        if (authenticate(inputPassword) && amount > 0) {
            balance += amount;
            System.out.println("存款成功，当前余额: " + balance);
            return true;
        }
        System.out.println("存款失败");
        return false;
    }
    
    public boolean withdraw(double amount, String inputPassword) {
        if (authenticate(inputPassword) && amount > 0 && amount <= balance) {
            balance -= amount;
            System.out.println("取款成功，当前余额: " + balance);
            return true;
        }
        System.out.println("取款失败");
        return false;
    }
    
    public double getBalance(String inputPassword) {
        if (authenticate(inputPassword)) {
            return balance;
        }
        System.out.println("密码错误");
        return -1;
    }
    
    // 私有方法，外部无法直接调用
    private boolean authenticate(String inputPassword) {
        return this.password.equals(inputPassword);
    }
    
    // 包级方法
    void changeBranch(String newBranch) {
        this.branch = newBranch;
        System.out.println("分行变更为: " + newBranch);
    }
    
    // 受保护方法
    protected void setBankName(String bankName) {
        this.bankName = bankName;
    }
    
    @Override
    public String toString() {
        return String.format("账户: %s, 持有人: %s, 余额: %.2f %s, 银行: %s, 分行: %s",
                           accountNumber, accountHolder, balance, CURRENCY, bankName, branch);
    }
}

public class EncapsulationDemo {
    public static void main(String[] args) {
        BankAccount account = new BankAccount("622848001234567890", "张三", 1000.0, "123456");
        
        // 只能通过公共方法访问
        account.deposit(500.0, "123456");
        account.withdraw(200.0, "123456");
        
        double balance = account.getBalance("123456");
        System.out.println("当前余额: " + balance);
        
        // 直接访问公共常量
        System.out.println("货币类型: " + BankAccount.CURRENCY);
        
        // 以下代码会编译错误（封装保护）
        // account.balance = 1000000; // 错误: balance是private
        // account.password = "hack"; // 错误: password是private
        
        System.out.println(account);
    }
}
```

```java
// 抽象类
abstract class Animal {
    protected String name;
    protected int age;
    
    public Animal(String name, int age) {
        this.name = name;
        this.age = age;
    }
    
    // 抽象方法，没有实现，必须由子类实现
    public abstract void makeSound();
    
    // 具体方法
    public void eat() {
        System.out.println(name + "在吃东西");
    }
    
    public void sleep() {
        System.out.println(name + "在睡觉");

// 接口
interface Flyable {
    // 常量（默认public static final）
    int MAX_ALTITUDE = 10000;
    
    // 抽象方法（默认public abstract）
    void fly();
    
    // 默认方法（Java 8+）
    default void takeOff() {
        System.out.println("准备起飞");
    }
    
    default void land() {
        System.out.println("准备降落");
    }
    
    // 静态方法（Java 8+）
    static void showMaxAltitude() {
        System.out.println("最大飞行高度: " + MAX_ALTITUDE + "米");
    }
}

interface Swimmable {
    void swim();
    
    default void dive() {
        System.out.println("潜入水中");
    }
}
```

### 方法重写和重载
```java
class Calculator {
    // 方法重载：同名方法，不同参数
    public int add(int a, int b) {
        return a + b;
    }
    
    public double add(double a, double b) {
        return a + b;
    }
    
    public int add(int a, int b, int c) {
        return a + b + c;
    }
    
    public String add(String a, String b) {
        return a + b;
    }
    
    // 可变参数方法
    public int add(int... numbers) {
        int sum = 0;
        for (int num : numbers) {
            sum += num;
        }
        return sum;
    }
}

class ScientificCalculator extends Calculator {
    // 方法重写：子类重写父类方法
    @Override
    public int add(int a, int b) {
        System.out.println("使用科学计算器进行加法");
        return super.add(a, b); // 调用父类方法
    }
    
    // 新增方法
    public double power(double base, double exponent) {
        return Math.pow(base, exponent);
    }
    
    public double squareRoot(double number) {
        return Math.sqrt(number);
    }
}

public class PolymorphismDemo {
    public static void main(String[] args) {
        Calculator calc = new Calculator();
        ScientificCalculator sciCalc = new ScientificCalculator();
        
        System.out.println("=== 方法重载演示 ===");
        System.out.println("整数加法: " + calc.add(5, 3));
        System.out.println("浮点数加法: " + calc.add(2.5, 3.7));
        System.out.println("三数加法: " + calc.add(1, 2, 3));
        System.out.println("字符串连接: " + calc.add("Hello", " World"));
        System.out.println("可变参数加法: " + calc.add(1, 2, 3, 4, 5));
        
        System.out.println("\n=== 方法重写演示 ===");
        System.out.println("普通计算器: " + calc.add(10, 20));
        System.out.println("科学计算器: " + sciCalc.add(10, 20));
        
        System.out.println("\n=== 多态演示 ===");
        Calculator polyCalc = new ScientificCalculator(); // 父类引用指向子类对象
        
        // 编译时类型是Calculator，运行时类型是ScientificCalculator
        System.out.println("多态加法: " + polyCalc.add(15, 25)); // 调用子类重写的方法
        
        // 以下代码会编译错误
        // polyCalc.power(2, 3); // 错误: Calculator没有power方法
        
        // 类型检查和转换
        if (polyCalc instanceof ScientificCalculator) {
            ScientificCalculator castCalc = (ScientificCalculator) polyCalc;
            System.out.println("平方根: " + castCalc.squareRoot(16));
            System.out.println("幂运算: " + castCalc.power(2, 3));
        }
        
        // 方法分派演示
        System.out.println("\n=== 方法分派 ===");
        testMethodDispatch(calc);
        testMethodDispatch(sciCalc);
        testMethodDispatch(polyCalc);
    }
    
    // 静态分派（重载）
    public static void testMethodDispatch(Calculator calculator) {
        // 编译时确定调用哪个重载方法
        System.out.println("计算结果: " + calculator.add(100, 200));
    }
}
```

### 对象类型转换
```java

public class TypeCastingDemo {
    public static void main(String[] args) {
        // 向上转型（隐式）
        Vehicle vehicle1 = new Car("丰田", 180, 4);
        Vehicle vehicle2 = new Bicycle("捷安特", 30, 21);
        
        System.out.println("=== 向上转型演示 ===");
        vehicle1.start();
        vehicle1.showInfo(); // 动态绑定，调用子类方法
        vehicle1.stop();
        
        System.out.println();
        
        vehicle2.start();
        vehicle2.showInfo(); // 动态绑定，调用子类方法
        vehicle2.stop();
        
        // 向下转型（需要显式转换）
        System.out.println("\n=== 向下转型演示 ===");
        
        if (vehicle1 instanceof Car) {
            Car car = (Car) vehicle1; // 向下转型
            car.drive(); // 可以调用Car特有方法
            car.showInfo();
        }
        
        if (vehicle2 instanceof Bicycle) {
            Bicycle bicycle = (Bicycle) vehicle2; // 向下转型
            bicycle.pedal(); // 可以调用Bicycle特有方法
            bicycle.showInfo();
        }
        
        // 错误的向下转型（会导致ClassCastException）
        try {
            Car wrongCar = (Car) vehicle2; // vehicle2实际上是Bicycle
            wrongCar.drive();
        } catch (ClassCastException e) {
            System.out.println("类型转换错误: " + e.getMessage());
        }
        
        // 安全的类型检查和转换
        System.out.println("\n=== 安全类型转换 ===");
        processVehicle(vehicle1);
        processVehicle(vehicle2);
        
        // 处理Vehicle数组
        Vehicle[] vehicles = {
            new Car("宝马", 220, 4),
            new Bicycle("美利达", 25, 18),
            new Car("本田", 160, 2),
            new Bicycle("凤凰", 20, 7)
        };
        
        System.out.println("\n=== 处理车辆数组 ===");
        for (Vehicle vehicle : vehicles) {
            processVehicle(vehicle);
        }
    }
    
    public static void processVehicle(Vehicle vehicle) {
        vehicle.start();
        
        // 根据实际类型执行不同操作
        if (vehicle instanceof Car) {
            Car car = (Car) vehicle;
            car.drive();
        } else if (vehicle instanceof Bicycle) {
            Bicycle bicycle = (Bicycle) vehicle;
            bicycle.pedal();
        }
        
        vehicle.showInfo();
        vehicle.stop();
        System.out.println("---");
    }
}
```

## 总结

本章详细介绍了Java面向对象编程的核心概念：

1. **类和对象**：类的定义、对象创建、构造方法重载
2. **封装**：访问控制修饰符、Getter/Setter方法、数据保护
3. **继承**：类继承、方法重写、super关键字、final关键字
4. **多态**：方法重载和重写、动态绑定、对象类型转换
5. **抽象类和接口**：抽象方法、默认方法、静态方法、多实现

面向对象编程的核心思想是通过对象来模拟现实世界，提高代码的可重用性、可维护性和可扩展性。在实际开发中，要合理运用这些概念，设计出良好的类层次结构。
