# Java异常处理

## 概述
异常处理是Java编程中处理运行时错误的重要机制。Java提供了完善的异常处理体系，帮助开发者编写健壮、可靠的程序。本章详细介绍了Java异常处理的概念、分类、处理方式和最佳实践。

### 检查异常 vs 非检查异常
```java
import java.io.*;

public class CheckedVsUnchecked {
    
    // 检查异常（Checked Exception）- 必须处理
    public static void readFile(String filename) throws IOException {
        FileReader reader = new FileReader(filename);
        // 读取文件操作...
        reader.close();
    }
    
    // 非检查异常（Unchecked Exception）- 运行时异常
    public static void divideNumbers(int a, int b) {
        if (b == 0) {
            throw new IllegalArgumentException("除数不能为零");
        }
        System.out.println("结果: " + (a / b));
    }
    
    public static void main(String[] args) {
        System.out.println("=== 检查异常 vs 非检查异常 ===");
        
        // 处理检查异常
        try {
            readFile("nonexistent.txt");
        } catch (IOException e) {
            System.out.println("文件读取异常: " + e.getMessage());
        }
        
        // 处理非检查异常
        try {
            divideNumbers(10, 0);
        } catch (IllegalArgumentException e) {
            System.out.println("数学运算异常: " + e.getMessage());
        }
        
        // 常见的检查异常
        demonstrateCheckedExceptions();
        
        // 常见的非检查异常
        demonstrateUncheckedExceptions();
    }
    
    public static void demonstrateCheckedExceptions() {
        System.out.println("\n=== 常见检查异常 ===");
        
        // 1. IOException
        try {
            File file = new File("test.txt");
            FileWriter writer = new FileWriter(file);
            writer.write("Hello World");
            writer.close();
        } catch (IOException e) {
            System.out.println("IO异常: " + e.getMessage());
        }
        
        // 2. ClassNotFoundException
        try {
            Class<?> clazz = Class.forName("com.example.NonExistentClass");
        } catch (ClassNotFoundException e) {
            System.out.println("类未找到异常: " + e.getMessage());
        }
        
        // 3. InterruptedException
        try {
            Thread.sleep(1000);
        } catch (InterruptedException e) {
            System.out.println("线程中断异常: " + e.getMessage());
            Thread.currentThread().interrupt(); // 恢复中断状态
        }
    }
    
    public static void demonstrateUncheckedExceptions() {
        System.out.println("\n=== 常见非检查异常 ===");
        
        // 1. NullPointerException
        try {
            String text = null;
            text.length();
        } catch (NullPointerException e) {
            System.out.println("空指针异常: " + e.getMessage());
        }
        
        // 2. ArrayIndexOutOfBoundsException
        try {
            int[] numbers = {1, 2, 3};
            System.out.println(numbers[5]);
        } catch (ArrayIndexOutOfBoundsException e) {
            System.out.println("数组越界异常: " + e.getMessage());
        }
        
        // 3. IllegalArgumentException
        try {
            if (System.currentTimeMillis() < 0) {
                throw new IllegalArgumentException("时间不能为负");
            }
        } catch (IllegalArgumentException e) {
            System.out.println("非法参数异常: " + e.getMessage());
        }
        
        // 4. ClassCastException
        try {
            Object obj = "Hello";
            Integer number = (Integer) obj;
        } catch (ClassCastException e) {
            System.out.println("类型转换异常: " + e.getMessage());
        }
    }
}
```

## 异常处理机制

### try-catch-finally块
```java
public class TryCatchFinally {
    
    public static void riskyOperation(int value) {
        System.out.println("\n执行操作，参数: " + value);
        
        try {
            if (value < 0) {
                throw new IllegalArgumentException("参数不能为负数");
            }
            if (value % 2 == 0) {
                throw new ArithmeticException("偶数参数不被支持");
            }
            
            System.out.println("操作成功完成");
            
        } catch (IllegalArgumentException e) {
            System.out.println("捕获IllegalArgumentException: " + e.getMessage());
            
        } catch (ArithmeticException e) {
            System.out.println("捕获ArithmeticException: " + e.getMessage());
            
        } finally {
            System.out.println("finally块总是执行");
        }
    }
    
    public static void multipleCatchBlocks() {
        System.out.println("\n=== 多个catch块 ===");
        
        try {
            // 可能抛出多种异常
            String[] data = {"1", "2", "three"};
            
            for (int i = 0; i <= data.length; i++) {
                int number = Integer.parseInt(data[i]);
                System.out.println("数字: " + number);
            }
            
        } catch (NumberFormatException e) {
            System.out.println("数字格式异常: " + e.getMessage());
            
        } catch (ArrayIndexOutOfBoundsException e) {
            System.out.println("数组越界异常: " + e.getMessage());
            
        } catch (Exception e) {
            System.out.println("其他异常: " + e.getMessage());
            
        } finally {
            System.out.println("清理资源完成");
        }
    }
    
    public static void tryWithResources() {
        System.out.println("\n=== try-with-resources ===");
        
        // Java 7+ 自动资源管理
        try (FileInputStream input = new FileInputStream("test.txt");
             FileOutputStream output = new FileOutputStream("copy.txt")) {
            
            byte[] buffer = new byte[1024];
            int bytesRead;
            while ((bytesRead = input.read(buffer)) != -1) {
                output.write(buffer, 0, bytesRead);
            }
            System.out.println("文件复制成功");
            
        } catch (IOException e) {
            System.out.println("IO异常: " + e.getMessage());
        }
        // 资源会自动关闭，不需要finally块
    }
    
    public static void main(String[] args) {
        System.out.println("=== 异常处理机制 ===");
        
        // 测试不同的参数
        riskyOperation(5);   // 正常情况
        riskyOperation(-1);  // 非法参数异常
        riskyOperation(4);   // 算术异常
        
        multipleCatchBlocks();
        tryWithResources();
        
        // finally块的特殊情况
        demonstrateFinallyBehavior();
    }
    
    public static void demonstrateFinallyBehavior() {
        System.out.println("\n=== finally块行为 ===");
        
        System.out.println("测试1 - 正常返回:");
        System.out.println("返回值: " + testFinally1());
        
        System.out.println("\n测试2 - 异常返回:");
        System.out.println("返回值: " + testFinally2());
        
        System.out.println("\n测试3 - finally中返回:");
        System.out.println("返回值: " + testFinally3());
    }
    
    public static String testFinally1() {
        try {
            return "try块返回值";
        } finally {
            System.out.println("finally块执行");
        }
    }
    
    public static String testFinally2() {
        try {
            throw new RuntimeException("测试异常");
        } catch (RuntimeException e) {
            System.out.println("捕获异常: " + e.getMessage());
            return "catch块返回值";
        } finally {
            System.out.println("finally块执行");
        }
    }
    
    public static String testFinally3() {
        try {
            return "try块返回值";
        } finally {
            System.out.println("finally块执行并返回");
            return "finally块返回值"; // 会覆盖try块的返回值
        }
    }
}
```

### 抛出异常
```java
public class ThrowExceptions {
    
    // 抛出检查异常
    public static void validateAge(int age) throws IllegalArgumentException {
        if (age < 0 || age > 150) {
            throw new IllegalArgumentException("年龄必须在0-150之间: " + age);
        }
        System.out.println("年龄验证通过: " + age);
    }
    
    // 抛出自定义异常
    public static void processPayment(double amount) throws PaymentException {
        if (amount <= 0) {
            throw new PaymentException("支付金额必须大于0: " + amount);
        }
        if (amount > 10000) {
            throw new PaymentException("单笔支付金额不能超过10000: " + amount);
        }
        System.out.println("支付成功: $" + amount);
    }
    
    // 异常链：包装原始异常
    public static void readConfigFile(String filename) throws ConfigException {
        try {
            // 模拟文件读取
            if (!filename.endsWith(".properties")) {
                throw new IOException("不支持的配置文件格式");
            }
            // 读取配置...
            
        } catch (IOException e) {
            // 包装原始异常，提供更多上下文信息
            throw new ConfigException("配置文件读取失败: " + filename, e);
        }
    }
    
    // 重新抛出异常
    public static void processData(String data) {
        try {
            if (data == null || data.trim().isEmpty()) {
                throw new IllegalArgumentException("数据不能为空");
            }
            // 处理数据...
            
        } catch (IllegalArgumentException e) {
            System.out.println("数据验证失败，记录日志");
            // 重新抛出异常
            throw e;
        }
    }
    
    public static void main(String[] args) {
        System.out.println("=== 抛出异常 ===");
        
        // 测试参数验证
        try {
            validateAge(25);   // 正常
            validateAge(-5);   // 异常
        } catch (IllegalArgumentException e) {
            System.out.println("年龄验证异常: " + e.getMessage());
        }
        
        // 测试自定义异常
        try {
            processPayment(500.0);   // 正常
            processPayment(0.0);    // 异常
        } catch (PaymentException e) {
            System.out.println("支付异常: " + e.getMessage());
            System.out.println("异常详情: " + e.getDetailedMessage());
        }
        
        // 测试异常链
        try {
            readConfigFile("config.xml");
        } catch (ConfigException e) {
            System.out.println("配置异常: " + e.getMessage());
            System.out.println("根本原因: " + e.getCause().getMessage());
            e.printStackTrace();
        }
        
        // 测试重新抛出
        try {
            processData(null);
        } catch (IllegalArgumentException e) {
            System.out.println("重新抛出的异常: " + e.getMessage());
        }
    }
}
```

## 异常处理最佳实践

### 异常处理原则
```java
import java.io.*;
import java.nio.file.*;
import java.util.logging.*;

public class ExceptionBestPractices {
    private static final Logger logger = Logger.getLogger(ExceptionBestPractices.class.getName());
    
    // 1. 具体异常优于通用异常
    public static void readFileBest(String filename) {
        try {
            Path path = Paths.get(filename);
            String content = Files.readString(path);
            System.out.println("文件内容: " + content);
            
        } catch (NoSuchFileException e) {
            System.err.println("文件不存在: " + filename);
            
        } catch (AccessDeniedException e) {
            System.err.println("没有文件访问权限: " + filename);
            
        } catch (IOException e) {
            System.err.println("文件读取错误: " + e.getMessage());
            
        }
    }
    
    // 2. 不要忽略异常
    public static void processDataWithLogging(String data) {
        try {
            // 处理数据
            int number = Integer.parseInt(data);
            System.out.println("处理后的数字: " + number);
            
        } catch (NumberFormatException e) {
            // 错误做法：空catch块
            // 正确做法：记录日志或采取适当措施
            logger.log(Level.WARNING, "数据格式错误: " + data, e);
            System.err.println("数据格式错误，使用默认值0");
            // 提供默认值或恢复策略
            int defaultValue = 0;
            System.out.println("使用默认值: " + defaultValue);
        }
    }
    
    // 3. 使用异常链保留根本原因
    public static void loadConfiguration() throws ConfigurationException {
        try {
            // 加载配置
            Properties props = new Properties();
            props.load(new FileInputStream("config.properties"));
            
        } catch (IOException e) {
            // 保留原始异常信息
            throw new ConfigurationException("配置加载失败", e);
        }
    }
    
    // 4. 尽早抛出异常，晚捕获异常
    public static void validateUserInput(String username, String password) {
        // 尽早验证
        if (username == null || username.trim().isEmpty()) {
            throw new IllegalArgumentException("用户名不能为空");
        }
        if (password == null || password.length() < 6) {
            throw new IllegalArgumentException("密码长度至少6位");
        }
        
        // 业务逻辑...
        System.out.println("用户输入验证通过");
    }
    
    // 5. 使用try-with-resources管理资源
    public static void copyFile(String source, String destination) {
        try (InputStream in = new FileInputStream(source);
             OutputStream out = new FileOutputStream(destination)) {
            
            byte[] buffer = new byte[1024];
            int length;
            while ((length = in.read(buffer)) > 0) {
                out.write(buffer, 0, length);
            }
            System.out.println("文件复制成功");
            
        } catch (IOException e) {
            System.err.println("文件复制失败: " + e.getMessage());
        }
    }
    
    // 6. 为可恢复异常提供恢复策略
    public static void connectToDatabase() {
        String[] urls = {"jdbc:mysql://primary", "jdbc:mysql://secondary"};
        
        for (String url : urls) {
            try {
                System.out.println("尝试连接: " + url);
                // 模拟数据库连接
                if (url.contains("primary")) {
                    throw new RuntimeException("主数据库连接失败");
                }
                System.out.println("连接成功: " + url);
                return; // 连接成功，退出循环
                
            } catch (RuntimeException e) {
                System.err.println("连接失败: " + url + " - " + e.getMessage());
                // 继续尝试下一个URL
            }
        }
        
        System.err.println("所有数据库连接尝试失败");
    }
    
    // 7. 使用断言进行内部验证
    public static void calculateDiscount(double price, double discountRate) {
        // 使用断言验证内部不变量
        assert price >= 0 : "价格不能为负数: " + price;
        assert discountRate >= 0 && discountRate <= 1 : "折扣率必须在0-1之间: " + discountRate;
        
        double discount = price * discountRate;
        double finalPrice = price - discount;
        
        System.out.println("原价: $" + price);
        System.out.println("折扣: $" + discount);
        System.out.println("最终价格: $" + finalPrice);
    }
    
    public static void main(String[] args) {
        System.out.println("=== 异常处理最佳实践 ===");
        
        // 测试具体异常处理
        readFileBest("nonexistent.txt");
        
        // 测试异常日志记录
        processDataWithLogging("abc");
        processDataWithLogging("123");
        
        // 测试异常链
        try {
            loadConfiguration();
        } catch (ConfigurationException e) {
            System.err.println("配置异常: " + e.getMessage());
            if (e.getCause() != null) {
                System.err.println("根本原因: " + e.getCause().getMessage());
            }
        }
        
        // 测试输入验证
        try {
            validateUserInput("", "short");
        } catch (IllegalArgumentException e) {
            System.err.println("输入验证失败: " + e.getMessage());
        }
        
        // 测试资源管理
        copyFile("source.txt", "destination.txt");
        
        // 测试恢复策略
        connectToDatabase();
        
        // 测试断言
        calculateDiscount(100.0, 0.2);
        // calculateDiscount(-50.0, 0.2); // 启用断言后会失败
    }
}
```

### 性能考虑
```java
public class ExceptionPerformance {
    
    // 异常创建的性能开销
    public static void testExceptionCreation() {
        System.out.println("=== 异常创建性能测试 ===");
        
        long startTime, endTime;
        int iterations = 100000;
        
        // 测试正常返回的性能
        startTime = System.nanoTime();
        for (int i = 0; i < iterations; i++) {
            normalReturn(i);
        }
        endTime = System.nanoTime();
        System.out.println("正常返回耗时: " + (endTime - startTime) / 1000000 + "ms");
        
        // 测试异常返回的性能
        startTime = System.nanoTime();
        for (int i = 0; i < iterations; i++) {
            try {
                exceptionReturn(i);
            } catch (IllegalArgumentException e) {
                // 忽略异常
            }
        }
        endTime = System.nanoTime();
        System.out.println("异常返回耗时: " + (endTime - startTime) / 1000000 + "ms");
        
        // 测试填充堆栈跟踪的性能
        startTime = System.nanoTime();
        for (int i = 0; i < iterations; i++) {
            try {
                exceptionWithStackTrace(i);
            } catch (IllegalArgumentException e) {
                // 忽略异常
            }
        }
        endTime = System.nanoTime();
        System.out.println("带堆栈跟踪的异常耗时: " + (endTime - startTime) / 1000000 + "ms");
    }
    
    public static int normalReturn(int value) {
        return value * 2;
    }
    
    public static int exceptionReturn(int value) {
        if (value < 0) {
            throw new IllegalArgumentException("值不能为负数");
        }
        return value * 2;
    }
    
    public static int exceptionWithStackTrace(int value) {
        if (value < 0) {
            IllegalArgumentException e = new IllegalArgumentException("值不能为负数");
            e.fillInStackTrace(); // 显式填充堆栈跟踪
            throw e;
        }
        return value * 2;
    }
    
    // 使用异常控制流 vs 条件判断
    public static void testControlFlow() {
        System.out.println("\n=== 控制流性能测试 ===");
        
        long startTime, endTime;
        int iterations = 100000;
        
        // 使用条件判断
        startTime = System.nanoTime();
        for (int i = 0; i < iterations; i++) {
            conditionalFlow(i);
        }
        endTime = System.nanoTime();
        System.out.println("条件判断耗时: " + (endTime - startTime) / 1000000 + "ms");
        
        // 使用异常控制流（不推荐）
        startTime = System.nanoTime();
        for (int i = 0; i < iterations; i++) {
            try {
                exceptionFlow(i);
            } catch (IllegalArgumentException e) {
                // 忽略异常
            }
        }
        endTime = System.nanoTime();
        System.out.println("异常控制流耗时: " + (endTime - startTime) / 1000000 + "ms");
    }
    
    public static void conditionalFlow(int value) {
        if (value >= 0) {
            // 正常处理
            int result = value * value;
        } else {
            // 错误处理
            System.err.println("无效值: " + value);
        }
    }
    
    public static void exceptionFlow(int value) {
        if (value < 0) {
            throw new IllegalArgumentException("无效值: " + value);
        }
        // 正常处理
        int result = value * value;
    }
    
    public static void main(String[] args) {
        System.out.println("=== 异常处理性能考虑 ===");
        
        testExceptionCreation();
        testControlFlow();
        
        System.out.println("\n性能优化建议:");
        System.out.println("1. 避免在正常流程中使用异常");
        System.out.println("2. 预检查条件，避免不必要的异常");
        System.out.println("3. 重用异常对象（如果安全）");
        System.out.println("4. 考虑使用断言进行内部验证");
        System.out.println("5. 对于频繁发生的错误，使用错误码或返回值");
    }
}
```

## 总结

Java异常处理是编写健壮程序的关键技术：

### 核心概念
1. **异常体系**：Throwable → Error/Exception → RuntimeException/其他检查异常
2. **检查异常**：必须处理，编译时检查
3. **非检查异常**：运行时异常，不强制处理

### 处理机制
1. **try-catch-finally**：基本异常处理结构
2. **try-with-resources**：自动资源管理

### 最佳实践
1. **具体异常**：捕获最具体的异常类型
2. **不要忽略**：正确处理或记录所有异常
3. **异常链**：保留原始异常信息
4. **资源管理**：使用try-with-resources
5. **性能考虑**：避免在正常流程中使用异常

通过合理使用异常处理机制，可以编写出更加健壮、可维护的Java程序。
