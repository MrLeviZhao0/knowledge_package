# Java集合框架

## 概述
Java集合框架（Java Collections Framework）是一组用于存储和操作对象的类和接口。它提供了高效的数据结构和算法，是Java编程中最重要的组成部分之一。

## 集合框架层次结构

### 核心接口
```java
import java.util.*;
import java.util.stream.Collectors;

public class CollectionHierarchy {
    public static void main(String[] args) {
        // Collection接口的主要子接口
        System.out.println("=== Collection接口层次 ===");
        
        // List - 有序集合，允许重复
        List<String> list = new ArrayList<>();
        list.add("Apple");
        list.add("Banana");
        list.add("Apple"); // 允许重复
        System.out.println("List: " + list);
        
        // Set - 无序集合，不允许重复
        Set<String> set = new HashSet<>();
        set.add("Apple");
        set.add("Banana");
        set.add("Apple"); // 重复元素不会被添加
        System.out.println("Set: " + set);
        
        // Queue - 队列，FIFO
        Queue<String> queue = new LinkedList<>();
        queue.offer("First");
        queue.offer("Second");
        queue.offer("Third");
        System.out.println("Queue: " + queue);
        System.out.println("出队: " + queue.poll());

        Map<String, Integer> map = new HashMap<>();
        map.put("Apple", 10);
        map.put("Banana", 5);
        map.put("Orange", 8);
```

### Set常用操作
```java
import java.util.*;

public class SetOperations {
    public static void main(String[] args) {
        Set<Integer> set1 = new HashSet<>();
        Set<Integer> set2 = new HashSet<>();
        
        // 添加元素
        set1.addAll(Arrays.asList(1, 2, 3, 4, 5));
        set2.addAll(Arrays.asList(4, 5, 6, 7, 8));
        
        System.out.println("Set1: " + set1);
        System.out.println("Set2: " + set2);
        
        // 并集
        Set<Integer> union = new HashSet<>(set1);
        union.addAll(set2);
        System.out.println("并集: " + union);
        
        // 交集
        Set<Integer> intersection = new HashSet<>(set1);
        intersection.retainAll(set2);
        System.out.println("交集: " + intersection);
        
        // 差集
        Set<Integer> difference = new HashSet<>(set1);
        difference.removeAll(set2);
        System.out.println("差集(Set1 - Set2): " + difference);
        
        // 对称差集
        Set<Integer> symmetricDifference = new HashSet<>(union);
        symmetricDifference.removeAll(intersection);
        System.out.println("对称差集: " + symmetricDifference);
        
        // 子集检查
        Set<Integer> subset = new HashSet<>(Arrays.asList(2, 3));
        System.out.println("子集检查: " + set1.containsAll(subset));
        
        // 自定义对象在Set中的使用
        System.out.println("\n=== 自定义对象Set ===");
        
        Set<Student> students = new HashSet<>();
        students.add(new Student("Alice", 20));
        students.add(new Student("Bob", 22));
        students.add(new Student("Alice", 20)); // 重复元素
        
        System.out.println("学生Set大小: " + students.size());
        students.forEach(System.out::println);
    }
}

class Student {
    private String name;
    private int age;
    
    public Student(String name, int age) {
        this.name = name;
        this.age = age;
    }
    
    // 重写equals和hashCode方法
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Student student = (Student) o;
        return age == student.age && Objects.equals(name, student.name);
    }
    
    @Override
    public int hashCode() {
        return Objects.hash(name, age);
    }
    
    @Override
    public String toString() {
        return "Student{name='" + name + "', age=" + age + "}";
    }
}
```

### HashMap vs TreeMap vs LinkedHashMap
```java
import java.util.*;

public class MapComparison {
    public static void main(String[] args) {
        // HashMap - 基于哈希表，无序
        Map<String, Integer> hashMap = new HashMap<>();
        
        // TreeMap - 基于红黑树，按键排序
        Map<String, Integer> treeMap = new TreeMap<>();
        
        // LinkedHashMap - 保持插入顺序
        Map<String, Integer> linkedHashMap = new LinkedHashMap<>();
        
        // 添加相同的键值对
        String[] keys = {"Banana", "Apple", "Orange", "Grape"};
        Integer[] values = {5, 10, 8, 3};
        
        for (int i = 0; i < keys.length; i++) {
            hashMap.put(keys[i], values[i]);
            treeMap.put(keys[i], values[i]);
            linkedHashMap.put(keys[i], values[i]);
        }
        
        System.out.println("HashMap (无序): " + hashMap);
        System.out.println("TreeMap (按键排序): " + treeMap);
        System.out.println("LinkedHashMap (插入顺序): " + linkedHashMap);
        
        // 自定义排序的TreeMap
        Map<String, Integer> reverseTreeMap = new TreeMap<>(Collections.reverseOrder());
        for (int i = 0; i < keys.length; i++) {
            reverseTreeMap.put(keys[i], values[i]);
        }
        System.out.println("逆序TreeMap: " + reverseTreeMap);
        
        // 访问顺序的LinkedHashMap
        LinkedHashMap<String, Integer> accessOrderMap = new LinkedHashMap<>(16, 0.75f, true);
        for (int i = 0; i < keys.length; i++) {
            accessOrderMap.put(keys[i], values[i]);
        }
        
        // 访问一些元素
        accessOrderMap.get("Apple");
        accessOrderMap.get("Banana");
        
        System.out.println("访问顺序LinkedHashMap: " + accessOrderMap);
    }
}
```

```java
import java.util.*;

public class MapOperations {
    public static void main(String[] args) {
        Map<String, Integer> inventory = new HashMap<>();
        
        // 添加键值对
        inventory.put("Apple", 50);
        inventory.put("Banana", 30);
        inventory.put("Orange", 25);
        inventory.put("Grape", 40);
        
        System.out.println("库存: " + inventory);
        
        // 获取值
        System.out.println("Apple库存: " + inventory.get("Apple"));
        System.out.println("不存在的键: " + inventory.get("Mango"));
        
        // 使用getOrDefault
        System.out.println("Mango库存(默认0): " + inventory.getOrDefault("Mango", 0));
        
        // 检查键和值
        System.out.println("包含Apple键: " + inventory.containsKey("Apple"));
        System.out.println("包含库存50: " + inventory.containsValue(50));
        
        // 修改值
        inventory.put("Apple", 45); // 替换现有值
        inventory.put("Mango", 20); // 添加新键值对
        System.out.println("修改后: " + inventory);
        
        // 使用putIfAbsent
        inventory.putIfAbsent("Apple", 60); // 不会替换，因为键已存在
        inventory.putIfAbsent("Peach", 15); // 添加新键值对
        System.out.println("putIfAbsent后: " + inventory);
        
        // 删除键值对
        inventory.remove("Grape");
        System.out.println("删除Grape后: " + inventory);
        
        // 条件删除
        inventory.remove("Apple", 50); // 不会删除，因为值不匹配
        inventory.remove("Apple", 45); // 会删除，因为值匹配
        System.out.println("条件删除后: " + inventory);
        
        // 替换值
        inventory.replace("Banana", 35);
        inventory.replace("Orange", 25, 30); // 条件替换
        System.out.println("替换后: " + inventory);
        
        // 遍历方式
        System.out.println("\n=== 遍历方式 ===");
        
        // 1. 遍历键
        System.out.println("键集合:");
        for (String key : inventory.keySet()) {
            System.out.print(key + " ");
        }
        System.out.println();
        
        // 2. 遍历值
        System.out.println("值集合:");
        for (Integer value : inventory.values()) {
            System.out.print(value + " ");
        }
        System.out.println();
        
        // 3. 遍历键值对
        System.out.println("键值对集合:");
        for (Map.Entry<String, Integer> entry : inventory.entrySet()) {
            System.out.println(entry.getKey() + " -> " + entry.getValue());
        }
        
        // 4. forEach方法
        System.out.println("forEach方法:");
        inventory.forEach((key, value) -> System.out.println(key + " : " + value));
        
        // 计算操作
        System.out.println("\n=== 计算操作 ===");
        
        // compute - 计算新值
        inventory.compute("Banana", (key, value) -> value + 10);
        System.out.println("compute后: " + inventory);
        
        // computeIfAbsent - 键不存在时计算
        inventory.computeIfAbsent("Lemon", key -> 5);
        System.out.println("computeIfAbsent后: " + inventory);
        
        // computeIfPresent - 键存在时计算
        inventory.computeIfPresent("Orange", (key, value) -> value * 2);
        System.out.println("computeIfPresent后: " + inventory);
        
        // merge - 合并值
        inventory.merge("Banana", 5, (oldValue, newValue) -> oldValue + newValue);
        System.out.println("merge后: " + inventory);
        
        // 统计信息
        System.out.println("\n=== 统计信息 ===");
        System.out.println("总库存量: " + inventory.values().stream().mapToInt(Integer::intValue).sum());
        System.out.println("平均库存: " + inventory.values().stream().mapToInt(Integer::intValue).average().orElse(0));
        System.out.println("最大库存: " + inventory.values().stream().mapToInt(Integer::intValue).max().orElse(0));
        System.out.println("最小库存: " + inventory.values().stream().mapToInt(Integer::intValue).min().orElse(0));
    }
}
```

```java
import java.util.*;

public class QueueStackDemo {
    public static void main(String[] args) {
        System.out.println("=== 队列操作 ===");
        
        // Queue - FIFO
        Queue<String> queue = new LinkedList<>();
        
        // 添加元素
        queue.offer("First");
        queue.offer("Second");
        queue.offer("Third");
        
        System.out.println("队列: " + queue);
        
        // 查看队首元素
        System.out.println("队首元素: " + queue.peek());
        
        // 出队
        while (!queue.isEmpty()) {
            System.out.println("出队: " + queue.poll());
        }
        
        System.out.println("\n=== 双端队列操作 ===");
        
        // Deque - 双端队列
        Deque<String> deque = new LinkedList<>();
        
        // 从头部和尾部添加
        deque.offerFirst("Front");
        deque.offerLast("Middle");
        deque.offerLast("End");
        
        System.out.println("双端队列: " + deque);
        
        // 从头部和尾部移除
        System.out.println("头部移除: " + deque.pollFirst());
        System.out.println("尾部移除: " + deque.pollLast());
        
        System.out.println("剩余队列: " + deque);
        
        System.out.println("\n=== 栈操作 ===");
        
        // 使用Deque作为栈
        Deque<String> stack = new ArrayDeque<>();
        
        // 压栈
        stack.push("First");
        stack.push("Second");
        stack.push("Third");
        
        System.out.println("栈: " + stack);
        
        // 弹栈
        while (!stack.isEmpty()) {
            System.out.println("弹栈: " + stack.pop());
        }
        
        System.out.println("\n=== 优先级队列 ===");
        
        // PriorityQueue - 优先级队列
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        
        pq.offer(5);
        pq.offer(1);
        pq.offer(3);
        pq.offer(2);
        pq.offer(4);
        
        System.out.println("优先级队列: " + pq);
        
        // 按优先级出队
        while (!pq.isEmpty()) {
            System.out.println("出队: " + pq.poll());
        }
        
        // 自定义优先级
        PriorityQueue<Student> studentQueue = new PriorityQueue<>(
            Comparator.comparing(Student::getAge).reversed()
        );
        
        studentQueue.offer(new Student("Alice", 20));
        studentQueue.offer(new Student("Bob", 22));
        studentQueue.offer(new Student("Charlie", 18));
        
        System.out.println("\n按年龄降序排列:");
        while (!studentQueue.isEmpty()) {
            System.out.println(studentQueue.poll());
        }
    }
}
```

```java
        );
        
        Set<String> immutableSet = Collections.unmodifiableSet(
            new HashSet<>(Arrays.asList("X", "Y", "Z"))
        );
        
        Map<String, Integer> immutableMap = Collections.unmodifiableMap(
            Map.of("One", 1, "Two", 2, "Three", 3)
        );
        
        System.out.println("不可变列表: " + immutableList);
        System.out.println("不可变集合: " + immutableSet);
        System.out.println("不可变映射: " + immutableMap);
        
        // Java 9+ 的工厂方法
        List<String> list = List.of("Java", "Python", "C++");
        Set<Integer> set = Set.of(1, 2, 3);
        Map<String, Integer> map = Map.of("A", 1, "B", 2, "C", 3);
        
        System.out.println("\n工厂方法创建的不可变集合:");
        System.out.println("List: " + list);
        System.out.println("Set: " + set);
        System.out.println("Map: " + map);
    }
}
```

## 总结

Java集合框架提供了丰富的数据结构和算法：

1. **List接口**：有序集合，允许重复
   - ArrayList：基于数组，随机访问快
   - LinkedList：基于链表，插入删除快
   - Vector：线程安全的ArrayList

2. **Set接口**：无序集合，不允许重复
   - HashSet：基于哈希表，查找快
   - TreeSet：基于红黑树，有序
   - LinkedHashSet：保持插入顺序

3. **Map接口**：键值对映射
   - HashMap：基于哈希表
   - TreeMap：基于红黑树，按键排序
   - LinkedHashMap：保持插入顺序

4. **Queue接口**：队列操作
   - LinkedList：可作队列和双端队列
   - PriorityQueue：优先级队列
   - ArrayDeque：高效的双端队列

5. **工具类**：Collections提供各种实用方法

选择正确的集合类型对于程序性能至关重要，需要根据具体需求选择最合适的实现。
