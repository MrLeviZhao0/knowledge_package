## 项目一：Android TV系统启动优化

### 项目背景
某智能电视厂商需要优化Android TV系统的启动时间，要求从按下电源键到显示主界面的时间控制在15秒以内。原有系统启动时间超过30秒，严重影响用户体验。

### 技术实现

```java
// 启动时间监控框架
public class BootTimeTracker {
    private static final String TAG = "BootTimeTracker";
    
    // 关键时间节点
    private static long sPowerOnTime;
    private static long sBootloaderStartTime;
    private static long sKernelStartTime;
    private static long sInitStartTime;
    private static long sZygoteStartTime;
    private static long sSystemServerStartTime;
    private static long sLauncherStartTime;
    
    // 阶段耗时统计
    private static final Map<String, Long> sStageDurations = new ConcurrentHashMap<>();
    
    // Bootloader阶段时间记录（通过内核命令行参数传递）
    public static native void recordBootloaderTime(long duration);
    
    // Kernel阶段时间记录
    public static void recordKernelTime(long startTime, long endTime) {
        long duration = endTime - startTime;
        sStageDurations.put("Kernel", duration);
        Log.i(TAG, "Kernel boot time: " + duration + "ms");
    }
    
    // Init阶段时间记录
    public static void recordInitTime(long startTime, long endTime) {
        long duration = endTime - startTime;
        sStageDurations.put("Init", duration);
        Log.i(TAG, "Init boot time: " + duration + "ms");
        
        // 记录到持久化存储
        storeBootTime("init_time", duration);
    }
    
    // Zygote阶段时间记录
    public static void recordZygoteTime(long startTime, long endTime) {
        long duration = endTime - startTime;
        sStageDurations.put("Zygote", duration);
        Log.i(TAG, "Zygote start time: " + duration + "ms");
    }
    
    // SystemServer阶段时间记录
    public static void recordSystemServerTime(long startTime, long endTime) {
        long duration = endTime - startTime;
        sStageDurations.put("SystemServer", duration);
        Log.i(TAG, "SystemServer start time: " + duration + "ms");
        
        // 详细记录各服务启动时间
        recordServiceStartTimes();
    }
    
    // 生成启动性能报告
    public static void generateBootReport() {
        StringBuilder report = new StringBuilder();
        report.append("=== Android TV Boot Performance Report ===\n");
        
        long totalTime = sLauncherStartTime - sPowerOnTime;
        report.append("Total boot time: ").append(totalTime).append("ms\n\n");
        
        report.append("Stage breakdown:\n");
        for (Map.Entry<String, Long> entry : sStageDurations.entrySet()) {
            report.append(entry.getKey()).append(": ").append(entry.getValue()).append("ms (");
            report.append(String.format("%.1f", (entry.getValue() * 100.0 / totalTime))).append("%)\n");
        }
        
        // 上报到云端分析系统
        uploadBootReport(report.toString());
        
        Log.i(TAG, report.toString());
    }
    
    private static void recordServiceStartTimes() {
        // 记录关键系统服务的启动时间
        String[] criticalServices = {
            "activity", "package", "window", "power", "display"
        };
        
        for (String service : criticalServices) {
            try {
                long startTime = getServiceStartTime(service);
                long endTime = getServiceReadyTime(service);
                sStageDurations.put("Service_" + service, endTime - startTime);
            } catch (Exception e) {
                Log.w(TAG, "Failed to record service time: " + service, e);
            }
        }
    }
}
```

#### 3. 服务启动依赖优化
```java
// 系统服务启动优化管理器
public class ServiceStartupOptimizer {
    private static final String TAG = "ServiceStartupOptimizer";
    
    // 服务依赖关系图
    private final Map<String, Set<String>> mDependencyGraph = new HashMap<>();
    
    // 服务启动时间记录
    private final Map<String, Long> mServiceStartTimes = new ConcurrentHashMap<>();
    
    public ServiceStartupOptimizer() {
        buildDependencyGraph();
    }
    
    private void buildDependencyGraph() {
        // 定义服务依赖关系
        mDependencyGraph.put("activity", new HashSet<>(Arrays.asList("package")));
        mDependencyGraph.put("package", new HashSet<>(Arrays.asList("installd")));
        mDependencyGraph.put("window", new HashSet<>(Arrays.asList("display", "activity")));
        mDependencyGraph.put("display", new HashSet<>(Arrays.asList("surfaceflinger")));
        mDependencyGraph.put("power", new HashSet<>()); // 无依赖
    }
    
    // 优化服务启动顺序
    public List<String> getOptimizedStartOrder() {
        List<String> startOrder = new ArrayList<>();
        Set<String> startedServices = new HashSet<>();
        
        // 拓扑排序确定启动顺序
        while (startedServices.size() < mDependencyGraph.size()) {
            boolean progress = false;
            
            for (String service : mDependencyGraph.keySet()) {
                if (!startedServices.contains(service)) {
                    Set<String> dependencies = mDependencyGraph.get(service);
                    
                    // 检查依赖是否都已启动
                    boolean canStart = true;
                    for (String dep : dependencies) {
                        if (!startedServices.contains(dep)) {
                            canStart = false;
                            break;
                        }
                    }
                    
                    if (canStart) {
                        startOrder.add(service);
                        startedServices.add(service);
                        progress = true;
                    }
                }
            }
            
            if (!progress) {
                // 检测到循环依赖
                Log.w(TAG, "Circular dependency detected in service startup");
                break;
            }
        }
        
        return startOrder;
    }
    
    // 并行启动无依赖关系的服务
    public void startIndependentServices() {
        ExecutorService executor = Executors.newFixedThreadPool(4);
        
        for (String service : mDependencyGraph.keySet()) {
            if (mDependencyGraph.get(service).isEmpty()) {
                // 无依赖服务可以并行启动
                executor.execute(() -> {
                    long startTime = System.currentTimeMillis();
                    startService(service);
                    long endTime = System.currentTimeMillis();
                    
                    mServiceStartTimes.put(service, endTime - startTime);
                    Log.i(TAG, "Service " + service + " started in " + (endTime - startTime) + "ms");
                });
            }
        }
        
        executor.shutdown();
        try {
            executor.awaitTermination(30, TimeUnit.SECONDS);
        } catch (InterruptedException e) {
            Log.e(TAG, "Service startup interrupted", e);
        }
    }
}
```

### 技术难点与解决方案

#### 难点1：服务依赖关系复杂
**问题**：系统服务间存在复杂的依赖关系，并行启动容易出错
**解决方案**：
- 建立完整的服务依赖关系图
- 使用拓扑排序确定启动顺序
- 实现依赖检测和循环依赖处理

#### 难点2：硬件初始化耗时
**问题**：电视硬件初始化（显示、音频、网络）耗时较长
**解决方案**：
- 实现硬件初始化的并行执行
- 优化驱动加载顺序
- 使用预初始化技术减少等待时间

#### 优化效果
- **启动时间**：从30秒优化到12秒（减少60%）
- **用户体验**：开机响应速度显著提升
- **系统稳定性**：优化后系统稳定性未受影响

## 项目二：车载系统快速启动方案

### 项目背景
某汽车厂商需要实现车载信息娱乐系统的快速启动，要求在车辆点火后5秒内显示导航界面。传统Android系统启动时间无法满足要求。

### 技术实现

#### 1. 休眠唤醒机制
```c
// 快速启动的休眠唤醒实现
#define SUSPEND_MAGIC 0x5USP3ND

typedef struct {
    uint32_t magic;
    uint32_t checksum;
    uint64_t suspend_time;
    uint32_t memory_pages;
    // 系统状态保存
    system_state_t system_state;
    // 进程状态保存
    process_state_t process_states[MAX_PROCESSES];
} suspend_header_t;

// 进入快速休眠
int enter_fast_suspend() {
    // 1. 冻结用户进程
    freeze_processes();
    
    // 2. 保存系统状态
    suspend_header_t header;
    header.magic = SUSPEND_MAGIC;
    header.suspend_time = get_current_time();
    
    // 3. 保存内存状态到特定区域
    if (save_memory_state(&header) != 0) {
        return -1;
    }
    
    // 4. 计算校验和
    header.checksum = calculate_checksum(&header);
    
    // 5. 写入休眠头信息
    write_suspend_header(&header);
    
    // 6. 进入低功耗模式
    enter_low_power_mode();
    
    return 0;
}

// 快速唤醒
int perform_fast_resume() {
    // 1. 读取休眠头信息
    suspend_header_t header;
    if (read_suspend_header(&header) != 0) {
        return -1; // 需要完整启动
    }
    
    // 2. 验证魔术字和校验和
    if (header.magic != SUSPEND_MAGIC || 
        header.checksum != calculate_checksum(&header)) {
        return -1; // 数据损坏，需要完整启动
    }
    
    // 3. 恢复内存状态
    if (restore_memory_state(&header) != 0) {
        return -1;
    }
    
    // 4. 恢复系统状态
    restore_system_state(&header.system_state);
    
    // 5. 解冻进程
    thaw_processes();
    
    // 6. 恢复运行
    resume_normal_operation();
    
    return 0;
}
```

```java
// 车载系统关键服务预启动管理器
public class CarCriticalServiceManager {
    private static final String TAG = "CarCriticalServiceManager";
    
    // 关键服务列表（必须快速启动的服务）
    private static final String[] CRITICAL_SERVICES = {
        "car_navigation",
        "car_audio", 
        "car_telephony",
        "car_bluetooth",
        "car_sensors"
    };
    
    // 预启动服务线程池
    private final ExecutorService mPreStartExecutor = 
        Executors.newFixedThreadPool(CRITICAL_SERVICES.length);
    
    public void preStartCriticalServices() {
        // 在系统启动早期预启动关键服务
        for (String service : CRITICAL_SERVICES) {
            mPreStartExecutor.execute(() -> {
                try {
                    preStartService(service);
                    Log.i(TAG, "Pre-started critical service: " + service);
                } catch (Exception e) {
                    Log.e(TAG, "Failed to pre-start service: " + service, e);
                }
            });
        }
    }
    
    private void preStartService(String serviceName) {
        // 服务预启动逻辑
        switch (serviceName) {
            case "car_navigation":
                preStartNavigationService();
                break;
            case "car_audio":
                preStartAudioService();
                break;
            case "car_telephony":
                preStartTelephonyService();
                break;
            // ... 其他服务
        }
    }
    
    private void preStartNavigationService() {
        // 预加载地图数据
        preloadMapData();
        
        // 初始化导航引擎
        initNavigationEngine();
        
        // 预计算常用路线
        precomputeCommonRoutes();
    }
    
    private void preStartAudioService() {
        // 预初始化音频设备

        // 预连接蓝牙音频
        preConnectBluetoothAudio();
    }
    
    // 等待所有预启动服务完成
    public boolean awaitPreStartCompletion(long timeout, TimeUnit unit) {
        mPreStartExecutor.shutdown();
        try {
            return mPreStartExecutor.awaitTermination(timeout, unit);
        } catch (InterruptedException e) {
            Log.e(TAG, "Pre-start await interrupted", e);
```

```java
// 资源预加载管理器
public class ResourcePreloader {
    private static final String TAG = "ResourcePreloader";
    
    // 预加载的资源分类
    private static final int PRIORITY_CRITICAL = 1;   // 关键资源（必须预加载）
    private static final int PRIORITY_IMPORTANT = 2;  // 重要资源（建议预加载）
    private static final int PRIORITY_NORMAL = 3;     // 普通资源（按需加载）
    
    private final Map<Integer, List<PreloadTask>> mPreloadTasks = new HashMap<>();
    
    public ResourcePreloader() {
        mPreloadTasks.put(PRIORITY_CRITICAL, new ArrayList<>());
        mPreloadTasks.put(PRIORITY_IMPORTANT, new ArrayList<>());
        mPreloadTasks.put(PRIORITY_NORMAL, new ArrayList<>());
        
        initializePreloadTasks();
    }
    
    private void initializePreloadTasks() {
        // 关键资源预加载任务
        addPreloadTask(PRIORITY_CRITICAL, new PreloadTask("导航地图数据") {
            @Override
            public void execute() {
                preloadNavigationMapData();
            }
        });
        
        addPreloadTask(PRIORITY_CRITICAL, new PreloadTask("系统字体") {
            @Override
            public void execute() {
                preloadSystemFonts();
            }
        });
        
        // 重要资源预加载任务
        addPreloadTask(PRIORITY_IMPORTANT, new PreloadTask("音频资源") {
            @Override
            public void execute() {
                preloadAudioResources();
            }
        });
        
        // 普通资源预加载任务
        addPreloadTask(PRIORITY_NORMAL, new PreloadTask("应用图标") {
            @Override
            public void execute() {
                preloadAppIcons();
            }
        });
    }
    
    public void startPreloading() {
        // 按优先级顺序预加载
        Executors.newSingleThreadExecutor().execute(() -> {
            preloadPriority(PRIORITY_CRITICAL);
            preloadPriority(PRIORITY_IMPORTANT);
            preloadPriority(PRIORITY_NORMAL);
        });
    }
    
    private void preloadPriority(int priority) {
        List<PreloadTask> tasks = mPreloadTasks.get(priority);
        if (tasks == null) return;
        
        for (PreloadTask task : tasks) {
            try {
                long startTime = System.currentTimeMillis();
                task.execute();
                long duration = System.currentTimeMillis() - startTime;
                
                Log.i(TAG, "Preload completed: " + task.getDescription() + 
                      " (" + duration + "ms)");
            } catch (Exception e) {
                Log.e(TAG, "Preload failed: " + task.getDescription(), e);
            }
        }
    }
}
```

```c
#define BOOT_MODE_NORMAL     0
#define BOOT_MODE_SAFE       1
#define BOOT_MODE_RECOVERY   2
#define BOOT_MODE_UPDATE     3

// 启动模式检测
int detect_boot_mode() {
    // 检查硬件按键
    if (is_recovery_key_pressed()) {
        return BOOT_MODE_RECOVERY;
    }
    
    if (is_safe_mode_key_pressed()) {
        return BOOT_MODE_SAFE;
    }

    if (strstr(cmdline, "androidboot.mode=update")) {
        return BOOT_MODE_UPDATE;
    }
    
    // 检查系统状态
    if (is_system_corrupted()) {
        return BOOT_MODE_SAFE;
    }
    
    return BOOT_MODE_NORMAL;
}

// 根据启动模式选择init.rc
```

#### 3. 系统健康检查
```java
// 系统健康检查服务
public class SystemHealthChecker {
    private static final String TAG = "SystemHealthChecker";
    
    // 健康检查项目
    private final List<HealthCheck> mHealthChecks = new ArrayList<>();
    
    public SystemHealthChecker() {
        initializeHealthChecks();
    }
    
    private void initializeHealthChecks() {
        // 文件系统检查
        mHealthChecks.add(new HealthCheck("Filesystem", this::checkFilesystem));
        
        // 系统服务检查
        mHealthChecks.add(new HealthCheck("SystemServices", this::checkSystemServices));
        
        // 内存检查
        mHealthChecks.add(new HealthCheck("Memory", this::checkMemory));
        
        // 存储检查
        mHealthChecks.add(new HealthCheck("Storage", this::checkStorage));
    }
    
    public HealthCheckResult performHealthCheck() {
        HealthCheckResult result = new HealthCheckResult();
        
        for (HealthCheck check : mHealthChecks) {
            try {
                boolean passed = check.performCheck();
                result.addCheckResult(check.getName(), passed);
                
                if (!passed) {
                    Log.w(TAG, "Health check failed: " + check.getName());
                }
            } catch (Exception e) {
                Log.e(TAG, "Health check error: " + check.getName(), e);
                result.addCheckResult(check.getName(), false);
            }
        }
        
        return result;
    }
    
    private boolean checkFilesystem() {
        // 检查文件系统完整性
        return checkFilesystemIntegrity("/system") &&
               checkFilesystemIntegrity("/data") &&
               checkFilesystemIntegrity("/cache");
    }
    
    private boolean checkSystemServices() {
        // 检查关键系统服务状态
        String[] criticalServices = {"activity", "package", "window"};
        
        for (String service : criticalServices) {
            if (!isServiceRunning(service)) {
                return false;
            }
        }
        
        return true;
    }
    
    private boolean checkMemory() {
        // 检查内存使用情况
        ActivityManager.MemoryInfo memoryInfo = new ActivityManager.MemoryInfo();
        ActivityManager activityManager = (ActivityManager) 
            getSystemService(Context.ACTIVITY_SERVICE);
        
        activityManager.getMemoryInfo(memoryInfo);
        
        // 确保有足够可用内存
        return memoryInfo.availMem > MIN_AVAILABLE_MEMORY &&
               !memoryInfo.lowMemory;
    }
    
    // 启动时自动健康检查
    public void performStartupHealthCheck() {
        HealthCheckResult result = performHealthCheck();
        
        if (!result.isHealthy()) {
            Log.w(TAG, "System health check failed, suggesting safe mode");
            
            // 提示用户进入安全模式
            suggestSafeMode();
        }
    }
}
```
